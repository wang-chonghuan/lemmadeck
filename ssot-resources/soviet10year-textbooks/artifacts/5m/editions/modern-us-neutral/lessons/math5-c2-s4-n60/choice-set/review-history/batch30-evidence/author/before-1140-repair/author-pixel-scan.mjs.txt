];
const decoded = new Map();
const scans = [];
for (const [number, label, axis, fixed, lo, hi, threshold] of requests) {
  if (!decoded.has(number)) {
    const filename = path.join(out, `author-fig-${number}-native.png`);
    const bytes = fs.readFileSync(filename);
    decoded.set(number, {
      png: PNG.sync.read(bytes),
      screenshot_sha256: crypto.createHash("sha256").update(bytes).digest("hex"),
    });
  }
  const { png, screenshot_sha256 } = decoded.get(number);
  const runs = [];
  let start = null;
  for (let t = lo; t <= hi + 1; t++) {
    const x = axis === "row" ? t : fixed;
    const y = axis === "column" ? t : fixed;
    const offset = 4 * (y * png.width + x);
    const ink = t <= hi && png.data[offset + 3] > 0 &&
      Math.max(...png.data.subarray(offset, offset + 3)) < threshold;
    if (ink && start === null) start = t;
    if (!ink && start !== null) {
      runs.push({ first: start, last: t - 1, midpoint: (start + t - 1) / 2 });
      start = null;
    }
  }
  scans.push({ figure_id: `fig-${number}`, screenshot_sha256, label, axis, fixed, range: [lo, hi], threshold, runs });
}
const report = {
  method: "Decode existing native screenshots with the PNG decoder bundled in the existing Playwright. Scan raster RGB thresholds; do not read SVG geometry or create/reopen an image in the model context.",
  context_image_loads: 5,
  scans,
};
fs.writeFileSync(path.join(out, "author-pixel-record.json"), JSON.stringify(report, null, 2) + "\n", { flag: "wx" });
