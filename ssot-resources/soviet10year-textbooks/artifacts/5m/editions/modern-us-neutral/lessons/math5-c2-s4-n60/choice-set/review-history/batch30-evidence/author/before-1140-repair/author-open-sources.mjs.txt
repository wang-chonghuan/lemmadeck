import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import { createRequire } from "node:module";
import { pathToFileURL } from "node:url";

const root = "/Users/yong/.codex/worktrees/d9b1/lemmadeck";
const out = path.join(root, ".intentfold/tickets/STEMROBIN-182/tmp/5m-n60");
const require = createRequire(path.join(root, ".agents/skills/package.json"));
const { chromium } = require("playwright-core");
const seed = JSON.parse(fs.readFileSync(path.join(out, "seed.json"), "utf8"));
const reportPath = path.join(out, "author-render-record.json");
if (fs.existsSync(reportPath)) throw new Error("Refusing to overwrite render record");
const report = {
  command: ["node", ".intentfold/tickets/STEMROBIN-182/tmp/5m-n60/author-open-sources.mjs"],
  workdir: root,
  started: new Date().toISOString(),
  network: "No network; file URLs only; http/https requests aborted.",
  headless: true,
  isolated_context: true,
  browser_closed: false,
  figures: [],
  errors: [],
};
let browser;
try {
  browser = await chromium.launch({ headless: true });
  const context = await browser.newContext({ deviceScaleFactor: 1 });
  await context.route(/^https?:/, route => route.abort());
  const page = await context.newPage();
  for (const figure of seed.figures) {
    const asset = figure.assets[0];
    const absolute = path.join(root, asset.path);
    const sha256 = crypto.createHash("sha256").update(fs.readFileSync(absolute)).digest("hex");
    if (sha256 !== asset.sha256) throw new Error(`Frozen asset changed: ${figure.id}`);
    const svg = fs.readFileSync(absolute, "utf8");
    const tag = svg.match(/<svg\b[^>]*>/)[0];
    const width = Number(tag.match(/\bwidth="(\d+)"/)[1]);
    const height = Number(tag.match(/\bheight="(\d+)"/)[1]);
    await page.setViewportSize({ width, height });
    await page.goto(pathToFileURL(absolute).href);
    await page.locator("svg").waitFor({ state: "visible" });
    const screenshot = path.join(out, `author-${figure.id}-native.png`);
    if (fs.existsSync(screenshot)) throw new Error(`Refusing to overwrite ${screenshot}`);
    await page.locator("svg").screenshot({ path: screenshot });
    const bounds = await page.locator("svg").boundingBox();
    report.figures.push({
      figure_id: figure.id,
      asset_path: asset.path,
      asset_sha256: sha256,
      native_size: { width, height },
      rendered_bounds: bounds,
      screenshot_path: path.relative(root, screenshot),
      screenshot_sha256: crypto.createHash("sha256").update(fs.readFileSync(screenshot)).digest("hex"),
    });
  }
  await context.close();
} catch (error) {
  report.errors.push(String(error.stack || error));
  process.exitCode = 1;
} finally {
  if (browser) {
    await browser.close();
    report.browser_closed = true;
  }
  report.finished = new Date().toISOString();
  fs.writeFileSync(reportPath, JSON.stringify(report, null, 2) + "\n", { flag: "wx" });
}
console.log(JSON.stringify(report, null, 2));
