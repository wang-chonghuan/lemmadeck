const reviews = []
for (const [group, suffix, file, loads] of reviewSpecs) {
  const source = filename(`reviewer-batch31-c3-s3-${suffix}`, file)
  const original = read(source)
  const set = sets[group]
  const montageRows = original.montages ?? [original.montage]
  for (const montage of montageRows) {
    const position = montage.index ?? montage.position ?? montage.zero_based_position
      ?? montage.array_position ?? montage.array_index
    assert(Number.isInteger(position), suffix)
    assert(!set.observed_montages.has(position), `${suffix} duplicate montage`)
    set.observed_montages.add(position)
    const inventory = set.inventory.montages[position]
    assert.equal(montage.path, inventory.path)
    assert.equal(montage.sha256, inventory.sha256)
    assert.equal(hash(montage.path), montage.sha256)
    const members = montage.members ?? original.members.filter(member =>
      member.montage_array_position === position || original.montage)
    assert.equal(members.length, inventory.members.length, `${suffix} members`)
    for (const member of members) {
      const id = member.capture_id ?? member.id
      const expected = inventory.members.find(row => row.capture_id === id)
      assert(expected, `${suffix} capture ${id}`)
      assert.equal(member.path, expected.path)
      assert.equal(member.sha256, expected.sha256)
      assert.equal(hash(member.path), member.sha256)
      assert(!set.observations.has(id), `${group} duplicate member ${id}`)
      const status = member.status ?? (member.ranges.partial.length ? 'partial' : 'pass')
      assert(['pass', 'partial', 'complete'].includes(status), `${suffix} ${status}`)
      set.observations.set(id, {
        group, capture_id: id, original_status: status,
        observation_file: bind(source), montage_position: position,
        observation: member,
      })
    }
  }
  assert.equal(montageRows.length, loads, suffix)
  reviews.push({ group, ...bind(source), actual_image_loads: loads, original_observation: original })
}
