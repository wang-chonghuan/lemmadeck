def file_hash(file_path):
    return hashlib.sha256(file_path.read_bytes()).hexdigest()


def canonical_hash(value):
    return hashlib.sha256(json.dumps(
        value, sort_keys=True, ensure_ascii=True, separators=(",", ":")
    ).encode("utf8")).hexdigest()


def read_json(file_path):
    return json.loads(file_path.read_bytes())

def polygon_area(points):
    pairs = [[Decimal(coordinate) for coordinate in point] for point in points]
    return abs(sum(
        point[0] * pairs[(index + 1) % len(pairs)][1]
        - pairs[(index + 1) % len(pairs)][0] * point[1]
        for index, point in enumerate(pairs)
    )) / 2


# This tolerance checks arithmetic consistency of native asset coordinates, not physical measurement precision.
arithmetic_tolerance = Decimal("1e-10")
plot = next(item for item in observations if item["figure_id"] == "fig-195")
geometry = plot["geometry_native_px"]
bar = Decimal(geometry["bar_endpoints"][1][0]) - Decimal(geometry["bar_endpoints"][0][0])
net_are = (polygon_area(geometry["outer_vertices"])
           - polygon_area(geometry["blank_rectangle_vertices"])) / bar**2
assert abs(net_are - Decimal(plot["area_data_approx"]["shaded_ground_are"])) < arithmetic_tolerance
crop = next(item for item in observations if item["figure_id"] == "fig-196")
geometry = crop["geometry_native_px"]
bar = Decimal(geometry["bar_endpoints"][1][0]) - Decimal(geometry["bar_endpoints"][0][0])
for name in ["potatoes", "cucumbers", "cabbage"]:
    area_are = polygon_area(geometry[f"{name}_vertices"]) / bar**2
    recorded = Decimal(crop["area_data_approx"][f"{name}_ground_are"])
    assert abs(area_are - recorded) < arithmetic_tolerance
