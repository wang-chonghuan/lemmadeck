import hashlib
import json
import pathlib
import shutil


ROOT = pathlib.Path("/Users/yong/.codex/worktrees/d9b1/lemmadeck")
TICKET_TMP = ROOT / ".intentfold/tickets/STEMROBIN-182/tmp"


for batch in ("5m-n56", "5m-n57"):
    source = TICKET_TMP / batch / "seed.json"
    destination = TICKET_TMP / ("reviewer-" + batch)
    raw = source.read_bytes()
    seed = json.loads(raw)
    canonical = json.dumps(
        seed, sort_keys=True, ensure_ascii=True, separators=(",", ":")
    ).encode("utf-8")
    image_references = []
    for figure in seed.get("figures", []):
        image_references.append({"kind": "figure", "figure": figure})
    for exercise in seed.get("exercises", []):
        for figure_id in exercise.get("figure_ids", []):
            image_references.append(
                {"kind": "exercise", "source_number": exercise["number"],
                 "figure_id": figure_id}
            )
    for figure_id in seed.get("source", {}).get("concept_figure_ids", []):
        image_references.append({"kind": "concept", "figure_id": figure_id})
    for section in ("exercises", "concepts"):
        for item in seed.get(section, []):
            text = item.get("text", "")
            if "![" in text or "<img" in text.lower() or "data:image/" in text:
                image_references.append(
                    {"kind": "embedded_reference", "section": section,
                     "identifier": item.get("number", item.get("block")),
                     "text": text}
                )
    snapshot = destination / "input-snapshot"
    snapshot.mkdir()
    frozen = snapshot / "seed.json"
    shutil.copyfile(source, frozen)
    frozen_raw = frozen.read_bytes()
    if frozen_raw != raw:
        raise RuntimeError("Source changed while copying: " + str(source))
    manifest = {
        "schema": "s10y-choice-source-input-freeze-v1",
        "batch": batch,
        "authority_path": str(source.relative_to(ROOT)),
        "snapshot_path": str(frozen.relative_to(ROOT)),
        "canonicalization": (
            'json.dumps(value, sort_keys=True, ensure_ascii=True, '
            'separators=(",", ":")).encode("utf-8")'
        ),
        "seed_sha256": hashlib.sha256(canonical).hexdigest(),
        "seed_bytes_sha256": hashlib.sha256(raw).hexdigest(),
        "seed_byte_count": len(raw),
        "source_count": len(seed.get("exercises", [])),
        "concept_count": len(seed.get("concepts", [])),
        "image_reference_count": len(image_references),
        "image_references": image_references,
        "source_phase_status": (
            "blocked_image_reference" if image_references else "input_frozen"
        ),
    }
    with (snapshot / "freeze.json").open("x", encoding="utf-8") as stream:
        json.dump(manifest, stream, ensure_ascii=True, indent=2)
        stream.write("\n")
    print(json.dumps(manifest, ensure_ascii=True, indent=2))
