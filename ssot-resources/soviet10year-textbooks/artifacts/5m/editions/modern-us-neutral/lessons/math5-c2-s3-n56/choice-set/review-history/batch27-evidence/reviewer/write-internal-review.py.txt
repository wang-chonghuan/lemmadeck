def hashes(raw):
    canonical = json.dumps(
        json.loads(raw), sort_keys=True, ensure_ascii=True, separators=(",", ":")
    ).encode("utf-8")
    return {
        "canonical_sha256": hashlib.sha256(canonical).hexdigest(),
        "bytes_sha256": hashlib.sha256(raw).hexdigest(),
        "byte_count": len(raw),
    }


def read_json(path):
    raw = path.read_bytes()
    return json.loads(raw), raw
