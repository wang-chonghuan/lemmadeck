function collect(reviews, files, expectedInventory) {
  const montages = new Map()
  const members = new Map()
  reviews.forEach((review, batch) => {
    assert(review.image_count <= 8)
    assert.equal(review.image_count, review.montages.length)
    for (const montage of review.montages) {
      const expected = expectedInventory.montages[montage.index]
      assert(expected && !montages.has(montage.index))
      assert.equal(montage.actualopened, true)
      assert.equal(montage.path, expected.path)
      assert.equal(montage.sha256, expected.sha256)
      assert.equal(hash(montage.path), montage.sha256)
      assert.equal(montage.members.length, expected.members.length)
      montages.set(montage.index, { ...montage, actual_review: bind(files[batch]) })
      for (const member of montage.members) {
        const original = expected.members.find(row => row.capture_id === member.capture_id)
        assert(original && !members.has(member.capture_id))
        assert.equal(member.path, original.path)
        assert.equal(member.sha256, original.sha256)
        assert.equal(hash(member.path), member.sha256)
        assert(member.visible_evidence && Object.keys(member.visible_evidence).length)
        members.set(member.capture_id, { ...member, actual_review: bind(files[batch]) })
      }
    }
  })
  return { montages, members }
}
