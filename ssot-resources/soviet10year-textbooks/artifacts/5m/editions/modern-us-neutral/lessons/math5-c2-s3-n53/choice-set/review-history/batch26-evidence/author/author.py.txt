def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(",", ":")).encode()).hexdigest()


def file_hash(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def rational(value):
    value = F(value)
    return str(value.numerator) if value.denominator == 1 else f"{value.numerator}/{value.denominator}"


def latex(value):
    value = F(value)
    if value.denominator == 1:
        return str(value.numerator)
    sign = "-" if value < 0 else ""
    return sign + r"\frac{" + str(abs(value.numerator)) + "}{" + str(value.denominator) + "}"


def m(integer, numerator, denominator):
    return F(integer) + (-1 if integer < 0 else 1) * F(numerator, denominator)


def vals(number, indices=None):
    rows = solutions[str(number)]["subtasks"]
    if indices is not None:
        rows = [rows[index - 1] for index in indices]
    return [F(row["result"]) for row in rows]


def table(expressions):
    return "| 项目 | 算式 |\n|---|---|\n" + "\n".join(
        f"| {i + 1} | ${expression}$ |" for i, expression in enumerate(expressions))


def result_table(values, labels=None, unit=""):
    labels = labels or [str(i + 1) for i in range(len(values))]
    return "| 项目 | 结果 |\n|---|---|\n" + "\n".join(
        f"| {label} | ${latex(value)}$" + (f" {unit}" if unit else "") + " |"
        for label, value in zip(labels, values))

def add_question(identifier, numbers, stem, candidates, task, unit_reason, limitations=None,
                 figures=None, mode="single"):
    numbers = [str(number) for number in numbers]
    figures = figures or []
    candidate_contents = [item["text"] for item in candidates]
    if len(set(candidate_contents)) != len(candidate_contents):
        raise RuntimeError("Duplicate visible candidate")
    rng.shuffle(candidates)
    options = []
    rationales = []
    correct = []
    for label, candidate in zip("ABCD", candidates):
        truth = candidate["truth"]
        options.append({"id": label, "text": candidate["text"], "figure_ids": []})
        rationales.append({"option_id": label, "truth": "correct" if truth else "incorrect",
                           "derivation": candidate["derivation"],
                           "misconception": None if truth else candidate["procedure"]})
        if truth:
            correct.append(label)
        else:
            program_records.append({"question_id": identifier, "option_id": label,
                                    "procedure": candidate["procedure"],
                                    "actual_result": candidate.get("actual_result", candidate["text"]),
                                    "visible_candidate": candidate["text"],
                                    "execution": candidate.get("execution", "explicit symbolic/geometric calculation in derivation"),
                                    "false_under_current_stem": True})
    if not correct or (mode == "single" and len(correct) != 1):
        raise RuntimeError("Invalid native true set")
    used = set(figures)
    for number in numbers:
        used.update(next(item["figure_ids"] for item in source["exercises"] if item["number"] == number))
    source_solution = "\n".join(f"{number}: {solutions[number]['independent_solution']}" for number in numbers)
    question = {
        "id": identifier, "source_numbers": numbers,
        "view": {"mode": mode, "stem": stem, "figure_ids": figures, "options": options},
        "intent": {"task": task, "source_solution": source_solution, "unit_reason": unit_reason,
                   "concept_blocks": [0], "limitations": limitations or [],
                   "figure_observations": [{"figure_id": figure, "observation": source_figure_observations[figure]}
                                           for figure in sorted(used)]},
        "key": {"correct": correct, "rationales": rationales},
    }
    questions.append(question)
    unit_records.append({"question_id": identifier, "source_numbers": numbers, "unit_reason": unit_reason,
                         "task": task, "limitations": limitations or []})


def numeric(identifier, number, stem, correct, wrong1, procedure1, wrong2, procedure2,
            unit_reason, labels=None, unit="", limitations=None, figures=None, task=None):
    correct, wrong1, wrong2 = [[F(value) for value in row] for row in (correct, wrong1, wrong2)]
    if len(correct) != len(wrong1) or len(correct) != len(wrong2):
        raise RuntimeError("Incomplete numeric candidate")
    if correct == wrong1 or correct == wrong2 or wrong1 == wrong2:
        raise RuntimeError("A false program coincides with a true or duplicate complete output")
    numbers = number if isinstance(number, list) else [number]
    full_solution = "\n".join(solutions[str(item)]["independent_solution"] for item in numbers)
    candidates = []
    for values, truth, procedure in [(correct, True, None), (wrong1, False, procedure1), (wrong2, False, procedure2)]:
        exact = [rational(value) for value in values]
        candidates.append({
            "text": result_table(values, labels, unit), "truth": truth, "procedure": procedure,
            "actual_result": exact,
            "execution": "Python Fraction evaluates the explicitly authored operations; exact materialized outputs, not sampled nearby numbers.",
            "derivation": (full_solution if truth else procedure) + " Exact ordered output: " + ", ".join(exact) + ".",
        })
    add_question(identifier, numbers, stem, candidates, task or "Determine every complete result in the displayed order.",
                 unit_reason, limitations, figures)
