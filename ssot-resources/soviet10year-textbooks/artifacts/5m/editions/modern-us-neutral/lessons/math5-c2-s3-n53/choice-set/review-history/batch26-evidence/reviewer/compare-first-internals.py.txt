def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(",", ":")).encode()


def load(name):

def save(name, value):
    with (OWN / name).open("x", encoding="utf-8") as handle:
        json.dump(value, handle, indent=2, ensure_ascii=False)
        handle.write("\n")


def normal(value):
    if isinstance(value, F):
        return str(value)
    if isinstance(value, (list, tuple)):
        return [normal(item) for item in value]
    if isinstance(value, dict):
        return {key: normal(item) for key, item in value.items()}
    return value

def S(number, label):

def improper_bug(a, b):
    return F(a.numerator - b.numerator, a.denominator)


def fraction_difference_bug(a, b):
    return F(a.numerator - b.numerator, a.denominator - b.denominator)


def borrow_bug(w1, n1, d1, w2, n2, d2):
    return F(w1 - w2 + 1) + F(n1, d1) - F(n2, d2)


def printed_fraction_sum_bug(w1, n1, d1, w2, n2, d2):
    return F(w1 + w2) + F(n1 + n2, d1 + d2)


def discard_carry(w1, n1, d1, w2, n2, d2):
    fraction_sum = F(n1, d1) + F(n2, d2)
    return F(w1 + w2) + fraction_sum - fraction_sum.numerator // fraction_sum.denominator
