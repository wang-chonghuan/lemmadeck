def sha(data):
    return hashlib.sha256(data).hexdigest()


def load(name):
    return json.loads((OWN / name).read_bytes())


def save(name, value):
    with (OWN / name).open("x", encoding="utf-8") as handle:
        json.dump(value, handle, indent=2, ensure_ascii=False)
        handle.write("\n")

def parse_value(cell):
    match = re.fullmatch(r"\$(.+)\$(?:\s+(.+))?", cell.strip())
    assert match, f"Unexpected visible result cell: {cell!r}"
    expression, units = match.groups()
    fraction = re.fullmatch(r"(-?)\\frac\{(\d+)\}\{(\d+)\}", expression)
    if fraction:
        sign, numerator, denominator = fraction.groups()
        value = F(int(numerator), int(denominator))
        if sign:
            value = -value
    else:
        assert re.fullmatch(r"-?\d+(?:\.\d+)?", expression), expression
        value = F(expression)
    return value, units


def table_rows(text):
    rows = [line.strip() for line in text.splitlines() if line.strip()]
    assert len(rows) >= 3 and rows[0] == "| 项目 | 结果 |" and rows[1] == "|---|---|"
    parsed = []
    for row in rows[2:]:
        cells = row.split("|")
        assert len(cells) == 4 and not cells[0].strip() and not cells[-1].strip()
        value, units = parse_value(cells[2])
        parsed.append({"label": cells[1].strip(), "value": value, "units": units})
    return parsed
