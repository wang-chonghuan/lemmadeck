def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(",", ":")).encode()


def new_json(name, value):
    with (OWN / name).open("x", encoding="utf-8") as handle:
        json.dump(value, handle, indent=2, ensure_ascii=False)
        handle.write("\n")


def mixed(whole, numerator, denominator):
    return F(whole) + F(numerator, denominator)


def display(value):
    value = F(value)
    sign = "-" if value < 0 else ""
    numerator, denominator = abs(value.numerator), value.denominator
    whole, remainder = divmod(numerator, denominator)
    if not remainder:
        return sign + str(whole)
    fraction = f"{remainder}/{denominator}"
    return sign + (f"({whole}+{fraction})" if whole else fraction)


def exact_decimal(value):
    value = F(value)
    denominator = value.denominator
    for prime in (2, 5):
        while denominator % prime == 0:
            denominator //= prime
    if denominator != 1:
        return None
    sign = "-" if value < 0 else ""
    numerator, denominator = abs(value.numerator), value.denominator
    whole, remainder = divmod(numerator, denominator)
    digits = []
    while remainder:
        digit, remainder = divmod(remainder * 10, denominator)
        digits.append(str(digit))
    return sign + str(whole) + ("." + "".join(digits) if digits else "")


def result(label, expression, value, working, *, units=None, check=None):
    value = F(value)
    entry = {
        "subtask": label,
        "source_expression_or_target": expression,
        "exact_result": str(value),
        "signed_mixed_result": display(value),
        "finite_decimal_result": exact_decimal(value),
        "working": working,
        "units": units,
    }
    if check:
        entry["verification"] = check
    return entry
