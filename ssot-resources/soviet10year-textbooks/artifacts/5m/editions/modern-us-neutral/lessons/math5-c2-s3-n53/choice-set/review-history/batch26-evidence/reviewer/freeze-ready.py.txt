import hashlib
import json
from pathlib import Path


OWN = Path(__file__).resolve().parent


def sha(data):
    return hashlib.sha256(data).hexdigest()


def load(name):
    return json.loads((OWN / name).read_bytes())


with (OWN / "ready-run.started.json").open("x") as handle:
    json.dump({"once_guard": True, "scope": "freeze source-phase receipts only"}, handle)
    handle.write("\n")

freeze = load("input-freeze.json")
captures = load("capture-receipt.json")
solutions = load("source-solutions.json")
observations = load("source-image-observations.json")
solution_receipt = load("solution-receipt.json")
for item in freeze["allowed_inputs"]:
    assert sha(Path(item["path"]).read_bytes()) == item["sha256"]
source_path = Path(freeze["allowed_inputs"][0]["path"])
assert source_path.read_bytes() == (OWN / "source.frozen.json").read_bytes()
seed = json.loads(source_path.read_bytes())
assert [item["source_number"] for item in solutions["source_reviews"]] == [str(n) for n in range(1006, 1033)]
for original, solved in zip(seed["exercises"], solutions["source_reviews"]):
    assert original["number"] == solved["source_number"]
    assert original["text"] == solved["source_text"]
    assert original["figure_ids"] == solved["source_figure_ids"]
    assert solved["independent_solution"] and solved["subtasks"] and solved["domain_and_conventions"]
assert solutions["source_image_observations_sha256"] == sha((OWN / "source-image-observations.json").read_bytes())
assert solution_receipt["output_sha256"] == sha((OWN / "source-solutions.json").read_bytes())
assert solutions["solver_script_sha256"] == sha((OWN / "solve-source.py").read_bytes())
assert captures["script_sha256"] == sha((OWN / "capture-source.mjs").read_bytes())
assert captures["browser_closed"] and captures["headless"]
for item in captures["captures"]:
    assert sha(Path(item["path"]).read_bytes()) == item["sha256"]
declared = {(item["path"], item["sha256"]) for item in freeze["allowed_inputs"] if "figure_id" in item}
observed = {(item["asset_path"], item["asset_sha256"]) for item in observations["asset_observations"]}
assert declared == observed and len(observed) == 3
assert not observations["unresolved_visual_blockers"]
