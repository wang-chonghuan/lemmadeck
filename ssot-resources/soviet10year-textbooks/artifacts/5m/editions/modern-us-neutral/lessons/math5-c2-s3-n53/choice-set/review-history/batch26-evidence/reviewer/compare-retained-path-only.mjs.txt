function canonical(value) {
  if (typeof value === 'string') return JSON.stringify(value).replace(/[\u007f-\uffff]/g,
    char => `\\u${char.charCodeAt(0).toString(16).padStart(4, '0')}`);
  if (Array.isArray(value)) return `[${value.map(canonical).join(',')}]`;
  if (value !== null && typeof value === 'object')
    return `{${Object.keys(value).sort().map(key => `${canonical(key)}:${canonical(value[key])}`).join(',')}}`;
  return JSON.stringify(value);
}
const digest = value => sha(canonical(value));
function delta(before, after, keys = []) {
  if (canonical(before) === canonical(after)) return [];
  if (before !== null && after !== null && typeof before === 'object' && typeof after === 'object' &&
      Array.isArray(before) === Array.isArray(after) && canonical(Object.keys(before)) === canonical(Object.keys(after)))
    return Object.keys(before).flatMap(key => delta(before[key], after[key], [...keys, key]));
  return [{ path: '/' + keys.join('/'), before, after }];
}
