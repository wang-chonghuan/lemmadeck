import hashlib
import json
from fractions import Fraction as F
from pathlib import Path


OWN = Path(__file__).resolve().parent


def sha(data):
    return hashlib.sha256(data).hexdigest()


def save(name, value):
    with (OWN / name).open("x", encoding="utf-8") as handle:
        json.dump(value, handle, indent=2, ensure_ascii=False)
        handle.write("\n")


capture = json.loads((OWN / "new-figure-capture-receipt.json").read_bytes())
snapshot = json.loads((OWN / "opened-view-snapshot.json").read_bytes())
for input_record in capture["inputs"]:
    assert sha(Path(input_record["path"]).read_bytes()) == input_record["sha256"]
assert sha(Path(capture["capture"]["path"]).read_bytes()) == capture["capture"]["sha256"]
geometry = capture["visible_svg_geometry"]
vertical = [line for line in geometry["lines"] if "_ld-fine-grid-v-" in line["id"]]
horizontal = [line for line in geometry["lines"] if "_ld-fine-grid-h-" in line["id"]]
assert len(vertical) == 41 and len(horizontal) == 46
unit = F(800, 17)
origin = (F(4640, 17), F(8000, 17))
step_x, step_y = unit / 4, unit / 3
for i, line in enumerate(vertical):
    assert abs(float(line["x1"]) - float(origin[0] + (-4 + F(i, 4)) * unit)) < 1e-8
for i, line in enumerate(horizontal):
    assert abs(float(line["y1"]) - float(origin[1] - (-6 + F(i, 3)) * unit)) < 1e-8
drawn_line = next(line for line in geometry["lines"] if line["id"].endswith("_ld-derived-line"))
assert abs(float(drawn_line["y2"]) - float(drawn_line["y1"]) + F(4, 3) * (float(drawn_line["x2"]) - float(drawn_line["x1"]))) < 1e-8
assert geometry["min_text_px"] == 18

]
query_checks = []
for label, x, y in queries:
    assert y == F(4, 3) * x
    column, row_from_bottom = (x + 4) * 4, (y + 6) * 3
    assert column.denominator == row_from_bottom.denominator == 1

    px = origin[0] + unit * x
    py = origin[1] - unit * y

        "query": label, "exact_point": [str(x), str(y)],

    })
