def token(value):
    if isinstance(value,str):
        return value
    value=F(value)
    if value.denominator==1:
        return str(value.numerator)
    return ("-" if value<0 else "")+r"\frac{"+str(abs(value.numerator))+"}{"+str(value.denominator)+"}"


def record_text(values,unit=""):
    if isinstance(values,str):
        return values
    if len(values)==1:
        return "$"+token(values[0])+"$"+(" "+unit if unit else "")
    return "; ".join(f"{index}) ${token(value)}$"+(" "+unit if unit else "")
                     for index,value in enumerate(values,1))


def linear(coefficient,constant,variable):
    coeff="" if coefficient==1 else "-" if coefficient==-1 else token(coefficient)
    return coeff+variable+("+" if constant>=0 else "-")+token(abs(constant))
