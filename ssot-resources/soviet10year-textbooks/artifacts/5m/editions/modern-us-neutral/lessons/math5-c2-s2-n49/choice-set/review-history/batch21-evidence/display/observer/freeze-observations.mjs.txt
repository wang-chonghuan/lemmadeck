const intersects = (rect, item) => rect.y < item.y + item.height && rect.y + rect.height > item.y;
function covered(rect, ranges) {
  let endpoint = rect.y;
  for (const row of [...ranges].sort((a, b) => a.y - b.y)) {
    if (row.y > endpoint + 0.01) return false;
    if (row.y + row.height > endpoint) endpoint = row.y + row.height;
    if (endpoint >= rect.y + rect.height - 0.01) return true;
  }
  return false;
}
