function q(number, suffix, mode, stem, reason, options, concepts = [], figures = []) {
  const s = sources.get(number);
  const records = options.map(([text, derivation, misconception], i) => ({
    id: String.fromCharCode(65 + i), text, derivation, misconception
  }));
  questions.push({
    id: `n49-${number}${suffix ? `-${suffix}` : ""}`, source_numbers: [number],
    view: { mode, stem, figure_ids: figures, options: records.map(o => ({ id: o.id, text: o.text, figure_ids: [] })) },
    intent: {
      task: s.task, source_solution: s.solution, unit_reason: reason,
      concept_blocks: concepts, limitations: [...s.production_losses],
      figure_observations: figures.map(id => frozen.figure_observations.find(f => f.figure_id === id))
    },
    key: {
      correct: records.filter(o => o.misconception === null).map(o => o.id),
      rationales: records.map(o => ({
        option_id: o.id, truth: o.misconception === null ? "correct" : "incorrect",
        derivation: o.derivation, misconception: o.misconception
      }))
    }
  });
}
