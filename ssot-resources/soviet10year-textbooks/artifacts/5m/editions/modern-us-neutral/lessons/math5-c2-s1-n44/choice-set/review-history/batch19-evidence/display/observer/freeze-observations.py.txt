questions = []
for index, (labels, note) in enumerate(notes, 1):
    checks = {}
    for width in [1440, 390]:
        key = f"choices-{width}"
        current_question = 0
        option_nodes = []
        for node in geometry[key]["nodes"]:
            if node["class"] == "choice-heading":
                current_question += 1
            if current_question == index and node["tag"] == "LI" and node["class"] == "":
                option_nodes.append(node)
        options = []
        for label, node in zip(labels, option_nodes):
            y = node["rect"]["y"]
            end = y + node["rect"]["height"]
            images = []
            for purpose, region in zip(["first", "last"], plan["existing_observed_page_intervals"][key]):
                if y < region[1] and end > region[0]:
                    images.append(str(ROOT.parent / "offline-evidence/5m" / f"math5-c2-s1-n44-{key}-{purpose}.png"))
            for image in capture["captures"]:
                r = image["rect"]
                if image["combination"] == key and y < r["y"] + r["height"] and end > r["y"]:
                    images.append(image["path"])
            options.append({"id": label, "status": "pass", "actual_pixel_images": images, "viewed_native_montages": sorted({montage_for[file] for file in images}), "evidence": note})

    questions.append({"number": index, "option_count": len(labels), "visible_display_evidence": note, "viewports": checks})
