const locate = (document, pointer) => {
  const parts = pointer.split('/').slice(1);
  let current = document;
  for (const part of parts.slice(0, -1)) {
    current = Array.isArray(current)
      ? current.find((item) => item.id === part || item.source_number === part || item.option_id === part)
      : current[part];
    if (!current) throw new Error(`Missing disclosed repair path:${pointer}`);
  }
  return [current, parts.at(-1)];
};

    throw new Error(`Current field differs from repaired value:${change.path}`);
  }
  const [oldTarget, oldKey] = locate(preimage, change.path);
  oldTarget[oldKey] = structuredClone(change.before);
  return { ...change, verdict: 'pass', current_value_matches_after: true };
});
