def binding(path):
    path = Path(path)
    return {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}


def primary_judgment(path):
    match = re.fullmatch(
        r"math5-c2-s2-n47-(text|exercises|choices)-(1440|390)-(first|last|figure-[12](?:-end)?)",
        path.stem,
    )
    assert match
    surface, width, part = match.groups()
    width = int(width)
    standalone = True
    if part in ("first", "last"):
        evidence = endpoint_evidence[(surface, width, part)]
    elif surface == "text":
        if width == 1440:
            evidence = "Full actual fig150 two-layer composite painted: five-part parallelogram rink, right two shaded parts, complete cleaner/tree artwork and bottom Cleaned:800m-squared (2/5) label. No figure cropping or label occlusion."
        elif part.endswith("-end"):
            evidence = "Native mobile fig150 right window fully paints shaded right portion, cleaner and complete bottom 800m-squared (2/5) label. Left [0,358] and right [282,640] overlap to cover full width."
        else:
            evidence = "Actual mobile fig150 left window painted. The 640-wide element screenshot has white outside the visible 358px scroll window; rejected as standalone full-asset evidence. Full width is covered jointly with the already opened right endpoint."
            standalone = False
    elif width == 1440:
        evidence = "Actual fig151 composite painted: tree above B, dotted AB/BC, solid AC baseline, river crossing, A60-degrees/C45-degrees and 600m label. Exercise surface has slight horizontal overflow; its primary start/end pair jointly covers full width."
    elif surface == "choices" and part.endswith("-end"):
        evidence = "Native choice-mobile fig151 right window paints C,45-degrees and complete 600m. Its 358px left/right windows overlap and jointly cover full width."
    elif surface == "choices":
        evidence = "Actual choice-mobile fig151 left window paints A,60-degrees, tree/B and dotted lines. White outside visible scroll window is not full-asset coverage; independently opened right window completes the figure."
        standalone = False
    elif part.endswith("-end"):
        evidence = "Actual exercise-mobile fig151 right window paints C/45-degrees, but cuts leading 6 from 600m. Left/right 310px windows leave about 20px middle gap. Rejected as standalone complete-label evidence; actually opened scroll165 middle supplement paints full 600m and connects both windows."
        standalone = False
    else:
        evidence = "Actual exercise-mobile fig151 left window paints A/60-degrees, tree/B and river/dotted crossing. The 641-wide element screenshot has white outside its about310px scroll window. Rejected as standalone full-asset evidence; left/middle/right actual windows provide continuous coverage."
        standalone = False
    return surface, width, part, evidence, standalone
