import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { rebindOptionPositions } from './option-position-permutation.mjs'

const dir = path.dirname(fileURLToPath(import.meta.url))
const seed = JSON.parse(fs.readFileSync(path.join(dir, 'source.json'), 'utf8'))
const frozen = JSON.parse(fs.readFileSync(path.join(dir, 'source-solutions.json'), 'utf8'))
const questions = []
const c = (text, derivation) => ({ text, truth: 'correct', derivation, misconception: null })
const w = (text, derivation, misconception) => ({ text, truth: 'incorrect', derivation, misconception })
function q(number, suffix, mode, stem, choices, unit) {
  const source = seed.exercises.find(s => s.number === number)
  const solution = frozen.solutions.find(s => s.source_number === number)
  const options = choices.map((o, i) => ({ id: String.fromCharCode(65 + i), text: o.text, figure_ids: [] }))
  questions.push({
    id: `n47-${number}${suffix}`, source_numbers: [number],
    view: { mode, stem, figure_ids: source.figure_ids, options },
    intent: { task: solution.task, source_solution: solution.solution, unit_reason: unit,
      concept_blocks: [], limitations: solution.losses,
      figure_observations: source.figure_ids.map(figure_id => ({
        figure_id, observation: frozen.figure_observations[figure_id],
      })) },
    key: {
      correct: options.filter((_, i) => choices[i].truth === 'correct').map(o => o.id),
      rationales: choices.map((o, i) => ({ option_id: options[i].id, truth: o.truth,
        derivation: o.derivation, misconception: o.misconception })),
    },
  })
}
function list(number, suffix, expressions, choices, unit) {
  q(number, suffix, 'single', `依次${expressions}。哪一行全部正确？`, choices, unit)
}
