import copy
import hashlib
import json
from fractions import Fraction
from pathlib import Path


OWN = Path(__file__).resolve().parent
AUTHOR = OWN.parent / "5m-n47"
HISTORY = OWN / "history-before-870b-internal-rationale-repair"
INPUTS = {}


def load(path):
    raw = path.read_bytes()
    INPUTS[path] = hashlib.sha256(raw).hexdigest()
    return json.loads(raw)


def binding(path):
    return {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}


def digest(value):
    raw = json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(",", ":"))
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def save(path, value, mode="x"):
    with path.open(mode, encoding="utf-8") as output:
        json.dump(value, output, ensure_ascii=True, indent=2)
        output.write("\n")
