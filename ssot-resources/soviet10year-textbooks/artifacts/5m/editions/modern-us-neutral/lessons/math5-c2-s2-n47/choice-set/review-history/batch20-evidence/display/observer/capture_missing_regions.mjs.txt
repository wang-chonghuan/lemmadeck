import fs from 'node:fs'
import path from 'node:path'
import crypto from 'node:crypto'
import { createRequire } from 'node:module'
import { fileURLToPath, pathToFileURL } from 'node:url'

delete process.env.PWDEBUG
process.env.PLAYWRIGHT_HTML_OPEN = 'never'
const require = createRequire(path.join(process.cwd(), 'package.json'))
const { chromium } = require('@playwright/test')
const repo = path.resolve(process.cwd(), '..')
const own = path.dirname(fileURLToPath(import.meta.url))
const manifestPath = path.join(repo, '.intentfold/tickets/STEMROBIN-182/tmp/batch20-render/5m-math5-c2-s2-n47-capture-manifest.json')
const initial = JSON.parse(fs.readFileSync(manifestPath, 'utf8'))
const hash = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')
const rendered = initial.rendered.map(row => ({ ...row, file: path.join(repo, row.path) }))
for (const row of rendered) {
  if (hash(row.file) !== row.sha256) throw new Error(`HTML changed: ${row.file}`)
}
const output = path.join(own, 'supplements')
if (fs.existsSync(output)) throw new Error('Refusing existing supplemental directory')
fs.mkdirSync(output)
const report = {
  lesson: initial.lesson,
  scope: 'Missing prose/question/exercise regions, genuinely hidden formula ends and fig151 mobile middle only; existing HTML, no source/internal/renderer edits or reruns.',
  unique_surface_viewport_combinations_added: 0,
  existing_unique_surface_viewport_combinations: 6,
  rendered_bindings: rendered.map(({ file, sha256 }) => ({ path: file, sha256 })),
  primary_files_not_modified: true, headless: true, isolated_contexts: true,
  PWDEBUG_cleared: !('PWDEBUG' in process.env), PLAYWRIGHT_HTML_OPEN: process.env.PLAYWRIGHT_HTML_OPEN,
  captures: [], dom_inventory_for_capture_selection_only: [], browser_closed: false,
}
const browser = await chromium.launch({ headless: true })
async function settle(page) {
  await page.evaluate(async () => {
    await document.fonts.ready
    await new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(resolve)))
  })
}
async function capture(locator, name, metadata) {
  const filename = path.join(output, name)
  if (fs.existsSync(filename)) throw new Error(`Refusing overwrite: ${filename}`)
  await locator.scrollIntoViewIfNeeded()
  await settle(locator.page())
  await locator.screenshot({ path: filename, animations: 'disabled' })
  report.captures.push({ path: filename, sha256: hash(filename), ...metadata })
}
async function captureTextBeforeFigure(page, item, name, metadata) {
  const figure = item.locator('.fig').first()
  await item.scrollIntoViewIfNeeded()
  await settle(page)
  const itemBox = await item.boundingBox()
  const figBox = await figure.boundingBox()
  const filename = path.join(output, name)
  if (fs.existsSync(filename)) throw new Error('Output exists')
  const clip = { x: itemBox.x, y: itemBox.y, width: itemBox.width, height: figBox.y - itemBox.y }
  if (clip.height <= 0) throw new Error('Missing text-before-figure bounds')
  await page.screenshot({ path: filename, clip, animations: 'disabled' })
  report.captures.push({ path: filename, sha256: hash(filename), clip, ...metadata })
}
try {
  for (const viewport of [{ width: 1440, height: 960 }, { width: 390, height: 844 }]) {
    const context = await browser.newContext({ viewport, deviceScaleFactor: 1 })
    try {
      const page = await context.newPage()
      for (const surface of ['text', 'exercises', 'choices']) {
        const entry = rendered.find(row => row.path.endsWith(`/${surface}.html`))
        await page.goto(pathToFileURL(entry.file).href)
        await settle(page)
        const selector = surface === 'text' ? '.para' : surface === 'choices' ? '.choice' : 'li.ex'
        const items = page.locator(selector)
        const inventory = {
          surface, viewport, count: await items.count(),
          option_count: surface === 'choices' ? await page.locator('.choice-options>li').count() : null,
          media_dom_count: await page.locator('img,svg').count(),
          table_count: await page.locator('table').count(),
        }
        report.dom_inventory_for_capture_selection_only.push(inventory)
        if (surface === 'text' && viewport.width === 390) {
          const target = items.filter({ hasText: '问题2.' })
          await capture(target, 'math5-c2-s2-n47-text-390-missing-problem2-stem.png', {
            kind: 'missing_region', surface, viewport, region: 'problem2-stem',
          })
        }
        const start = surface === 'exercises' ? (viewport.width === 1440 ? 3 : 2)
          : surface === 'choices' ? (viewport.width === 1440 ? 3 : 2) : 0
        const end = surface === 'exercises' ? (viewport.width === 1440 ? 27 : 29)
          : surface === 'choices' ? (viewport.width === 1440 ? 40 : 41) : -1
        for (let index = start; index <= end; index++) {
          const item = items.nth(index)
          const number = surface === 'choices' ? index + 1 : 862 + index
          const metadata = {
            kind: 'missing_region', surface, viewport,
            item_index: index, item_number: number,
            option_count: surface === 'choices' ? await item.locator('.choice-options>li').count() : null,
          }
          const base = `math5-c2-s2-n47-${surface}-${viewport.width}-region-${number}`
          if (await item.locator('.fig').count()) {
            if (surface === 'choices') {
              await capture(item.locator('.choice-heading'), `${base}-heading.png`, { ...metadata, region: 'heading' })
              await capture(item.locator('.choice-stem'), `${base}-stem-left.png`, { ...metadata, region: 'stem' })
              await capture(item.locator('.choice-options'), `${base}-options-left.png`, { ...metadata, region: 'options' })
            } else {
              await captureTextBeforeFigure(page, item, `${base}-text-left.png`, { ...metadata, region: 'text-before-already-observed-figure' })
            }
            if (await item.locator('.figcap').count())
              await capture(item.locator('.figcap'), `${base}-caption.png`, { ...metadata, region: 'caption' })
          } else {
            await capture(item, `${base}-left.png`, metadata)
          }
        }
        const scrollNodes = await page.evaluate(() =>
          [...document.querySelectorAll('*')].filter(node =>
            node.clientWidth > 0 && node.scrollWidth > node.clientWidth + 1
            && ['auto', 'scroll'].includes(getComputedStyle(node).overflowX)
          ).map((node, index) => {
            node.setAttribute('data-observer-overflow-index', String(index))
            const choice = node.closest('.choice')
            const exercise = node.closest('li.ex')
            return {
              index, class_name: node.className,
              contains_figure: !!node.querySelector('.figure-media'),
              question_number: choice ? [...document.querySelectorAll('.choice')].indexOf(choice) + 1 : null,
              exercise_number: exercise ? [...document.querySelectorAll('li.ex')].indexOf(exercise) + 862 : null,
              option_id: node.closest('.choice-options>li')?.querySelector('.option-id')?.textContent || null,
              client_width: node.clientWidth, scroll_width: node.scrollWidth,
              visible_formula_tex: [...node.querySelectorAll('annotation[encoding="application/x-tex"]')].map(n => n.textContent),
            }
          }),
        )
        inventory.horizontal_regions = scrollNodes
        for (const row of scrollNodes) {
          const region = page.locator(`[data-observer-overflow-index="${row.index}"]`)
          let positions
          if (row.contains_figure) {
            if (surface !== 'exercises' || viewport.width !== 390) continue
            positions = ['middle']
          } else {
            positions = row.scroll_width > row.client_width * 2 ? ['middle', 'right'] : ['right']
          }
          for (const position of positions) {
            const metrics = await region.evaluate((node, position) => {
              node.scrollLeft = position === 'right' ? node.scrollWidth : (node.scrollWidth - node.clientWidth) / 2
              return { scroll_left: node.scrollLeft, maximum: node.scrollWidth - node.clientWidth,
                client_width: node.clientWidth, scroll_width: node.scrollWidth }
            }, position)
            if (position === 'right' && Math.abs(metrics.scroll_left - metrics.maximum) > 1)
              throw new Error('Did not reach horizontal right end')
            await capture(region, `math5-c2-s2-n47-${surface}-${viewport.width}-horizontal-${row.index + 1}-${position}.png`, {
              kind: row.contains_figure ? 'figure_middle_gap' : 'horizontal_end',
              surface, viewport, ...row, endpoint: position, scroll_metrics: metrics,
            })
          }
          await region.evaluate(node => { node.scrollLeft = 0 })
        }
      }
    } finally {
      await context.close()
    }
  }
} finally {
  await browser.close()
  report.browser_closed = true
  report.html_byte_hashes_unchanged = rendered.every(row => hash(row.file) === row.sha256)
  report.supplemental_png_count = report.captures.length
  fs.writeFileSync(path.join(own, 'supplement-capture-manifest.json'), JSON.stringify(report, null, 2) + '\n', { flag: 'wx' })
}
if (!report.html_byte_hashes_unchanged) throw new Error('HTML changed')
console.log(JSON.stringify({
  manifest: path.join(own, 'supplement-capture-manifest.json'),
  supplemental_png_count: report.captures.length,
  formula_ends: report.captures.filter(row => row.kind === 'horizontal_end').length,
  figure_middle_gaps: report.captures.filter(row => row.kind === 'figure_middle_gap').length,
  browser_closed: report.browser_closed, html_byte_hashes_unchanged: report.html_byte_hashes_unchanged,
}))
