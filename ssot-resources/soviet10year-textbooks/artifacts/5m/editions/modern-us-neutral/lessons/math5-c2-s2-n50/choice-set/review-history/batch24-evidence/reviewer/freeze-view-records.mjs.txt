function positionEvidence(questions) {
  const labels = questions.map((q) => q.reviewer_true_set.join(""));
  const runs = [];
  for (let i = 0; i < labels.length;) {
    let j = i + 1;
    while (j < labels.length && labels[j] === labels[i]) j++;
    if (j - i >= 3) {
      runs.push({
        start_ordinal: i + 1, end_ordinal: j, label: labels[i],
        length: j - i, question_ids: questions.slice(i, j).map((q) => q.id)
      });
    }
    i = j;
  }
  const periodicBlocks = [];
  for (let period = 2; period <= Math.floor(labels.length / 3); period++) {
    for (let i = 0; i + 3 * period <= labels.length; i++) {
      const pattern = labels.slice(i, i + period);
      if (new Set(pattern).size === 1) continue;
      // Record only primitive patterns; a repeated shorter word is one block.
      let primitive = true;
      for (let smaller = 1; smaller < period; smaller++) {
        if (period % smaller === 0 &&
            pattern.every((v, k) => v === pattern[k % smaller])) primitive = false;
      }
      if (!primitive) continue;
      let j = i + period;
      while (j < labels.length && labels[j] === labels[i + (j - i) % period]) j++;
      if (j - i >= 3 * period &&
          (i === 0 || labels[i - 1] !== labels[i - 1 + period])) {
        periodicBlocks.push({
          start_ordinal: i + 1, end_ordinal: j, period,
          pattern: pattern.join(""), complete_repetitions: Math.floor((j - i) / period),
          question_ids: questions.slice(i, j).map((q) => q.id)
        });
      }
    }
  }
  return {
    scope: "Every single-choice position in original visible order; all adjacent primitive periods fitting three complete repetitions were inspected, not just lesson totals.",
    reviewer_established_sequence: labels.join(""),
    label_counts: Object.fromEntries(["A", "B", "C"].map((id) => [id, labels.filter((v) => v === id).length])),
    maximum_identical_run: Math.max(1, ...runs.map((r) => r.length)),
    identical_runs_length_at_least_three: runs,
    adjacent_periodic_blocks_at_least_three_repetitions: periodicBlocks,
    ordinary_short_runs_policy: "No exact balancing quota and no automatic failure of every double/triple/fourfold run. Decisions below distinguish a conspicuous repeated local cycle plus a fivefold content-block run from ordinary short clusters.",
    whole_sequence_is_three_repeat_cycle: periodicBlocks.some((b) => b.start_ordinal === 1 && b.end_ordinal === labels.length)
  };
}
