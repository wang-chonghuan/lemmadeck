montages = []
member_to_montage = {}
for index in (primary, supplement):
    for montage in index["montages"]:
        if digest(montage["path"]) != montage["sha256"]:
            raise RuntimeError("Opened montage hash changed")
        with Image.open(montage["path"]) as canvas:
            canvas = canvas.convert("RGB")
            for member in montage["members"]:
                if digest(member["path"]) != member["sha256"]:
                    raise RuntimeError("Observed source screenshot hash changed")
                x, y, width, height = member["montage_rect"]
                with Image.open(member["path"]) as original:
                    if ImageChops.difference(original.convert("RGB"), canvas.crop((x, y, x + width, y + height))).getbbox():
                        raise RuntimeError("Native pixel proof no longer holds")
                member_to_montage[member["path"]] = montage["path"]
        montages.append({**montage, "actually_opened": True, "observer": OBSERVER,
                         "open_method": "functions.view_image detail=original; every native-copy member inspected"})

def covered(start, end, ranges):
    cursor = start
    for a, b in sorted(ranges):
        if b <= cursor:
            continue
        if a > cursor + 0.01:
            return False
        cursor = max(cursor, b)
        if cursor >= end - 0.01:
            return True
    return cursor >= end - 0.01
