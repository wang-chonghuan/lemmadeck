function leaves(before, after, pointer = "") {
  if (JSON.stringify(before) === JSON.stringify(after)) return [];
  if (before !== null && after !== null &&
      typeof before === "object" && typeof after === "object" &&
      Array.isArray(before) === Array.isArray(after)) {
    assert.deepEqual(Object.keys(before), Object.keys(after));
    return Object.keys(before).flatMap((key) =>
      leaves(before[key], after[key], `${pointer}/${key.replaceAll("~", "~0").replaceAll("/", "~1")}`)
    );
  }
  return [{ pointer, before, after }];
}

function runs(labels, questions) {
  const found = [];
  for (let i = 0; i < labels.length;) {
    let end = i + 1;
    while (end < labels.length && labels[end] === labels[i]) end++;
    found.push({
      start_ordinal: i + 1, end_ordinal: end, label: labels[i], length: end - i,
      question_ids: questions.slice(i, end).map((q) => q.id)
    });
    i = end;
  }
  return found;
}

function blocks(labels, questions) {
  const found = [];
  for (let period = 2; period <= Math.floor(labels.length / 2); period++) {
    for (let i = 0; i + 2 * period <= labels.length; i++) {
      const pattern = labels.slice(i, i + period);
      if (new Set(pattern).size < 2) continue;
      let primitive = true;
      for (let smaller = 1; smaller < period; smaller++) {
        if (period % smaller === 0 &&
            pattern.every((v, k) => v === pattern[k % smaller])) primitive = false;
      }
      if (!primitive) continue;
      let end = i + period;
      while (end < labels.length && labels[end] === labels[i + (end - i) % period]) end++;
      if (end - i >= 2 * period &&
          (i === 0 || labels[i - 1] !== labels[i - 1 + period])) {
        found.push({
          start_ordinal: i + 1, end_ordinal: end, period,
          pattern: pattern.join(""), complete_repetitions: Math.floor((end - i) / period),
          trailing_prefix_length: (end - i) % period,
          question_ids: questions.slice(i, end).map((q) => q.id)
        });
      }
    }
  }
  return found;
}
