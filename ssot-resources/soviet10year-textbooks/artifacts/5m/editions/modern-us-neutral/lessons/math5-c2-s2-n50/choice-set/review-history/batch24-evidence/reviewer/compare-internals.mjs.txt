const gcd = (a, b) => {
  a = a < 0n ? -a : a;
  b = b < 0n ? -b : b;
  while (b) [a, b] = [b, a % b];
  return a;
};
class R {
  constructor(n, d = 1n) {
    n = BigInt(n); d = BigInt(d);
    assert.notEqual(d, 0n);
    if (d < 0n) { n = -n; d = -d; }
    const g = gcd(n, d);
    this.n = n / g; this.d = d / g;
  }
  add(other) { const r = q(other); return new R(this.n * r.d + r.n * this.d, this.d * r.d); }
  sub(other) { const r = q(other); return new R(this.n * r.d - r.n * this.d, this.d * r.d); }
  mul(other) { const r = q(other); return new R(this.n * r.n, this.d * r.d); }
  div(other) { const r = q(other); return new R(this.n * r.d, this.d * r.n); }
  pow(n) { return new R(this.n ** BigInt(n), this.d ** BigInt(n)); }
  neg() { return new R(-this.n, this.d); }
  abs() { return new R(this.n < 0n ? -this.n : this.n, this.d); }
  toString() { return this.d === 1n ? String(this.n) : `${this.n}/${this.d}`; }
}
function q(value) {
  if (value instanceof R) return value;
  const text = String(value);
  if (text.includes("/")) {
    const [a, b] = text.split("/");
    return new R(a, b);
  }
  if (text.includes(".")) {
    const negative = text.startsWith("-");
    const unsigned = negative ? text.slice(1) : text;
    const [whole, fraction] = unsigned.split(".");
    const n = BigInt((whole || "0") + fraction);
    return new R(negative ? -n : n, 10n ** BigInt(fraction.length));
  }
  return new R(text);
}
function normalized(value) {
  if (value instanceof R) return value.toString();

  if (value !== null && typeof value === "object") {
    return Object.fromEntries(Object.entries(value).map(([key, item]) => [key, normalized(item)]));
  }
  return value;
}
