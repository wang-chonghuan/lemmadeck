const dir = path.dirname(fileURLToPath(import.meta.url))
const frozenBytes = fs.readFileSync(path.join(dir, 'source-solutions.json'))
assert(frozenBytes.equals(fs.readFileSync(path.join(dir, 'source-first-frozen/source-solutions.json'))))
const frozen = JSON.parse(frozenBytes)
const c = (text, derivation) => ({ text, derivation, misconception: null })
const w = (text, derivation, misconception) => ({ text, derivation, misconception })
const questions = []
const independent = 'This complete expression has independent operands and a full result; it is not an intermediate arithmetic blank or a dependency on an earlier answer.'
const model = 'The full quantity/model with units is one dependent mathematical response, retaining every requested relationship.'
function q(number, suffix, stem, choices, unit, concepts = [], figures = []) {
  const s = frozen.solutions.find(s => s.source_number === number)
  questions.push({
    id: `n50-${number}${suffix}`, source_numbers: [number],
    view: {
      mode: 'single', stem, figure_ids: figures,
      options: choices.map((o, i) => ({ id: String.fromCharCode(65 + i), text: o.text, figure_ids: [] }))
    },
    intent: {
      task: s.task, source_solution: s.solution, unit_reason: unit,
      concept_blocks: concepts, limitations: [...s.losses],
      figure_observations: figures.map(id => frozen.figure_observations.find(f => f.figure_id === id))
    },
    key: {
      correct: choices.flatMap((o, i) => o.misconception === null ? [String.fromCharCode(65 + i)] : []),
      rationales: choices.map((o, i) => ({
        option_id: String.fromCharCode(65 + i), truth: o.misconception === null ? 'correct' : 'incorrect',
        derivation: o.derivation, misconception: o.misconception
      }))
    }
  })
}
