  const source = solutions.get(number)

    id: `n52-${number}${suffix}`, source_numbers: [number],
    view: { mode: 'single', stem, figure_ids: [],
      options: choices.map((row, i) => ({ id: ids[i], text: row[0], figure_ids: [] })) },
    intent: {
      task, source_solution: source.independent_solution, unit_reason: unit,
      concept_blocks: [], limitations: [...source.losses, ...extraLosses],
      figure_observations: [],
    },
    key: {
      correct: choices.flatMap((row, i) => row[1] === 'correct' ? [ids[i]] : []),
      rationales: choices.map((row, i) => ({
        option_id: ids[i], truth: row[1], derivation: row[2], misconception: row[3],
      })),
    },
  })
