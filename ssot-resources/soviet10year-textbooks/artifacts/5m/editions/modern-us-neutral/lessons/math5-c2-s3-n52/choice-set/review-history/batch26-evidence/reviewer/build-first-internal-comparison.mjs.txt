function compare(before, after, pointer = "") {
  if (before === after) return;
  if (before !== null && after !== null && typeof before === "object" && typeof after === "object") {
    assert.deepEqual(Object.keys(before), Object.keys(after));
    for (const key of Object.keys(before)) compare(before[key], after[key], `${pointer}/${key}`);
  } else actualDelta.push({ pointer, before, after });
}
