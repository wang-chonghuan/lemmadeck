function q(number, suffix, stem, choices, unit, concepts = [], figures = []) {
  const s = frozen.solutions.find(s => s.source_number === number)
  questions.push({
    id: `n51-${number}${suffix}`, source_numbers: [number],
    view: { mode: 'single', stem, figure_ids: figures, options: choices.map((o,i) => ({id:String.fromCharCode(65+i),text:o.text,figure_ids:[]})) },
    intent: {
      task:s.task, source_solution:s.solution, unit_reason:unit, concept_blocks:concepts,
      limitations:[...s.losses,...(candidateLosses[number]??[])], figure_observations:figures.map(id => frozen.figure_observations.find(f=>f.figure_id===id))
    },
    key: {
      correct:choices.flatMap((o,i)=>o.misconception===null?[String.fromCharCode(65+i)]:[]),
      rationales:choices.map((o,i)=>({option_id:String.fromCharCode(65+i),truth:o.misconception===null?'correct':'incorrect',derivation:o.derivation,misconception:o.misconception}))
    }
  })
}
