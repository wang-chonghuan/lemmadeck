class R {

    if (d < 0n) { n = -n; d = -d; }
    const g = gcd(n, d);

  add(v) { const r = q(v); return new R(this.n * r.d + r.n * this.d, this.d * r.d); }
  sub(v) { const r = q(v); return new R(this.n * r.d - r.n * this.d, this.d * r.d); }
  mul(v) { const r = q(v); return new R(this.n * r.n, this.d * r.d); }
  div(v) { const r = q(v); return new R(this.n * r.d, this.d * r.n); }
  pow(n) { return new R(this.n ** BigInt(n), this.d ** BigInt(n)); }
  neg() { return new R(-this.n, this.d); }
  abs() { return new R(this.n < 0n ? -this.n : this.n, this.d); }

}

}
