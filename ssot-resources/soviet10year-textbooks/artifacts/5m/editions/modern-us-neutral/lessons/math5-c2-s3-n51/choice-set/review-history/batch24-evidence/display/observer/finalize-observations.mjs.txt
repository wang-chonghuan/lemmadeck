const valid = (i) => i[1] > i[0];
const merged = (intervals) => {
  const result = [];
  for (const i of intervals.filter(valid).sort((a, b) => a[0] - b[0])) {
    const last = result.at(-1);
    if (last && i[0] <= last[1] + 0.01) last[1] = Math.max(last[1], i[1]);
    else result.push([...i]);
  }
  return result;
};
