async function makeMontages(records,prefix) {
  const sheets=[];
  let remaining=[...records];
  while(remaining.length) {
    const placements=[];
    let x=0,y=24,rowHeight=0,width=0;
    for(let i=0;i<remaining.length;) {
      const r=remaining[i];
      if(r.width>2200||r.height>2600)throw Error(`Oversize native input ${r.path}`);
      if(x+r.width>2200) {x=0;y+=rowHeight+24;rowHeight=0;}
      if(y+r.height>2624)break;
      placements.push({record:r,left:x,top:y,label_index:records.indexOf(r)+1});
      width=Math.max(width,x+r.width);
      x+=r.width+12;rowHeight=Math.max(rowHeight,r.height);
      remaining.splice(i,1);
    }
    if(!placements.length)throw Error("No native input fits montage.");
    const height=y+rowHeight+24;
    const filename=path.join(own,`${prefix}-${String(sheets.length+1).padStart(2,"0")}.png`);
    await sharp({create:{width,height,channels:3,background:"#c4c4c4"}}).composite(
      placements.map((p)=>({input:path.join(root,p.record.path),left:p.left,top:p.top}))
    ).png().toFile(filename);
    sheets.push({path:relative(filename),sha256:readHash(filename),width,height,
      native_unscaled:true,source_pngs:placements.map((p)=>({path:p.record.path,sha256:p.record.sha256,
        placement:{left:p.left,top:p.top,width:p.record.width,height:p.record.height}}))});
  }
  return sheets;
}
