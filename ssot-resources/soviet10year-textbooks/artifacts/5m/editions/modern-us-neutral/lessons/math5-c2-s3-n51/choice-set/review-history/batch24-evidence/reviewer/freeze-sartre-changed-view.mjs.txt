function positionRecord(questions) {
  const labels = questions.map((q) => q.reviewer_true_set[0]);
  const runs = [];
  for (let i = 0; i < labels.length;) {
    let end = i + 1;
    while (end < labels.length && labels[end] === labels[i]) end++;
    if (end - i >= 3) runs.push({
      start_ordinal: i + 1, end_ordinal: end, label: labels[i], length: end - i,
      question_ids: questions.slice(i, end).map((q) => q.id)
    });
    i = end;
  }
  const blocks = [];
  for (let period = 2; period <= Math.floor(labels.length / 3); period++) {
    for (let start = 0; start + period * 3 <= labels.length; start++) {
      const pattern = labels.slice(start, start + period);
      if (new Set(pattern).size < 2) continue;
      let primitive = true;
      for (let smaller = 1; smaller < period; smaller++) {
        if (period % smaller === 0 && pattern.every((v, i) => v === pattern[i % smaller])) primitive = false;
      }
      if (!primitive) continue;
      let end = start + period;
      while (end < labels.length && labels[end] === labels[start + (end - start) % period]) end++;
      if (end - start >= period * 3 &&
          (start === 0 || labels[start - 1] !== labels[start - 1 + period])) {
        blocks.push({
          start_ordinal: start + 1, end_ordinal: end, period, pattern: pattern.join(""),
          repetitions: Math.floor((end - start) / period),
          question_ids: questions.slice(start, end).map((q) => q.id)
        });
      }
    }
  }
  return {
    current_sequence: labels.join(""),
    label_counts: Object.fromEntries(["A","B","C"].map((id) => [id, labels.filter((v) => v === id).length])),
    longest_identical_run: Math.max(...runs.map((r) => r.length)),
    runs_at_least_three: runs,
    adjacent_primitive_period_blocks_with_three_full_repetitions: blocks,
    verdict: blocks.length === 0 ? "pass" : "revise",
    evidence: "Only the insertion/replacement effect on the full sequence is assessed. No three-full-repeat adjacent primitive period is present. New975C extends the prior974a/974b/976b C triple into a fourfold run; maximum run stays4, as in the original frozen n51 verdict. Ordinary fourfold clusters are explicitly recorded and are not by themselves a deterministic position rule."
  };
}
