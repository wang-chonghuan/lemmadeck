const escapePointer = segment => String(segment).replaceAll("~", "~0").replaceAll("/", "~1");
function differences(before, after, pointer = "") {
  if (JSON.stringify(before) === JSON.stringify(after)) return [];
  if (Array.isArray(before) && Array.isArray(after) && before.length === after.length) {
    return before.flatMap((value, index) => differences(value, after[index], `${pointer}/${index}`));
  }
  if (before && after && !Array.isArray(before) && !Array.isArray(after) &&
      typeof before === "object" && typeof after === "object" &&
      JSON.stringify(Object.keys(before)) === JSON.stringify(Object.keys(after))) {
    return Object.keys(before).flatMap(key =>
      differences(before[key], after[key], `${pointer}/${escapePointer(key)}`));
  }
  return [{ path: pointer, before, after }];
}
