import assert from 'node:assert/strict'
import crypto from 'node:crypto'
import fs from 'node:fs'
import path from 'node:path'
import { spawnSync } from 'node:child_process'

const [batch, authorName, reviewerName, seedName = 'seed.json', reviewName = 'review.json',
  canonicalName = 'canonical-validation.json'] = process.argv.slice(2)
assert(/^batch[0-9]+$/.test(batch))
assert(authorName && reviewerName)
const scratch = '.intentfold/tickets/STEMROBIN-182/tmp'
const read = file => JSON.parse(fs.readFileSync(file, 'utf8'))
const sha = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')
const records = files => files.map(file => ({ path: file, sha256: sha(file) }))
const writeNew = (file, value) => {
  fs.mkdirSync(path.dirname(file), { recursive: true })
  fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n', { flag: 'wx' })
}
const seed = read(`${scratch}/${authorName}/${seedName}`)
const review = read(`${scratch}/${reviewerName}/${reviewName}`)
assert(['pass', 'blocked'].includes(review.outcome))
const canonical = read(`${scratch}/${reviewerName}/${canonicalName}`)
assert.equal(canonical.exit_code, review.outcome === 'pass' ? 0 : 3)
const { book, lesson } = seed.source
const root = 'ssot-resources/soviet10year-textbooks/artifacts'
const lessonDir = `${root}/${book}/editions/modern-us-neutral/lessons/${lesson}`
const choiceDir = review.outcome === 'pass' ? `${lessonDir}/choice-set`
  : `${lessonDir}/choice-set/review-history/unresolved-source`
for (const [name, original] of [
  ['seed.json', `${scratch}/${authorName}/${seedName}`],
  ['draft.json', `${scratch}/${authorName}/draft.json`],
  ['review.json', `${scratch}/${reviewerName}/${reviewName}`],
]) assert(fs.readFileSync(`${choiceDir}/${name}`).equals(fs.readFileSync(original)))
const inputs = records(['lesson.json', 'exercises.json', 'figures.json']
  .map(name => `${lessonDir}/${name}`).concat(['seed.json', 'draft.json', 'review.json']
    .map(name => `${choiceDir}/${name}`)))
const out = `${scratch}/${batch}-render`
const prefix = `${book}-${lesson}`
const reportName = `${lesson}-${batch}-report.json`
const captures = `${scratch}/offline-evidence/${book}`
const render = `.tmp/ld-s10y-lesson/render/${book}/modern-us-neutral/${lesson}`
fs.mkdirSync(out, { recursive: true })
for (const suffix of ['render-command.json', 'capture-command.json', 'capture-manifest.json']) {
  assert(!fs.existsSync(`${out}/${prefix}-${suffix}`))
}
if (fs.existsSync(captures)) {
  assert(!fs.readdirSync(captures).some(name =>
    name.startsWith(lesson + '-') || name === reportName))
}
const renderArgs = ['.claude/skills/ld-s10y-lesson/tools/p2c.py', 'render',
  '--book', book, '--edition', 'modern-us-neutral', '--lesson', lesson]
const env = { ...process.env, PLAYWRIGHT_HTML_OPEN: 'never', S10Y_REPORT_NAME: reportName }
delete env.PWDEBUG
delete env.S10Y_CHOICE_DIRS
if (review.outcome === 'blocked') {
  const shadow = `${out}/${prefix}-shadow-artifacts`
  assert(!fs.existsSync(shadow))
  const shadowBook = `${shadow}/${book}`
  const shadowEdition = `${shadowBook}/editions/modern-us-neutral`
  const shadowLesson = `${shadowEdition}/lessons/${lesson}`
  fs.mkdirSync(`${shadowLesson}/choice-set`, { recursive: true })
  for (const [original, target] of [
    [`${root}/${book}/book.json`, `${shadowBook}/book.json`],
    [`${root}/${book}/editions/modern-us-neutral/book.json`, `${shadowEdition}/book.json`],
    ...['lesson.json', 'exercises.json', 'figures.json']
      .map(name => [`${lessonDir}/${name}`, `${shadowLesson}/${name}`]),
    ...['seed.json', 'draft.json', 'review.json']
      .map(name => [`${choiceDir}/${name}`, `${shadowLesson}/choice-set/${name}`]),
  ]) fs.copyFileSync(original, target, fs.constants.COPYFILE_EXCL)
  fs.cpSync(`${root}/${book}/editions/modern-us-neutral/figures`,
    `${shadowEdition}/figures`, { recursive: true, errorOnExist: true, force: false })
  renderArgs.push('--root', shadow)
  env.S10Y_CHOICE_DIRS = JSON.stringify({ [lesson]: choiceDir })
  writeNew(`${out}/${prefix}-blocked-shadow.json`, {
    semantic_outcome: 'blocked', durable_core_unchanged: true,
    purpose: 'Render the retained blocked view from a byte-copy shadow through the existing renderer, without adding an approved choice-set.',
    shadow_core: records(['seed.json', 'draft.json', 'review.json']
      .map(name => `${shadowLesson}/choice-set/${name}`)),
    renderer_changed: false,
  })
}
if (fs.existsSync(render)) {
  const existing = fs.readdirSync(render).filter(name => name.endsWith('.html'))
  if (existing.length) {
    const archive = `${out}/${prefix}-render-before`
    fs.mkdirSync(archive)
    for (const name of existing) {
      fs.copyFileSync(`${render}/${name}`, `${archive}/${name}`, fs.constants.COPYFILE_EXCL)
    }
    writeNew(`${archive}/manifest.json`, {
      purpose: 'Prior render bytes retained before the one new batch display execution.',
      originals: records(existing.map(name => `${render}/${name}`)),
      copies: records(existing.map(name => `${archive}/${name}`)),
    })
  }
}
function run(command, cwd, destination, dependencies) {
  assert(!fs.existsSync(destination))
  const started = new Date().toISOString()
  const result = spawnSync(command[0], command.slice(1), {
    cwd, env, encoding: 'utf8', maxBuffer: 32 * 1024 * 1024,
  })
  writeNew(destination, {
    command, cwd, started, finished: new Date().toISOString(),
    exit_code: result.status, signal: result.signal,
    error: result.error?.message ?? null, stdout: result.stdout, stderr: result.stderr,
    inputs: dependencies, PWDEBUG_cleared: true, PLAYWRIGHT_HTML_OPEN: 'never',
    S10Y_REPORT_NAME: reportName, S10Y_CHOICE_DIRS: env.S10Y_CHOICE_DIRS ?? null,
    execution_count: 1, semantic_outcome: review.outcome,
    scope: 'Offline display only; no new semantic/canonical verification or promotion.',
    model: null, usage: null,
  })
  console.log(JSON.stringify({ command_record: destination, exit_code: result.status }))
  assert.equal(result.status, 0, result.stderr || result.stdout)
}
run(['.claude/skills/ld-s10y-lesson/.venv/bin/python', ...renderArgs],
  process.cwd(), `${out}/${prefix}-render-command.json`, inputs)
const rendered = records(['text', 'exercises', 'choices'].map(name => `${render}/${name}.html`))
run(['node', '../.intentfold/tickets/STEMROBIN-182/tmp/offline-check.mjs', book, lesson],
  path.resolve('app'), `${out}/${prefix}-capture-command.json`,
  [...inputs, ...rendered, ...records([`${scratch}/offline-check.mjs`])])
assert.deepEqual(records(inputs.map(row => row.path)), inputs)
const report = read(`${captures}/${reportName}`)
assert.equal(report.status, 'pass')
assert.equal(report.checks.length, 6)
assert(report.checks.every(row => row.lesson === lesson))
const screenshots = records(fs.readdirSync(captures).filter(name =>
  name.startsWith(lesson + '-') && name.endsWith('.png')).sort()
  .map(name => `${captures}/${name}`))
writeNew(`${out}/${prefix}-capture-manifest.json`, {
  lesson, book, status: 'DOM pass; actual pixel observation pending',
  semantic_outcome: review.outcome, unique_surface_viewport_combinations: 6,
  inputs_unchanged: true, inputs, rendered, report: `${captures}/${reportName}`,
  screenshots, checks: report.checks,
  browser_launch: { headless: true, isolated_contexts: true, user_profile: false },
  browser_closed: 'Existing checker closes in finally before exit0.',
  model: null, usage: null,
})
console.log(JSON.stringify({ lesson, primary_PNGs: screenshots.length,
  combinations: 6, actual_pixel_approval_pending: true }))
