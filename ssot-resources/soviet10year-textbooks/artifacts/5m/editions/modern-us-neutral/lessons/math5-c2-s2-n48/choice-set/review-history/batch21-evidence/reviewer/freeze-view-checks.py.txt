def source_of(q):
    return q["id"].split("-")[1]


def freeze_fidelity(lesson, data):
    view_path = TMP / f"5m-{lesson}/view.json"
    source_path = TMP / f"reviewer-5m-{lesson}/source-solutions.json"
    view = load(view_path)
    source = load(source_path)
    records = []
    for q in view["questions"]:
        source_number = source_of(q)
        evidence, losses = data[source_number]
        records.append({
            "question_id": q["id"],
            "source_number_inferred_from_visible_id_and_original_content": source_number,
            "task_fidelity": {"verdict": "pass", "evidence": evidence},
            "redundancy": {
                "verdict": "pass",
                "evidence": (
                    f"{q['id']} represents its stated original mathematical subtask. "
                    "Other package units use distinct inputs, geometric domains or source subparts; "
                    "shared methods and complete-row recognition alone do not make this redundant. "
                    + ("A/B are different valid five-point choices, so their shared line/conclusions do not erase the original freedom of point selection." if source_number == "928" else "")
                ),
            },
            "answer_leakage_before_mathematics": {
                "verdict": "pass",
                "evidence": (
                    "No source selector, correct/incorrect marker, rationale, answer declaration or "
                    "grading cue appears in the learner stem/options. Candidate procedures and complete "
                    "models are alternatives, not facts predeclared true. Cross-unit numerical copying "
                    "does not supply this unit's answer."
                ),
                "qualification": "Position-pattern judgment will use only independently established true sets in the next phase, never an author key.",
            },
            "production_losses": losses,
        })
    coverage = []
    for s in source["source_reviews"]:
        number = s["source_number"]
        ids = [q["id"] for q in view["questions"] if source_of(q) == number]
        evidence, losses = data[number]
        coverage.append({
            "source_number": number,
            "visible_question_ids": ids,
            "verdict": "blocked" if lesson == "n49" and number == "934" else "pass",
            "evidence": evidence,
            "production_losses": losses,
            "omitted_or_unresolved_tasks": [
                "934(4) unique numerical value cannot be recovered from the unconfirmed divisor; the conditional/nonuniqueness task is covered, not the missing numerical result."
            ] if lesson == "n49" and number == "934" else [],
            "internal_loss_disclosure_assessment": "Not authorized/read in pure-view phase. These exact losses must survive later intent/coverage comparison; recognition coverage is not full production coverage.",
        })
    result = {
        "schema": "independent-pure-view-fidelity-first-v1",
        "stage": "fidelity/redundancy/leakage/coverage recorded before option-truth file",
        "view_path": str(view_path),
        "view_byte_sha256": sha(view_path),
        "source_freeze_path": str(source_path),
        "source_freeze_byte_sha256": sha(source_path),
        "source_images": "Reused hash-bound actual observations from own image-observations.json; no reopening or authored-image approval.",
        "read_boundary": "Pure view plus original seed and own frozen source evidence only; no draft, ready, author solutions, serializer, rationales, keys, history choices, validators, canonical or render.",
        "question_count": len(view["questions"]),
        "option_count": sum(len(q["options"]) for q in view["questions"]),
        "question_reviews": records,
        "source_coverage": coverage,
        "whole_package_dependencies": [
            "n48 geometry construction methods recur in 896/898/907 by source design; recognize correctness for each domain rather than infer a key from a copied declaration.",
            "Same numerical results on distinct n48-901 expressions do not eliminate those computational tasks.",
            "n49-925 has 11.25 as a competitor while n49-926 has its own distinct valid rate; neither question declares the other's result.",
            "n49-928 and 929 specify different full lines; 930/931 are closed segments and must not inherit a whole-line domain.",
            "n49-933 uses explicitly fixed common axes; no other question precomputes its meeting/rest/arrival tuple.",
            "Ratio/percentage source parallels are practice of distinct inputs, not duplicated answer slots.",
        ],
        "pending": "Independent complete option truth sets and position-cycle assessment.",
    }
    target = TMP / f"reviewer-5m-{lesson}/view-fidelity-first.json"
    if target.exists():
        raise RuntimeError(f"Do not overwrite frozen first-pass fidelity: {target}")
    target.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps({"lesson": lesson, "fidelity_file_sha256": sha(target),
                      "questions": result["question_count"], "options": result["option_count"]}))

            "option_evidence": dict(zip("ABCD", evidence))}

def freeze_truths(lesson, mathematics):
    reviewer_dir = TMP / f"reviewer-5m-{lesson}"
    view_path = TMP / f"5m-{lesson}/view.json"
    view = load(view_path)
    fidelity_path = reviewer_dir / "view-fidelity-first.json"
    fidelity = load(fidelity_path)
    result = {
        "schema": "independent-pure-view-checks-v1",
        "stage": "pure-view-all-truths-frozen-before-internals",
        "reviewer": "Independent staged choice reviewer, shared invocation",
        "view_path": str(view_path),
        "view_byte_sha256": sha(view_path),
        "source_freeze_path": fidelity["source_freeze_path"],
        "source_freeze_byte_sha256": fidelity["source_freeze_byte_sha256"],
        "fidelity_first_file": str(fidelity_path),
        "fidelity_first_byte_sha256": sha(fidelity_path),
        "review_order": ["Original source/actual-image independent freeze", "Pure-view fidelity, redundancy, prose leakage and whole-source coverage freeze", "Independently solve every visible option and complete true set", "Assess position leakage using only these independent true sets", "Stop before internals"],
        "read_boundary": fidelity["read_boundary"],
        "source_image_observations": {
            "path": str(reviewer_dir / "image-observations.json"),
            "sha256": sha(reviewer_dir / "image-observations.json"),
            "reopened": False,
            "new_figure_approval": "No new figure or extra figure approval; existing source observations reused.",
        },
        "question_reviews": [],
        "source_coverage": fidelity["source_coverage"],
        "whole_package_dependencies": fidelity["whole_package_dependencies"],
        "findings": [],
    }
    for index, q in enumerate(view["questions"]):
        suffix = q["id"].removeprefix(f"{lesson}-")
        m = mathematics[suffix]
        f = fidelity["question_reviews"][index]
        record = {
            **f,
            "visible_question_ordinal": index + 1,
            "answer_leakage": f["answer_leakage_before_mathematics"],
            "mathematics": {
                "verdict": "pass",
                "independent_solution": m["independent_solution"],
                "correct": m["correct"],
                "option_checks": [{
                    "option_id": o["id"],
                    "truth": "correct" if o["id"] in m["correct"] else "incorrect",
                    "evidence": m["option_evidence"][o["id"]],
                } for o in q["options"]],
                "author_key_or_rationale_compared": False,
            },
        }
        record.pop("answer_leakage_before_mathematics")
        result["question_reviews"].append(record)
    if lesson == "n48":
        runs = [result["question_reviews"][5:25]]
        bias = {"question_ids": [r["question_id"] for r in result["question_reviews"][1:5]],
                "independent_true_sets": [["A"]] * 4,
                "evidence": "The four initial single-choice geometry questions all use A; this is additional concentration, but the decisive defect is the later strict 20-unit periodic run."}
    else:
        runs = [
            result["question_reviews"][1:24],
            [r for r in result["question_reviews"][24:] if len(r["mathematics"]["correct"]) == 1],
        ]
        bias = None
    cycle_records = []
    affected = set()
    for run in runs:
        sequence = [r["mathematics"]["correct"][0] for r in run]
        cycle_records.append({
            "question_ids": [r["question_id"] for r in run],
            "single_question_count": len(run),
            "independent_correct_label_sequence": sequence,
            "independent_correct_display_position_sequence": [
                next(i + 1 for i, o in enumerate(view["questions"][r["visible_question_ordinal"] - 1]["options"])
                     if o["id"] == r["mathematics"]["correct"][0]) for r in run],
            "observed_pattern": "Strict repeated A,B,C / first,second,third with final prefix retained.",
            "evidence": "Every independent true singleton in this run follows A,B,C cyclically. Once a learner infers this repeated arrangement, later labels/positions can be chosen without resolving their distinct mathematical tasks. This is a package-level design cue, not an author key read.",
        })
        affected.update(r["question_id"] for r in run)
    for r in result["question_reviews"]:
        if r["question_id"] in affected:
            r["answer_leakage"] = {
                "verdict": "revise",
                "evidence": f"{r['question_id']} participates in the independently established long strict A/B/C correct-position cycle recorded below. Its prose alone has no declaration of truth, but whole-package position prediction supplies a nonmathematical answer cue.",
            }
    repair = {
        "severity": "revise",
        "scope": "Only displayed option content positions and corresponding displayed letter assignments of the affected units; preserve mathematical wording, stems, source domains, modes, figures and source relationships.",
        "affected_question_ids": [r["question_id"] for r in result["question_reviews"] if r["question_id"] in affected],
        "request": "Rearrange the same option contents into a nonperiodic whole-package order and relabel A/B/C (and D when present) by actual display position. Merely moving stable-letter objects while leaving all correct letters in the same A/B/C cycle does not remove the cue. No new distractor or semantic rewrite is required. Provide the revised pure view for a new frozen visible-set/order check before internals.",
        "must_not_change": "Source numbers/data, original domains/quantifiers, complete alternative mathematical contents, native modes, figures, and the honest 934(4) blocker.",
    }
    result["position_cycle_review"] = {
        "verdict": "revise", "runs": cycle_records,
        "initial_geometry_position_bias": bias,
        "multiple_units_not_in_cycle": [r["question_id"] for r in result["question_reviews"] if len(r["mathematics"]["correct"]) > 1],
        "minimal_repair": repair,
    }
    result["findings"].append({
        "verdict": "revise",
        "topic": "Whole-visible-package correct-position periodicity",
        "evidence": f"Independent true singleton sequences have strict A/B/C runs of lengths {[len(run) for run in runs]}; details and all affected IDs recorded in position_cycle_review.",
        "minimal_repair": repair,
    })
    if lesson == "n49":
        result["findings"].append({
            "verdict": "blocked",
            "topic": "Original source934(4) unique numerical recovery",
            "evidence": "The unresolved divisor is not supplied by any visible unit. n49-934-4-notice has determinate conditional true set B, but neither it nor 32 other valid sources resolves the missing original number.",
            "minimal_repair": "None from current input. Retain the honest blocked numerical task and conditional explanation; do not ask the author to guess.",
        })
    result["production_loss_review"] = {
        "verdict": "pass",
        "scope": "Recognition translation remains mathematically meaningful, but this is not full production coverage.",
        "by_source": [{"source_number": c["source_number"], "losses": c["production_losses"]}
                      for c in result["source_coverage"] if c["production_losses"]],
        "later_internal_requirement": "When separately authorized, compare these explicit construction, model-generation, endpoint-plotting, proof-writing and actual scale-measurement losses against intent/coverage. Their disclosure cannot be certified from a pure view and is not presumed to exist.",
    }
    options = [o for q in result["question_reviews"] for o in q["mathematics"]["option_checks"]]
    result["counts"] = {
        "questions": len(result["question_reviews"]),
        "options": len(options),
        "independently_checked_options": len(options),
        "correct_options": sum(o["truth"] == "correct" for o in options),
        "incorrect_options": sum(o["truth"] == "incorrect" for o in options),
        "complete_true_sets": len(result["question_reviews"]),
        "source_records": len(result["source_coverage"]),
        "mathematics_pass_questions": len(result["question_reviews"]),
        "author_key_reads": 0,
        "author_rationale_reads": 0,
    }
    result["all_visible_truths_frozen"] = True
    result["mathematics_outcome"] = "pass"
    result["view_design_outcome"] = "revise"
    result["outcome"] = "blocked" if lesson == "n49" else "revise"
    result["next_authorization"] = "Revised pure-view order repair only; no internals are authorized or read."
    target = reviewer_dir / "view-checks.json"
    if target.exists():
        raise RuntimeError(f"Frozen pure-view result already exists: {target}")
    target.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps({"lesson": lesson, "sha256": sha(target), "counts": result["counts"],
                      "outcome": result["outcome"], "cycle_lengths": [len(run) for run in runs]}))
