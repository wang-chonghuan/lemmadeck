for (const [kind, index] of [['primary', primary], ['supplementary', supplemental]]) {
  for (const montage of index.montages) {
    if (hash(await fs.readFile(montage.path)) !== montage.sha256) throw new Error('Montage changed');
    montage.opened = true;
    montage.open_method = 'functions.view_image detail=original';
    nativeMontages.push({
      path: montage.path, sha256: montage.sha256, dimensions: montage.dimensions,
      opened: true, native_scale: 1, resampled: false, cropped: false,
      source_files: montage.members.map(member => member.source_path),
      counted_as_additional_actual_PNG_input: false,
    });
    for (const member of montage.members) {
      const number = Number(member.label.match(/^[PS](\d+)/)[1]);
      const sourceBytes = await fs.readFile(member.source_path);
      if (hash(sourceBytes) !== member.sha256) throw new Error('Actual screenshot changed');
      if (hash(await fs.readFile(member.absolute_path)) !== member.sha256) throw new Error('Opened copy changed');
      const item = {
        id: `${kind === 'primary' ? 'P' : 'S'}${String(number).padStart(2, '0')}`,
        path: member.source_path, sha256: member.sha256,
        opened: true, opened_via: montage.path, native_montage_rect: member.montage_rect,
        all_source_pixels_included: true, native_scale: 1,
        observation: (kind === 'primary' ? primaryNotes : supplementaryNotes)[number - 1],
      };
      if (kind === 'primary') {
        item.opened_copy_path = member.absolute_path;
        item.copy_hash_verified = member.copy_hash_verified;
        primaryFiles.push(item);
      } else {
        const shot = capture.screenshots.find(source => source.path === member.source_path);
        Object.assign(item, {
          surface: shot.surface, viewport: shot.viewport,
          exact_render_sha256: shot.exact_render_sha256,
          page_scroll: shot.actual, full_viewport_context: true,
        });
        supplementalFiles.push(item);
      }
    }
  }
  await fs.writeFile(path.join(own, `${kind === 'primary' ? 'primary' : 'supplemental'}-montage-index.json`),
    JSON.stringify(index, null, 2) + '\n');
}
