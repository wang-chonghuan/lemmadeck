const root = ".intentfold/tickets/STEMROBIN-182/tmp/5m-n48";
const frozenBytes = fs.readFileSync(`${root}/source-solutions.json`);
const frozen = JSON.parse(frozenBytes);
const sources = new Map(frozen.source_solutions.map(s => [s.source_number, s]));
const seedDigest = spawnSync("python3", [".agents/skills/ld-s10y-choice/scripts/prepare.py", "digest", `${root}/source.json`], { encoding: "utf8" }).stdout.trim();
if (seedDigest !== frozen.seed_sha256) throw new Error("Frozen seed mismatch");
const questions = [];
function q(number, suffix, mode, stem, reason, options, figures = [], concepts = [], extraLosses = []) {
  const s = sources.get(number);
  const id = `n48-${number}${suffix ? `-${suffix}` : ""}`;
  const records = options.map(([text, derivation, misconception], i) => ({
    id: String.fromCharCode(65 + i), text, derivation, misconception
  }));
  questions.push({
    id, source_numbers: [number],
    view: {
      mode, stem, figure_ids: figures,
      options: records.map(o => ({ id: o.id, text: o.text, figure_ids: [] }))
    },
    intent: {
      task: s.task, source_solution: s.solution, unit_reason: reason,
      concept_blocks: concepts,
      limitations: [...s.production_losses, ...extraLosses],
      figure_observations: figures.map(id => frozen.figure_observations.find(f => f.figure_id === id))
    },
    key: {
      correct: records.filter(o => o.misconception === null).map(o => o.id),
      rationales: records.map(o => ({ option_id: o.id, truth: o.misconception === null ? "correct" : "incorrect", derivation: o.derivation, misconception: o.misconception }))
    }
  });
}
