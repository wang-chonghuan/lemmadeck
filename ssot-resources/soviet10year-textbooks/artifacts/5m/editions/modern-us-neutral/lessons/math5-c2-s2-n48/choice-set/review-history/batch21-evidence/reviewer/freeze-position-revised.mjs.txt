function periodicObservation(units) {
  const sequence = units.map((unit) => unit.correct[0]);
  const blocks = [];
  const periods = [];
  let comparedStartingWindows = 0;
  for (let period = 1; period <= Math.floor(sequence.length / 2); period += 1) {
    periods.push(period);
    for (let start = 0; start + 2 * period <= sequence.length; start += 1) {
      comparedStartingWindows += 1;
      if (start > 0 && sequence[start - 1] === sequence[start + period - 1]) continue;
      let end = start + period;
      while (end < sequence.length && sequence[end] === sequence[end - period]) end += 1;
      if (end - start < 2 * period) continue;
      const seed = sequence.slice(start, start + period);
      let primitive = true;
      for (let shorter = 1; shorter < period; shorter += 1) {
        if (period % shorter === 0 && seed.every((item, index) => item === seed[index % shorter])) {
          primitive = false;
        }
      }
      if (!primitive) continue;
      blocks.push({
        period,
        seed: seed.join(""),
        first_single_ordinal: start + 1,
        last_single_ordinal: end,
        length: end - start,
        full_copies: Math.floor((end - start) / period),
        sequence: sequence.slice(start, end).join(""),
        question_ids: units.slice(start, end).map((unit) => unit.id),
      });
    }
  }
  return {
    question_ids: units.map((unit) => unit.id),
    independent_correct_label_sequence: sequence,
    independent_correct_display_position_sequence: units.map((unit) => unit.positions[0]),
    sequence_text: sequence.join(""),
    position_counts: sequence.reduce((counts, label) => {
      counts[label] = (counts[label] ?? 0) + 1;
      return counts;
    }, {}),
    periods_examined: periods,
    possible_two_copy_starting_windows_examined: comparedStartingWindows,
    maximal_primitive_adjacent_periodic_blocks: blocks,
    nonconstant_two_copy_blocks: blocks.filter((block) => block.period > 1),
    longest_constant_run: Math.max(1, ...blocks.filter((block) => block.period === 1).map((block) => block.length)),
  };
}
