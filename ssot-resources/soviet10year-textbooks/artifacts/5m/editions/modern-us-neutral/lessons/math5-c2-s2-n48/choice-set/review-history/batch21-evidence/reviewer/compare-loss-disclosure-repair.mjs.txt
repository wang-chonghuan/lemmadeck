const hash = (bytes) => crypto.createHash("sha256").update(bytes).digest("hex");
const encode = (value) => JSON.stringify(value, null, 2) + "\n";
const read = (file) => {
  const bytes = fs.readFileSync(file);
  return { bytes, value: JSON.parse(bytes.toString("utf8")), sha256: hash(bytes) };
};
const sorted = (value) => Array.isArray(value)
  ? value.map(sorted)
  : value && typeof value === "object"
    ? Object.fromEntries(Object.keys(value).sort().map((key) => [key, sorted(value[key])]))
    : value;
const snapshotDigest = (value) => hash(JSON.stringify(sorted(value))
  .replace(/[^\x00-\x7f]/g, (character) => "\\u" + character.charCodeAt(0).toString(16).padStart(4, "0")));
const same = (left, right) => JSON.stringify(left) === JSON.stringify(right);
const assert = (condition, message) => {
  if (!condition) throw new Error(message);
};
const pointerEscape = (key) => String(key).replaceAll("~", "~0").replaceAll("/", "~1");
const at = (value, pointer) => pointer.split("/").slice(1).reduce((current, key) =>
  current[key.replaceAll("~1", "/").replaceAll("~0", "~")], value);

// Only byte/JSON identity and the authorized appended disclosure strings are
// inspected; unchanged mathematical/source/view evidence is not re-reviewed.
function leafDiff(before, after, pointer = "") {
  if (same(before, after)) return [];
  if (Array.isArray(before) && Array.isArray(after)) {
    const changes = [];
    for (let index = 0; index < Math.max(before.length, after.length); index += 1) {
      const child = `${pointer}/${index}`;
      if (index >= before.length) changes.push({ operation: "add", path: child, value: after[index] });
      else if (index >= after.length) changes.push({ operation: "remove", path: child, before: before[index] });
      else changes.push(...leafDiff(before[index], after[index], child));
    }
    return changes;
  }
  if (before && after && typeof before === "object" && typeof after === "object" &&
      !Array.isArray(before) && !Array.isArray(after)) {
    return [...new Set([...Object.keys(before), ...Object.keys(after)])].flatMap((key) => {
      const child = `${pointer}/${pointerEscape(key)}`;
      if (Object.hasOwn(before, key) && Object.hasOwn(after, key)) return leafDiff(before[key], after[key], child);
      if (Object.hasOwn(after, key)) return [{ operation: "add", path: child, value: after[key] }];
      return [{ operation: "remove", path: child, before: before[key] }];
    });
  }
  return [{ operation: "replace", path: pointer, before, after }];
}
