import copy
import hashlib
import json
from collections import Counter
from pathlib import Path


ROOT = Path(__file__).resolve().parent.parent
INPUTS = {}


def load(path):
    data = path.read_bytes()
    INPUTS[str(path)] = hashlib.sha256(data).hexdigest()
    return json.loads(data)


def digest(value, ascii=True):
    data = json.dumps(
        value, sort_keys=True, ensure_ascii=ascii, separators=(",", ":")
    ).encode("utf-8")
    return hashlib.sha256(data).hexdigest()


def binding(path):
    return {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}


def content(value, excluded):
    return {key: item for key, item in value.items() if key != excluded}


def project(draft):
    return {"questions": [{"id": q["id"], **q["view"]} for q in draft["questions"]]}


def reverse_replacements(current, replacements):
    previous = copy.deepcopy(current)
    seen = set()
    for change in replacements:
        assert change["op"] == "replace"
        assert change["path"] not in seen
        seen.add(change["path"])
        parts = change["path"].strip("/").split("/")
        parent = previous
        for part in parts[:-1]:
            parent = parent[int(part)] if isinstance(parent, list) else parent[part]
        key = int(parts[-1]) if isinstance(parent, list) else parts[-1]
        assert parent[key] == change["value"], change["path"]
        parent[key] = change["old_value"]
    return previous


def frozen_solution(row):
    mathematics = row["mathematics"]
    solution = mathematics.get("frozen_independent_solution") or mathematics[
        "independent_solution"
    ]
    return (
        "Reused independent pre-internals content solution; any option labels "
        "in the following historical prose refer to its prior frozen view: "
        + solution
        + " Current content-rebound complete true set: "
        + ", ".join(mathematics["correct"])
        + "."
    )


def common(number):
    own = ROOT / f"reviewer-5m-n{number}"
    author = ROOT / f"5m-n{number}"
    draft = load(author / "draft.json")
    source = load(own / "source-solutions.json")
    seed = load(own / "source-input-preserved.json")
    frozen = load(own / "view-checks-option-position-revised.json")
    frozen_view = load(own / "view-input-preserved-option-position-revised.json")
    assert digest(seed) == draft["seed_sha256"]
    assert project(draft) == frozen_view
    assert frozen["position_repair_stage_status"] == "pass"
    assert draft["new_figures"] == []
    assert len(draft["questions"]) == frozen["visible_question_count"]
    receipt = load(author / "option-position-permutation-receipt.json")
    diff = load(author / "option-position-json-diff.json")
    assert receipt["file_byte_sha256"]["draft.json"]["after"] == INPUTS[
        str(author / "draft.json")
    ]
    assert receipt["revised_view_byte_sha256"] == INPUTS[
        str(own / "view-input-preserved-option-position-revised.json")
    ]
    return own, author, draft, source, seed, frozen, receipt, diff


def standard_record(draft, seed, source, question_reviews, source_reviews):
    return {
        "schema": "s10y-choice-review-v1",
        "seed_sha256": digest(seed),
        "draft_sha256": digest(draft),
        "reviewer": {
            "kind": "independent_agent",
            "identity": "independent source-first reviewer; authorized current internal comparison",
        },
        "source_reviews": source_reviews,
        "figure_checks": [
            {
                key: figure[key]
                for key in ("figure_id", "asset_path", "asset_sha256", "observation")
            }
            for figure in source["figure_checks"]
        ],
        "new_figure_reviews": [],
        "question_reviews": question_reviews,
        "outcome": "blocked",
    }

    evidence = load(evidence_path)

    prior_view = load(own / "view-input-preserved-revised.json")
    original_view = load(own / "view-input-preserved.json")

    assert project(previous) == prior_view
    assert reverse_replacements(project(draft), diff["view"]) == prior_view

    change = direction["changes"][0]
    assert change["pointer"] == "/questions/41/view/stem"
    assert previous["questions"][41]["view"]["stem"] == change["after"]
    initial = copy.deepcopy(previous)
    initial["questions"][41]["view"]["stem"] = change["before"]
    assert project(initial) == original_view
    assert previous["coverage"] == initial["coverage"] == draft["coverage"]
    frozen_rows = {row["question_id"]: row for row in frozen["question_reviews"]}
    source_rows = {row["source_number"]: row for row in source["solutions"]}
    coverage_rows = {row["source_number"]: row for row in draft["coverage"]}
    assert len(coverage_rows) == len(draft["coverage"]) == len(seed["exercises"]) == 34
    assert set(coverage_rows) == set(source_rows)
    procedure_failures = {
        (finding["question_id"], finding["option_id"]): finding
        for finding in evidence["procedure_findings"]
    }
    expected_wrong = {
        (question["id"], rationale["option_id"])
        for question in draft["questions"]
        for rationale in question["key"]["rationales"]
        if rationale["truth"] == "incorrect"
    }
    assert {
        (qid, oid)
        for qid, procedures in evidence["wrong_procedure_evidence"].items()
        for oid in procedures
    } == expected_wrong

    comparisons = []
    standard_questions = []
    for question in draft["questions"]:

        independent = frozen_rows[qid]
        number = independent["source_numbers_independently_mapped"][0]
        assert question["source_numbers"] == [number]
        assert source_rows[number]["status"] == "pass"
        assert set(question["key"]["correct"]) == set(independent["mathematics"]["correct"])
        assert question["intent"]["concept_blocks"] == []
        assert question["intent"]["limitations"] == coverage_rows[number]["omitted_tasks"]
        assert question["intent"]["unit_reason"] and question["intent"]["task"]

        counts["complete_keys_matching"] += 1
        counts["intent_and_source_solution_checks"] += 1
        counts["local_production_loss_checks"] += 1
        independent_options = {
            check["option_id"]: check for check in independent["mathematics"]["option_checks"]
        }
        options = {option["id"]: option for option in question["view"]["options"]}
        rationales = {rationale["option_id"]: rationale for rationale in question["key"]["rationales"]}
        assert set(options) == set(rationales) == set(independent_options)

        canonical_checks = []

            rationale = rationales[oid]
            frozen_option = independent_options[oid]

            assert option["text"] == frozen_option["visible_text"]
            assert option["figure_ids"] == frozen_option["visible_figure_ids"]

            assert (rationale["misconception"] is None) if correct else bool(rationale["misconception"])
            procedure = (
                "The actual declared correct derivation agrees with the unchanged "
                "independent content-level calculation: " + frozen_option["evidence"]
                if correct else evidence["wrong_procedure_evidence"][qid][oid]
            )

                    "actual_declared_procedure"
                ]

            counts["declared_derivations_checked"] += 1

                    "option_id": oid,

                "verdict": "pass",
                "evidence": evidence["source_solution_evidence"][number],

                "evidence": (

                    + json.dumps(question["intent"]["limitations"], ensure_ascii=True)
                    + " These limit recognition versus independent production of the "

            },
            "figure_observations_check": {"verdict": "pass", "evidence": figure_evidence},
            "complete_key_check": {
                "verdict": "pass",
                "current_author_correct": question["key"]["correct"],
                "independently_frozen_correct": independent["mathematics"]["correct"],
                "evidence": "The entire current key agrees with the already frozen content-rebound true set; not only one marked option.",

            "option_rationale_checks": option_checks,
            "task_fidelity": independent["task_fidelity"],
            "redundancy": independent["redundancy"],
            "answer_leakage": independent["answer_leakage"],
            "mathematics_verdict": math_verdict,
        }
        comparisons.append(comparison)
        standard_questions.append(
            {
                "question_id": qid,
                "task_fidelity": independent["task_fidelity"],
                "redundancy": independent["redundancy"],
                "answer_leakage": independent["answer_leakage"],
                "mathematics": {
                    "verdict": math_verdict,
                    "independent_solution": frozen_solution(independent),
                    "correct": independent["mathematics"]["correct"],
                    "option_checks": canonical_checks,

            }

    assert counts["rationales_checked"] == counts["rationale_truth_matches"] == 134
    assert counts["correct"] == 46 and counts["incorrect"] == 88
    assert counts["declared_wrong_procedures_checked"] == 88
    assert counts["declared_wrong_procedures_pass"] == 87
    assert counts["declared_wrong_procedures_revise"] == 1
    assert counts["declared_derivations_pass"] == 133

            row["question_id"] for row in frozen["question_reviews"]
            if number in row["source_numbers_independently_mapped"]

            assert current["question_ids"] == []
            assert current["omitted_tasks"] and current["reason"].startswith("Blocked:")
            explanation = evidence["special_scope_checks"]["blocked882"]
