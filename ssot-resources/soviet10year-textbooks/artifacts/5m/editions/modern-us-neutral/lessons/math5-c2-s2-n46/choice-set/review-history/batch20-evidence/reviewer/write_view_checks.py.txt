def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def binding(path):
    return {"path": str(path), "sha256": sha(path)}


def save_new(path, value):
    with path.open("x", encoding="utf-8") as handle:
        json.dump(value, handle, ensure_ascii=True, indent=2)
        handle.write("\n")


def main():
    for number, judgments, expected_counts in (
        ("46", N46, (15, 48, 18, 30)),
        ("47", N47, (44, 134, 46, 88)),
    ):
        own = TMP / f"reviewer-5m-n{number}"
        view_path = TMP / f"5m-n{number}/view.json"
        preserved_path = own / "view-input-preserved.json"
        fidelity_path = own / "view-fidelity-checks.json"

        view = json.loads(view_path.read_text())
        fidelity = json.loads(fidelity_path.read_text())
        source = json.loads(source_path.read_text())
        assert sha(view_path) == sha(preserved_path) == fidelity["view_binding"]["sha256"]
        assert sha(source_path) == fidelity["frozen_source_binding"]["sha256"]
        assert {q["id"] for q in view["questions"]} == {
            f"n{number}-{suffix}" for suffix in judgments
        }
        stage_questions = {q["question_id"]: q for q in fidelity["question_reviews"]}
        reviews = []
        truth_counts = Counter()
        single_positions = Counter()
        for question in view["questions"]:
            checked = judgments[question["id"].split("-", 1)[1]]
            option_ids = [option["id"] for option in question["options"]]
            assert set(option_ids) == set(checked["options"])
            assert set(checked["correct"]) <= set(option_ids)
            if question["mode"] == "single":
                assert len(checked["correct"]) == 1
                single_positions.update(checked["correct"])
            option_checks = []
            for option in question["options"]:
                truth = "correct" if option["id"] in checked["correct"] else "incorrect"
                truth_counts[truth] += 1
                option_checks.append({
                    "option_id": option["id"],
                    "truth": truth,
                    "evidence": checked["options"][option["id"]],
                    "visible_text": option["text"],
                    "visible_figure_ids": option["figure_ids"],
                })
            review = dict(stage_questions[question["id"]])
            review["mode"] = question["mode"]
            review["stem"] = question["stem"]
            review["figure_ids"] = question["figure_ids"]
            review["mathematics"] = {
                "verdict": "pass",
                "scope": "Independent visible-option truth only; no author key comparison authorized or performed.",
                "wrong_procedure_evidence_scope": "Any procedure described here is independently reconstructed from visible values, not attributed to an unread author rationale. Matching the author's actual declared procedure remains a later check.",
                "independent_solution": checked["solution"],
                "correct": checked["correct"],
                "option_checks": option_checks,
                "all_visible_options_independently_checked": True,
                "complete_true_set": True,
            }
            reviews.append(review)
        actual_counts = (
            len(reviews),
            sum(len(q["mathematics"]["option_checks"]) for q in reviews),
            truth_counts["correct"],
            truth_counts["incorrect"],

        image_evidence = json.loads(figure_record.read_text())
