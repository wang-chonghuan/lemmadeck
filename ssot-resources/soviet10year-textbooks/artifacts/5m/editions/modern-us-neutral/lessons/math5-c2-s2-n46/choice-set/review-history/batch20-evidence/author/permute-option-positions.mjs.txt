const ticket = '.intentfold/tickets/STEMROBIN-182/tmp'
const phase = process.argv[2]
assert(['prepare', 'apply'].includes(phase))
const byteHash = value => crypto.createHash('sha256').update(value).digest('hex')
const json = p => JSON.parse(fs.readFileSync(p, 'utf8'))
const saveNew = (p, v) => fs.writeFileSync(p, JSON.stringify(v, null, 2) + '\n', { flag: 'wx' })

function positionAudit(draft) {
  const sequence = draft.questions.filter(q => q.view.mode === 'single').map(q => q.key.correct[0])
  const periods = []
  for (let p = 1; p <= Math.min(8, Math.floor(sequence.length / 2)); p++) {
    if (sequence.every((v, i) => i < p || v === sequence[i - p])) periods.push(p)
  }
  let longest = 0
  let run = 0
  let previous = null
  for (const id of sequence) {
    run = id === 'A' || id === 'B'
      ? (previous !== null && (previous === 'A' || previous === 'B') && previous !== id ? run + 1 : 1)
      : 0
    longest = Math.max(longest, run)
    previous = id
  }
  return {
    single_sequence: sequence,
    counts: sequence.reduce((v, id) => ({ ...v, [id]: (v[id] ?? 0) + 1 }), {}),
    whole_sequence_periods_1_to_8: periods,
    longest_contiguous_A_B_alternating_run: longest
  }
}

function leafDiff(a, b, at = '', result = []) {
  if (JSON.stringify(a) === JSON.stringify(b)) return result
  if (a && b && typeof a === 'object' && typeof b === 'object') {
    assert.deepEqual(Object.keys(a), Object.keys(b))
    for (const k of Object.keys(a)) leafDiff(a[k], b[k], `${at}/${k}`, result)
  } else {
    result.push({ op: 'replace', path: at, old_value: a, value: b })
  }
  return result
}

      questions: before.questions.map(q => {
        const ids = q.view.options.map(o => o.id)
        const shuffled = [...ids]
        for (let i = shuffled.length - 1; i > 0; i--) {
          const j = crypto.randomInt(i + 1)
          ;[shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]]
        }
        return {
          question_id: q.id, old_order_ids: ids, new_order_old_ids: shuffled,
          old_content_hashes: q.view.options.map(({ id, ...value }) => contentHash(value))
        }
      })
    }
    saveNew(planPath, plan)
    const candidate = structuredClone(before)
    rebindOptionPositions(candidate.questions, planPath)
    console.log(JSON.stringify({ lesson, plan: planPath, before: positionAudit(before), after: positionAudit(candidate) }, null, 2))
    continue
  }

  const plan = json(planPath)
  assert.equal(plan.base_draft_byte_sha256, byteHash(beforeBytes))
  const currentAuthor = fs.readFileSync(`${dir}/author.mjs`, 'utf8')
  let normalizedAuthor = currentAuthor
    .replace("import { rebindOptionPositions } from './option-position-permutation.mjs'\n", '')
    .replace("rebindOptionPositions(questions, path.join(dir, 'option-position-permutation-plan.json'))\n\n", '')
  if (lesson === 47) {

    assert(currentAuthor.includes(`q('894','','single','${current894}',[`))
    normalizedAuthor = normalizedAuthor.replace(current894, '三个数依次相差0.7，算术平均数是8.9。按从第一到第三的顺序，这三个数是多少？')

  const after = structuredClone(before)
  rebindOptionPositions(after.questions, planPath)
  const mapping = before.questions.map((oldQ, i) => {
    const newQ = after.questions[i]
    const permutation = plan.questions[i]
    const oldById = new Map(oldQ.view.options.map(o => [o.id, o]))
    const newByOldId = new Map(permutation.new_order_old_ids.map((id, j) => [id, newQ.view.options[j]]))
    const oldRationales = new Map(oldQ.key.rationales.map(r => [r.option_id, r]))
    const newRationales = new Map(newQ.key.rationales.map(r => [r.option_id, r]))
    const inverseId = new Map([...newByOldId].map(([oldId, o]) => [o.id, oldId]))
    const normalized = structuredClone(newQ)
    normalized.view.options = oldQ.view.options.map(oldO => ({ ...newByOldId.get(oldO.id), id: oldO.id }))
    normalized.key.correct = newQ.key.correct.map(id => inverseId.get(id))
    normalized.key.rationales = oldQ.key.rationales.map(oldR => ({
      ...newRationales.get(newByOldId.get(oldR.option_id).id), option_id: oldR.option_id
    }))
    assert.deepEqual(normalized, oldQ, `${oldQ.id}: non-permutation content changed`)
    return {
      question_id: oldQ.id,
      old_order_ids: permutation.old_order_ids,
      new_order_old_ids: permutation.new_order_old_ids,
      old_correct_ids: oldQ.key.correct,
      new_correct_ids: newQ.key.correct,
      old_to_new: oldQ.view.options.map((oldO, j) => {

        const { id: oi, ...oldContent } = oldById.get(oldO.id)
        const { id: ni, ...newContent } = newO
        const { option_id: ori, ...oldReason } = oldRationales.get(oldO.id)
        const { option_id: nri, ...newReason } = newRationales.get(newO.id)
        assert.deepEqual(oldContent, newContent)
        assert.deepEqual(oldReason, newReason)
        assert.equal(oldQ.key.correct.includes(oldO.id), newQ.key.correct.includes(newO.id))
        return {
          old_id: oldO.id, old_position: j + 1,
          new_id: newO.id, new_position: newQ.view.options.findIndex(o => o.id === newO.id) + 1,
          option_content_sha256_before: contentHash(oldContent),
          option_content_sha256_after: contentHash(newContent),
          option_text_byte_sha256: byteHash(Buffer.from(oldO.text)),

          rationale_content_sha256_after: contentHash(newReason),

    serializer_executed: false,
