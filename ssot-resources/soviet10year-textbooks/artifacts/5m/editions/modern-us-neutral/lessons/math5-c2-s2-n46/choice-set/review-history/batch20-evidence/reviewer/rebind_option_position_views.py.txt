def sha256(data):
    return hashlib.sha256(data).hexdigest()


def binding(path, data):
    return {"path": str(path), "sha256": sha256(data)}


def content(option):
    # Match the entire allowed content subtree, never its previous label.
    return json.dumps(
        {key: value for key, value in option.items() if key != "id"},
        sort_keys=True,
        ensure_ascii=True,
        separators=(",", ":"),
    )


def alternating_run(sequence, start):
    if start + 1 == len(sequence) or sequence[start] == sequence[start + 1]:
        return 1
    end = start + 2
    while (
        end < len(sequence)
        and sequence[end] == sequence[end - 2]
        and sequence[end] != sequence[end - 1]
    ):
        end += 1
    return end - start


def position_stats(question_reviews):
    rows = [
        {"question_id": row["question_id"], "position": row["mathematics"]["correct"][0]}
        for row in question_reviews
        if row["mode"] == "single"
    ]
    sequence = [row["position"] for row in rows]
    runs = []
    start = 0
    for end in range(1, len(sequence) + 1):
        if end == len(sequence) or sequence[end] != sequence[start]:
            runs.append(
                {
                    "position": sequence[start],
                    "single_ordinals": [start + 1, end],
                    "length": end - start,
                }
            )
            start = end
    transition_counts = Counter(
        before + after for before, after in zip(sequence, sequence[1:])
    )
    return {
        "single_sequence": rows,
        "sequence_string": " ".join(sequence),
        "single_question_count": len(sequence),
        "position_counts": {label: sequence.count(label) for label in "ABC"},
        "ordered_transition_counts": dict(sorted(transition_counts.items())),
        "total_transition_count": len(sequence) - 1,
        "same_position_transition_count": sum(
            count for pair, count in transition_counts.items() if pair[0] == pair[1]
        ),
        "initial_strict_two_position_alternating_run_length": alternating_run(sequence, 0),
        "longest_strict_two_position_alternating_run_length": max(
            alternating_run(sequence, index) for index in range(len(sequence))
        ),
        "longest_same_position_run_length": max(run["length"] for run in runs),
        "same_position_runs": runs,
    }


def build(number, config):

        q["id"] for q in new_view["questions"]

    frozen_rows = {row["question_id"]: row for row in old_checks["question_reviews"]}
    assert set(frozen_rows) == {question["id"] for question in old_view["questions"]}
    rows = []

            "id", "mode", "stem", "figure_ids", "options"
        }

            key: value for key, value in after.items() if key != "options"
        }, question_id
        assert all(
            set(option) == {"id", "text", "figure_ids"}
            for option in before["options"] + after["options"]
        )
        assert [option["id"] for option in before["options"]] == [
            option["id"] for option in after["options"]
        ]
        before_contents = Counter(content(option) for option in before["options"])
        after_contents = Counter(content(option) for option in after["options"])
        assert before_contents == after_contents, question_id
        assert all(count == 1 for count in before_contents.values()), question_id
        old_options = {content(option): option for option in before["options"]}
        frozen = frozen_rows[question_id]
        assert frozen["stem"] == before["stem"]
        assert frozen["mode"] == before["mode"]
        assert frozen["figure_ids"] == before["figure_ids"]
        assert frozen["mathematics"]["verdict"] == "pass"
        old_option_checks = {
            check["option_id"]: check
            for check in frozen["mathematics"]["option_checks"]
        }
        assert set(old_option_checks) == {option["id"] for option in before["options"]}
        frozen_true = {
            check["option_id"]
            for check in old_option_checks.values()
            if check["truth"] == "correct"
        }
        assert frozen_true == set(frozen["mathematics"]["correct"])
        assert frozen_true == set(old_checks["complete_true_sets"][question_id])

        row = copy.deepcopy(frozen)
        new_checks = []
        rebindings = []
        for new_ordinal, option in enumerate(after["options"], start=1):
            old_option = old_options[content(option)]
            old_check = old_option_checks[old_option["id"]]
            assert old_check["visible_text"] == old_option["text"]
            assert old_check["visible_figure_ids"] == old_option["figure_ids"]
            new_check = copy.deepcopy(old_check)
            new_check["option_id"] = option["id"]
            new_check["visible_text"] = option["text"]
            new_check["visible_figure_ids"] = option["figure_ids"]
            new_check["truth_reused_by_exact_content"] = True

                    "visible_figure_ids": option["figure_ids"],

                    "exact_content_match": True,
                }
            )
        correct = [
            check["option_id"] for check in new_checks if check["truth"] == "correct"
        ]
        assert correct and (after["mode"] != "single" or len(correct) == 1)

pending = [build(number, config) for number, config in CONFIG.items()]
for record_path, preserved_path, _, _, _ in pending:
    assert not record_path.exists(), f"Refuse to overwrite {record_path}"
    assert not preserved_path.exists(), f"Refuse to overwrite {preserved_path}"
for record_path, preserved_path, new_view_bytes, record, protected in pending:
    with preserved_path.open("xb") as output:
        output.write(new_view_bytes)
    with record_path.open("x", encoding="utf-8") as output:
        json.dump(record, output, ensure_ascii=True, indent=2)
        output.write("\n")
    for prior in protected:
        assert sha256(Path(prior["path"]).read_bytes()) == prior["sha256"]
    print(
        json.dumps(
            {
                "saved": binding(record_path, record_path.read_bytes()),
                "view_preserved": binding(preserved_path, preserved_path.read_bytes()),
                "position_repair_stage_status": record["position_repair_stage_status"],
                "aggregate_outcome": record["outcome"],
                "reused_option_count": record["visible_option_judgments_reused"],
                "true_false_counts": [
                    record["independently_true_option_count"],
                    record["independently_false_option_count"],
                ],
                "position_counts": record["position_distribution_notice"]["position_counts"],
            }
        )
    )
