ROOT = Path(__file__).resolve().parents[5]
TMP = Path(__file__).resolve().parent.parent
CHECKS = {"46": [], "47": []}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def mixed(whole, numerator, denominator):
    return F(whole) + F(numerator, denominator)


def check(lesson, source_number, values, expected):
    normalized = [F(value) for value in values]
    wanted = [F(value) for value in expected]
    assert normalized == wanted, (lesson, source_number, normalized, wanted)
    CHECKS[lesson].append({
        "source_number": str(source_number),
        "exact_results_checked": [str(value) for value in normalized],
        "status": "pass",
    })

def save_json(path, data):
    with path.open("x") as handle:
        json.dump(data, handle, indent=2, ensure_ascii=True)
        handle.write("\n")


def binding(path):
    return {"path": str(path.relative_to(ROOT)), "sha256": sha(path)}


def preserve_copy(source, target):
    with target.open("xb") as handle:
        handle.write(source.read_bytes())
    assert sha(source) == sha(target)
    return {"original": binding(source), "preserved": binding(target), "byte_identical": True}


def main():
    arithmetic()
    for number in ("46", "47"):
        own = TMP / f"reviewer-5m-n{number}"
        source_path = TMP / f"5m-n{number}/source.json"
        source = json.loads(source_path.read_text())
        reasoning_path = own / "source-reasoning.json"
        result = json.loads(reasoning_path.read_text())
        assert source["source"]["lesson"] == result["lesson"]
        assert [item["number"] for item in source["exercises"]] == [
            item["source_number"] for item in result["solutions"]
        ]
        assert len(result["solutions"]) == result["source_exercise_count"]
        provenance = [binding(source_path)]
        for item in source["trace"]:
            file = ROOT / item["path"]
            assert sha(file) == item["sha256"], f"Allowed source snapshot changed: {file}"
            provenance.append(binding(file))
        original_dir = ROOT / "ssot-resources/soviet10year-textbooks/artifacts/5m/lessons"
        for filename in ("lesson.json", "exercises.json"):
            provenance.append(binding(original_dir / result["lesson"] / filename))
        definitions = ["math5-c1-s2-n12", "math5-c2-s1-n38"] if number == "46" else [
            "math5-c1-s2-n15", "math5-c1-s2-n11", "math5-c1-s1-n2",
        ]
        provenance.extend(binding(original_dir / lesson / "lesson.json") for lesson in definitions)
        preserved_source = preserve_copy(source_path, own / "source-input-preserved.json")
        calculation = {
            "scope": "Independent rational arithmetic verification; not structural/canonical validation",
            "status": "pass", "checks": CHECKS[number],
            "symbolic_domains_and_geometric_conventions": "Reasoned separately in source-solutions; no convention inferred by this arithmetic program.",
        }
        save_json(own / "arithmetic-checks.json", calculation)
        result["source_file"] = binding(source_path)
        result["source_input_preservation"] = preserved_source
        result["allowed_input_bindings"] = provenance
        result["independent_reasoning_binding"] = binding(reasoning_path)
        result["arithmetic_verification"] = binding(own / "arithmetic-checks.json")
        result["prohibited_material_read"] = False
        result["structural_or_canonical_validator_run"] = False
        result["render_or_browser_run"] = False
        if number == "47":
            figure = source["figures"][0]
            assert figure["id"] == "fig-151"
            assets = []
            for asset in figure["assets"]:
                asset_path = ROOT / asset["path"]
                assert sha(asset_path) == asset["sha256"]
                assets.append(binding(asset_path))
            supplied = TMP / "5m-n47/source-assets"
            evidence_path = supplied / "render-evidence.json"
            evidence = json.loads(evidence_path.read_text())[0]
            assert evidence["asset"] == assets[0]["path"]
            assert evidence["sha256"] == assets[0]["sha256"]
            assets_dir = own / "source-assets"
            assets_dir.mkdir(exist_ok=True)
            png_source = supplied / "fig-151-browser.png"
            preservation = [
                preserve_copy(png_source, assets_dir / "fig-151-browser.png"),
                preserve_copy(ROOT / assets[1]["path"], assets_dir / "fig-151.artwork.png"),
            ]
            image_observations = []
            for note, copied in zip(result["figure_observation_notes"], preservation):
                viewed_path = ROOT / note["actual_opened_file"]
                with Image.open(viewed_path) as image:
                    image.load()
                    size = list(image.size)
                image_observations.append({
                    **note, "status": "pass", "scope": "source asset observation only",
                    "observer": "independent_agent", "actual_opened_file": str(viewed_path),
                    "sha256": sha(viewed_path), "size": size, "preserved_copy": copied,
                })
            image_record = {
                "lesson": result["lesson"], "figure_id": "fig-151",
                "status": "pass", "scope": "actually opened source layers; not product display or new-figure approval",
                "assets": assets, "rasterization_evidence": binding(evidence_path),
                "rasterization_evidence_is_not_author_visual_approval": True,
                "actually_opened_image_count": 2, "observations": image_observations,
                "full_two_layer_composite_actually_opened": False,
            }
            save_json(own / "source-image-observations.json", image_record)
            result["figure_checks"] = [{
                "figure_id": "fig-151", "asset_path": assets[0]["path"],
                "asset_sha256": assets[0]["sha256"],
                "observation": image_observations[0]["observation"],
                "actually_opened_raster": binding(png_source),
                "artwork_asset_opened": assets[1],
                "image_observations": binding(own / "source-image-observations.json"),
                "source_layer_observation_only": True,
            }]
        else:
            save_json(own / "source-image-observations.json", {
                "lesson": result["lesson"], "status": "not_applicable",
                "evidence": "The allowed source package has no referenced figures.",
                "actually_opened_image_count": 0, "observations": [],
            })
        result["source_image_observation_record"] = binding(own / "source-image-observations.json")
        save_json(own / "source-solutions.json", result)
        print(json.dumps({
            "saved": str(own / "source-solutions.json"), "status": result["status"],
            "source_exercises_addressed": len(result["solutions"]),
            "convention_blocked_sources": result["unresolved_source_numbers"],
            "arithmetic_check_groups": len(CHECKS[number]),
            "new_browser_or_render_run": False,
        }))


if __name__ == "__main__":
    main()
