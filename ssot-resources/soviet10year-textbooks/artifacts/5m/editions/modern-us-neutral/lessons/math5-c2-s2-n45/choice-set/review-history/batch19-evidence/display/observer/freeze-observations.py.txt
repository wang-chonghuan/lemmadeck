def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

bands = {key: [] for key in geometry}
for path, image in image_records.items():
    name = Path(path).name
    for combination in geometry:
        if f"-{combination}-" not in name:
            continue
        height = 960 if combination.endswith("1440") else 844
        if name.endswith("-first.png"):
            bands[combination].append((0, height, path))
        elif name.endswith("-last.png"):
            h = geometry[combination]["height"]
            bands[combination].append((h - height, h, path))
for entry in capture["captures"]:
    if entry["uncovered"]["kind"] == "middle-document-band":
        a, b = entry["uncovered"]["document_y_interval"]
        bands[entry["combination"]].append((a, b, entry["path"]))


def images_for(combination, rect):
    a, b = rect["y"], rect["y"] + rect["height"]
    result = []
    for start, end, path in bands[combination]:
        if start < b and end > a:
            result.append({
                "png": path,
                "sha256": image_records[path]["sha256"],
                "montage": image_records[path]["montage_path"],
                "observed_document_y_interval": [max(a, start), min(b, end)],
            })
    for entry in capture["captures"]:
        r = entry["rect"]
        if (entry["combination"] == combination
                and entry["uncovered"]["kind"] == "formula-right-end"
                and r["y"] < b and r["y"] + r["height"] > a):
            result.append({
                "png": entry["path"],
                "sha256": entry["sha256"],
                "montage": image_records[entry["path"]]["montage_path"],
                "observed_formula_source_x_interval": entry["uncovered"]["source_x_interval"],
                "math_index": entry["uncovered"]["math_index"],
            })
    return result
