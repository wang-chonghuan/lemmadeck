const falseChecks = [];
const correctChecks = [];
const setChecks = [];
for (const q of draft.questions) {
  const previous = frozen.question_reviews.find((item) => item.question_id === q.id);
  const sameSet = JSON.stringify([...q.key.correct].sort()) ===
    JSON.stringify([...previous.mathematics.correct].sort());
  if (!sameSet) throw new Error(`Author set disagrees with frozen set:${q.id}`);
  setChecks.push({
    question_id: q.id,
    verdict: 'pass',
    independently_frozen_correct: previous.mathematics.correct,
    author_correct: q.key.correct,
    evidence: previous.mathematics.independent_solution,
  });
  for (const rationale of q.key.rationales) {
    const truth = previous.mathematics.option_checks.find(
      (item) => item.option_id === rationale.option_id,
    );
    if (rationale.truth !== truth.truth) throw new Error(`Truth mismatch:${q.id}/${rationale.option_id}`);
    const common = {
      question_id: q.id,
      option_id: rationale.option_id,
      verdict: 'pass',
      author_derivation: rationale.derivation,
      author_misconception: rationale.misconception,
      independently_frozen_evidence: truth.evidence,
    };
    if (rationale.truth === 'incorrect') {
      const generation = programs[q.id]?.[rationale.option_id];
      if (!generation) throw new Error(`Missing individual adjudication:${q.id}/${rationale.option_id}`);
      falseChecks.push({
        ...common,
        generation,
        falsehood: truth.evidence,
        comparison: 'Current stated erroneous rule and its derivation generate the displayed alternative, including any correct unchanged components; frozen evidence establishes why the whole alternative is false.',
      });
    } else {
      correctChecks.push({
        ...common,
        proof_comparison: truth.evidence,
        comparison: 'Current correct derivation was individually read against this independently frozen proof/result, the source solution and requested method/domain. Correct components do not rely on author source-solutions or a key-derived mathematical solution.',
      });
    }
  }
}
