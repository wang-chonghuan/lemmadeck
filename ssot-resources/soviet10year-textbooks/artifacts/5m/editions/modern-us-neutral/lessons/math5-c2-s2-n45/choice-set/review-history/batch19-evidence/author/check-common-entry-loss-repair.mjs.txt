import fs from "node:fs";
import crypto from "node:crypto";
import { spawnSync } from "node:child_process";

const root = ".intentfold/tickets/STEMROBIN-182/tmp/5m-n45";

const after = JSON.parse(fs.readFileSync(`${root}/draft.json`, "utf8"));
const replacements = [];
function compare(a, b, path = "") {
  if (JSON.stringify(a) === JSON.stringify(b)) return;
  if (Array.isArray(a) && Array.isArray(b) && !path.endsWith("limitations") && !path.endsWith("omitted_tasks")) {
    assert.equal(a.length, b.length);
    a.forEach((v, i) => compare(v, b[i], `${path}/${i}`));
  } else if (a && b && !Array.isArray(a) && typeof a === "object" && typeof b === "object") {
    assert.deepEqual(Object.keys(a), Object.keys(b));
    Object.keys(a).forEach(k => compare(a[k], b[k], `${path}/${k}`));
  } else {
    replacements.push({ op: "replace", path, value: b, previous_value: a });
  }
}
