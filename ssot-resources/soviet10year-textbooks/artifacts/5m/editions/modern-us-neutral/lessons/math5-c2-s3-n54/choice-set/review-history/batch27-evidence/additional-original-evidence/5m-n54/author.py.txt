def token(value):
    if isinstance(value, str):
        return value
    value = F(value)
    if value.denominator == 1:
        return str(value.numerator)
    sign = "-" if value < 0 else ""
    return sign + r"\frac{" + str(abs(value.numerator)) + "}{" + str(value.denominator) + "}"


def render(values, unit=""):
    if isinstance(values, str):
        return values
    if len(values) == 1:
        return f"${token(values[0])}$" + (f" {unit}" if unit else "")
    return "; ".join(f"{index}) ${token(value)}$" + (f" {unit}" if unit else "")
                     for index, value in enumerate(values, 1))

def add(number, title, correct, wrong1, wrong2, program1, program2, stem=None,
        unit="", losses=None, task=None):
    number = str(number)
    original = by_number[number]
    correct_id = positions[len(questions)]
    wrong_ids = [label for label in "ABC" if label != correct_id]
    candidates = {
        correct_id: (render(correct, unit), "correct", original["independent_derivation"], None),
        wrong_ids[0]: (render(wrong1, unit), "incorrect", program1, program1),
        wrong_ids[1]: (render(wrong2, unit), "incorrect", program2, program2),
    }
    qid = f"n54-{number}-{title}"
    questions.append({
        "id": qid,
        "source_numbers": [number],
        "view": {
            "mode": "single", "stem": stem or prompt(number),
            "figure_ids": [],
            "options": [{"id": label, "text": candidates[label][0], "figure_ids": []} for label in "ABC"],
        },
        "intent": {
            "task": task or original["whole_original_task"],
            "source_solution": original["independent_derivation"],
            "unit_reason": "One complete response preserves all ordered subcases and the common mathematical task; no separate answer-blank clicks.",
            "concept_blocks": [0],
            "limitations": losses if losses is not None else original["production_losses_for_later_choice"] + original["notices"],
            "figure_observations": [],
        },
        "key": {
            "correct": [correct_id],
            "rationales": [{
                "option_id": label, "truth": candidates[label][1],
                "derivation": candidates[label][2], "misconception": candidates[label][3],
            } for label in "ABC"],
        },
    })
    programs.append({
        "question_id": qid,
        "execution_only_not_semantic_approval": True,
        "false_programs": [{
            "option_id": label, "executed_output": candidates[label][0],
            "operations": candidates[label][2],
        } for label in wrong_ids],
    })


def multiplication_errors(pairs):
    wrong1, wrong2 = [], []
    descriptions1, descriptions2 = [], []
    for index, (whole, numerator, denominator, multiplier) in enumerate(pairs):
        frac = F(numerator, denominator)
        value = whole + frac
        if index % 2 == 0:
            wrong1.append(whole + frac * multiplier)
            wrong2.append((whole + numerator) * multiplier)
            descriptions1.append(f"{index+1}: {whole}+({numerator}/{denominator})*{multiplier}")
            descriptions2.append(f"{index+1}: ({whole}+{numerator})*{multiplier}, dropping fractional denominator")
        else:
            wrong1.append(whole * multiplier + frac)
            wrong2.append(value / multiplier)
            descriptions1.append(f"{index+1}: {whole}*{multiplier}+{numerator}/{denominator}")
            descriptions2.append(f"{index+1}: ({whole}+{numerator}/{denominator})/{multiplier}, division instead of multiplication")
    return wrong1, wrong2, "Only one mixed-number part multiplied: " + "; ".join(descriptions1), "; ".join(descriptions2)
