"""Execute only the authorized 1043 arithmetic and a 1046 midpoint example."""

from fractions import Fraction as F
import json


def record(expression, result, offered, correct):
    assert result == F(offered), (expression, result, offered)
    assert result != F(correct), (expression, result, correct)
    return {
        "expression": expression,
        "executed_result": str(result),
        "offered_result": offered,
        "source_result": correct,
        "equals_offered": True,
        "false_under_source": True,
    }


mixed = 1 + F(25, 42)
bracket = 2 - mixed
product = F(7, 6) * bracket
first = F(11, 9) + product
assert (mixed, bracket, product, first) == (
    F(67, 42), F(17, 42), F(17, 36), F(61, 36)
)

# The last three true results are frozen source values, not a new source solve.
true_results = ["61/36", "2", "30", "13/12"]
a_cases = [
    ("(11/9+7/6)*(2-(1+25/42))",
     (F(11, 9) + F(7, 6)) * (2 - (1 + F(25, 42))), "731/756"),
    ("(4-2)*(10-8)", F((4 - 2) * (10 - 8)), "4"),
    ("(4+(5+1/6))/(1/3+1/36)",
     (4 + (5 + F(1, 6))) / (F(1, 3) + F(1, 36)), "330/13"),
    ("(6+5/12)/(2+3/4)+(11+1/4)*(1/9)",
     (6 + F(5, 12)) / (2 + F(3, 4)) + (11 + F(1, 4)) * F(1, 9),
     "43/12"),
]
c_cases = [
    ("11/9+(7/6)*(2-25/42)",
     F(11, 9) + F(7, 6) * (2 - F(25, 42)), "103/36"),
    ("(4-7/15)*(10-16/23)",
     (4 - F(7, 15)) * (10 - F(16, 23)), "11342/345"),
    ("(4+(5+1/6))*(1/3-1/36)",
     (4 + (5 + F(1, 6))) * (F(1, 3) - F(1, 36)), "605/216"),
    ("(6+5/12)*(2+3/4)-(11+1/4)*(1/9)",
     (6 + F(5, 12)) * (2 + F(3, 4)) - (11 + F(1, 4)) * F(1, 9),
     "787/48"),
]

wrong = {
    option: [record(expression, value, offered, true_results[index])
             for index, (expression, value, offered) in enumerate(cases)]
    for option, cases in [("A", a_cases), ("C", c_cases)]
}
midpoint = (F(7, 4), F(7, 4))
assert sum(midpoint) == F(7, 2)
assert midpoint != (F(0), F(0))
assert F(7, 2) ** 2 + F(7, 2) ** 2 < F(7) ** 2

print(json.dumps({
