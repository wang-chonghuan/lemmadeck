const pointerPart = (key) => String(key).replace(/~/g, '~0').replace(/\//g, '~1');

// Only an exact structured repair delta; this does not interpret unchanged mathematics.
const changes = (before, after, pointer = '') => {
  if (isDeepStrictEqual(before, after)) return [];
  if (Array.isArray(before) && Array.isArray(after)) {
    requireFact(before.length === after.length, `Out-of-scope array-length change: ${pointer}`);
    return before.flatMap((value, index) => changes(value, after[index], `${pointer}/${index}`));
  }
  if (before !== null && after !== null
      && typeof before === 'object' && typeof after === 'object'
      && !Array.isArray(before) && !Array.isArray(after)) {
    const keys = Object.keys(before).sort();
    requireFact(isDeepStrictEqual(keys, Object.keys(after).sort()),
      `Out-of-scope object-field change: ${pointer}`);
    return keys.flatMap((key) => changes(before[key], after[key], `${pointer}/${pointerPart(key)}`));
  }
  return [{ pointer, before, after }];
};
