import json
from pathlib import Path
import subprocess
import sys
from datetime import datetime, timezone


out = Path(sys.argv[1]).resolve()
stage = sys.argv[2]
command = sys.argv[3:]
record = out / f"{stage}-command.json"
assert not record.exists(), f"Do not repeat {stage}"
out.mkdir(parents=True, exist_ok=True)
started = datetime.now(timezone.utc).isoformat()
result = subprocess.run(command, text=True, capture_output=True)
value = {
    "command": command,
    "cwd": str(Path.cwd()),
    "execution_count": 1,
    "started": started,
    "finished": datetime.now(timezone.utc).isoformat(),
    "exit_code": result.returncode,
    "stdout": result.stdout,
    "stderr": result.stderr,
    "semantic_approval": False,
    "model": None,
    "usage": None,
}
record.write_text(json.dumps(value, ensure_ascii=True, indent=2) + "\n")
if stage == "inspect" and result.returncode == 0:
    inventory = json.loads(result.stdout)
    target = out / "inspect.json"
    assert not target.exists()
    target.write_text(json.dumps(inventory, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps(inventory, ensure_ascii=False, indent=2))
else:
    print(result.stdout, end="")
    print(result.stderr, end="", file=sys.stderr)
raise SystemExit(result.returncode)
