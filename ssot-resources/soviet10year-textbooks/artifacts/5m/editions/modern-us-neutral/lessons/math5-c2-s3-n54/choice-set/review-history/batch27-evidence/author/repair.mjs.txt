function canonical(value) {
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  if (value && typeof value === "object") {
    return `{${Object.keys(value).sort().map((key) =>
      `${escapeAscii(JSON.stringify(key))}:${canonical(value[key])}`).join(",")}}`;
  }
  return escapeAscii(JSON.stringify(value));
}

function command(argv) {
  const started = new Date().toISOString();
  const result = spawnSync(argv[0], argv.slice(1), {
    cwd: root, encoding: "utf8", maxBuffer: 16 * 1024 * 1024,
  });
  return {
    command: argv, cwd: root, execution_count: 1,
    started, finished: new Date().toISOString(),
    exit_code: result.status, stdout: result.stdout ?? "", stderr: result.stderr ?? "",
    signal: result.signal, spawn_error: result.error?.message ?? null,
    semantic_approval: false,
  };
}
function changes(before, after, pointer = "") {
  if (Object.is(before, after)) return [];
  if (Array.isArray(before) && Array.isArray(after)) {
    assert.equal(before.length, after.length, `array length changed at ${pointer}`);
    return before.flatMap((value, index) => changes(value, after[index], `${pointer}/${index}`));
  }
  if (before && after && typeof before === "object" && typeof after === "object") {
    assert.deepEqual(Object.keys(before), Object.keys(after), `fields changed at ${pointer}`);
    return Object.keys(before).flatMap((key) => changes(before[key], after[key],
      `${pointer}/${key.replace(/~/g, "~0").replace(/\//g, "~1")}`));
  }
  return [{ pointer, before, after }];
}
function exactChanges(before, after, allowed) {
  const result = changes(before, after);
  assert.deepEqual(result.map((row) => row.pointer).sort(), [...allowed].sort());
  return result;
}
