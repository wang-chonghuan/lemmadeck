const recordDelta = (before, after, pointer = '') => {
  if (isDeepStrictEqual(before, after)) return [];
  if (Array.isArray(before) && Array.isArray(after) && before.length === after.length) {
    return before.flatMap((item, index) => recordDelta(item, after[index], `${pointer}/${index}`));
  }
  if (before !== null && after !== null
      && typeof before === 'object' && typeof after === 'object'
      && !Array.isArray(before) && !Array.isArray(after)
      && isDeepStrictEqual(Object.keys(before).sort(), Object.keys(after).sort())) {
    return Object.keys(before).flatMap((key) =>
      recordDelta(before[key], after[key], `${pointer}/${key}`));
  }
  return [{ pointer, before, after }];
};
