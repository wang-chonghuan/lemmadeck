async function inspect(page, selector, spec) {
  return page.evaluate(({ selector, spec }) => {
    const svg = document.querySelector(`${selector} svg`)
    const objects = new Map(spec.objects.map(object => [object.id, object]))
    const p = objects.get(`f${spec.id.slice(4)}-P`).at
    const paths = [...svg.querySelectorAll('path')]
    const visible = node => {
      const style = getComputedStyle(node)
      return style.display !== 'none' && style.visibility !== 'hidden' && Number(style.opacity) !== 0
    }
    const nodeFor = id => [...svg.querySelectorAll('[id]')].find(node => node.id.endsWith(`_ld-${id}`))
    const smallObject = objects.get(`f${spec.id.slice(4)}-small-orbit`)
    const small = paths.find(node => node.getAttribute('d') === smallObject.d)
    const radius = Math.hypot(
      objects.get(`f${spec.id.slice(4)}-earth-center`).at[0] - p[0],
      objects.get(`f${spec.id.slice(4)}-earth-center`).at[1] - p[1],
    )
    const length = small.getTotalLength()
    const radiusDeviation = Math.max(...Array.from({ length: 65 }, (_, index) => {
      const point = small.getPointAtLength(length * index / 64)
      return Math.abs(Math.hypot(point.x - p[0], spec.canvas.height - point.y - p[1]) - radius)
    }))
    const texts = [...svg.querySelectorAll('text')].map(node => {
      const matrix = node.getScreenCTM()
      const box = node.getBBox()
      return {
        text: node.textContent,
        px: parseFloat(node.getAttribute('font-size')) * Math.hypot(matrix.a, matrix.b),
        box: { x: box.x, y: box.y, width: box.width, height: box.height },
        color: getComputedStyle(node).fill,
      }
    })
    const motions = ['large', 'small'].map(name => {
      const object = objects.get(`f${spec.id.slice(4)}-motion-${name}`)
      const node = paths.find(item => item.getAttribute('d') === object.d)
      const length = node.getTotalLength()
      const start = node.getPointAtLength(0)
      const end = node.getPointAtLength(length)
      const angles = [start, end].map(point =>
        Math.atan2(spec.canvas.height - point.y - p[1], point.x - p[0]))
      const headId = `f${spec.id.slice(4)}-motion-${name}-head`
      const arrowhead = [...svg.querySelectorAll('[data-arrowhead-for]')]
        .find(item => item.getAttribute('data-arrowhead-for').endsWith(`_ld-${headId}`))
      return {
        name,
        angularChange: angles[1] - angles[0],
        explicitArrowhead: Boolean(arrowhead && visible(arrowhead)),
        arrowheadFill: arrowhead ? getComputedStyle(arrowhead).fill : null,
      }
    })
    const construction = spec.objects
      .filter(object => /motion-(large|small)-(tip|tail)$/.test(object.id))
      .map(object => ({ id: object.id, visible: Boolean(nodeFor(object.id) && visible(nodeFor(object.id))) }))
    const points = ['P', 'earth-center', 'moon-center', 'rmoon-end'].map(name => {
      const node = nodeFor(`f${spec.id.slice(4)}-${name}`)
      return { name, visible: Boolean(node && visible(node)), x: node?.getAttribute('cx'), y: node?.getAttribute('cy') }
    })
    const media = document.querySelector(`${selector} .sr-figure-media`)
    const host = document.querySelector(`${selector} .sr-read-fig`)
    const maxScroll = host ? host.scrollWidth - host.clientWidth : 0
    if (host) host.scrollLeft = maxScroll
    return {
      svgWidth: svg.getBoundingClientRect().width,
      svgHeight: svg.getBoundingClientRect().height,
      mediaWidth: media?.getBoundingClientRect().width ?? null,
      hostWidth: host?.clientWidth ?? null,
      maxScroll,
      scrollEnd: host?.scrollLeft ?? null,
      rightEdgeAtScrollEnd: host ? media.getBoundingClientRect().right <= host.getBoundingClientRect().right + 1 : null,
      minTextPx: Math.min(...texts.map(text => text.px)),
      texts,
      points,
      construction,
      smallOrbit: {
        dash: small.getAttribute('stroke-dasharray'),
        fill: small.getAttribute('fill'),
        maximumRadiusDeviation: radiusDeviation,
      },
      motions,
      arrowheads: svg.querySelectorAll('[data-arrowhead-for]').length,
    }
  }, { selector, spec })
}

function check(metrics) {
  assert.equal(metrics.svgWidth, 640)
  assert.equal(metrics.svgHeight, 640)
  assert.ok(metrics.minTextPx >= 16)
  assert.ok(metrics.points.every(point => point.visible))
  assert.ok(metrics.construction.every(point => !point.visible))
  assert.ok(metrics.smallOrbit.dash)
  assert.equal(metrics.smallOrbit.fill, 'none')
  assert.ok(metrics.smallOrbit.maximumRadiusDeviation < 0.02)
  assert.ok(metrics.motions.every(motion => motion.angularChange > 0 && motion.explicitArrowhead))
  assert.equal(metrics.arrowheads, 3)
  if (metrics.mediaWidth !== null) {
    assert.equal(metrics.mediaWidth, 640)
    assert.equal(metrics.scrollEnd, metrics.maxScroll)
    assert.equal(metrics.rightEdgeAtScrollEnd, true)
  }
}
