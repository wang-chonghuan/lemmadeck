import assert from 'node:assert/strict'
import crypto from 'node:crypto'
import fs from 'node:fs'
import path from 'node:path'

const root = process.cwd()
assert.equal(root, '/Users/yong/work/lemmadeck-ws/lemmadeck--STEMROBIN-180')
const lessonPath = 'ssot-resources/soviet10year-textbooks/artifacts/8p/editions/modern-us-neutral/lessons/phy8-c2-s1-n26/lesson.json'
const lessonBytes = fs.readFileSync(lessonPath)
const lesson = JSON.parse(lessonBytes)
const fullText = lesson.prose.filter(block => block.kind === 'p').map(block => ({
  lesson: lesson.id,
  text: block.text,
}))
const hash = value => crypto.createHash('sha256').update(value).digest('hex')
const out = (directory, name, value) => {
  const destination = path.join(directory, name)
  assert.equal(fs.existsSync(destination), false, `preserve existing ${destination}`)
  fs.writeFileSync(destination, `${JSON.stringify(value, null, 2)}\n`)
}
const P = [310, 310]
const E = [270, 270]
const bigRadius = 245
const smallRadius = Math.hypot(E[0] - P[0], E[1] - P[1])
const M = [P[0] + bigRadius / Math.sqrt(2), P[1] + bigRadius / Math.sqrt(2)]
const polar = (radius, angle) => [
  P[0] + radius * Math.cos(angle * Math.PI / 180),
  P[1] + radius * Math.sin(angle * Math.PI / 180),
]
const screen = at => [at[0], 640 - at[1]]
const numbers = at => at.map(value => Number(value.toFixed(8))).join(' ')
const text = (id, at, value, placement = 'CENTER', fontSize = 22) => ({
  id, type: 'text', at, text: value, fontSize, fontWeight: 500,
  labelPlacement: { position: placement, gap: 12 },
  labelColor: 'ink',
})
const point = (id, at, size = 4) => ({
  id, type: 'point', at, size, stroke: 'ink', fill: 'ink', strokeWidth: 1,
})
const relationship = (id, type, fields) => ({ id, type, ...fields })

for (const number of [80, 81]) {
  const prefix = `f${number}`
  const directory = `.tmp/s10y-image/8p-n26/fig-${number}`
  const context = JSON.parse(fs.readFileSync(path.join(directory, 'context.json'), 'utf8'))
  const sourcePath = `ssot-resources/soviet10year-textbooks/artifacts/8p/figures/fig-${number}.png`
  assert.equal(context.sourceImage.path, sourcePath)
  assert.equal(context.sourceImage.sha256, hash(fs.readFileSync(sourcePath)))
  const sphere = number === 80
  const objects = [
    point(`${prefix}-P`, P),
    point(`${prefix}-earth-center`, E),
    point(`${prefix}-moon-center`, M, sphere ? 2 : 4),
    point(`${prefix}-rmoon-end`, [P[0] + bigRadius, P[1]], 1),
    {
      id: `${prefix}-large-orbit`, type: 'circle', center: `${prefix}-P`,
      radius: bigRadius, stroke: 'ink', strokeWidth: 2.5, fillOpacity: 0,
    },
  ]
  const assertions = [
    relationship(`${prefix}-earth-on-small`, 'pointOnCircle', {
      point: `${prefix}-earth-center`, center: `${prefix}-P`, radius: smallRadius,
    }),
    relationship(`${prefix}-moon-on-large`, 'pointOnCircle', {
      point: `${prefix}-moon-center`, center: `${prefix}-P`, radius: bigRadius,
    }),
    relationship(`${prefix}-rmoon-end-on-large`, 'pointOnCircle', {
      point: `${prefix}-rmoon-end`, center: `${prefix}-P`, radius: bigRadius,
    }),
    relationship(`${prefix}-earth-lower-left`, 'displacement', {
      from: `${prefix}-P`, to: `${prefix}-earth-center`, dx: -40, dy: -40,
    }),
    relationship(`${prefix}-moon-upper-right`, 'displacement', {
      from: `${prefix}-P`, to: `${prefix}-moon-center`,
      dx: M[0] - P[0], dy: M[1] - P[1],
    }),
    relationship(`${prefix}-rmoon-horizontal`, 'displacement', {
      from: `${prefix}-P`, to: `${prefix}-rmoon-end`, dx: bigRadius, dy: 0,
    }),
    relationship(`${prefix}-opposite-rays`, 'collinear', {
      a: `${prefix}-earth-center`, b: `${prefix}-P`, c: `${prefix}-moon-center`,
    }),
  ]
  if (sphere) {
    const spherePoints = Array.from({ length: 96 }, (_, index) => {
      const angle = index * Math.PI / 48
      return [E[0] + 78 * Math.cos(angle), E[1] + 78 * Math.sin(angle)]
    })
    objects.push({
      id: `${prefix}-earth`, type: 'polygon', points: spherePoints,
      stroke: 'ink', strokeWidth: 2, fill: 'accentSoft', fillOpacity: 0.18,
    })
    // Smooth geometric shading conveys a sphere without inventing geography.
    objects.push({
      id: `${prefix}-earth-shade`, type: 'svgPath',
      d: 'M 270 292 A 78 78 0 0 0 270 448 C 214 420 214 320 270 292 Z',
      fill: 'accentSoft', fillOpacity: 0.3, stroke: 'muted', strokeWidth: 1,
    }, {
      id: `${prefix}-earth-latitude`, type: 'svgPath',
      d: 'M 198 340 C 229 359 306 370 344 348',
      stroke: 'muted', strokeWidth: 1, fillOpacity: 0,
    }, {
      id: `${prefix}-earth-meridian`, type: 'svgPath',
      d: 'M 281 293 C 316 322 321 409 281 447',
      stroke: 'muted', strokeWidth: 1, fillOpacity: 0,
    }, {
      id: `${prefix}-moon`, type: 'circle', center: `${prefix}-moon-center`,
      radius: 12, stroke: 'ink', strokeWidth: 1, fill: 'muted', fillOpacity: 1,
    })
    assertions.push(
      relationship(`${prefix}-P-inside-earth`, 'inside', {
        point: `${prefix}-P`, container: `${prefix}-earth`, margin: 18,
      }),
      relationship(`${prefix}-center-inside-earth`, 'inside', {
        point: `${prefix}-earth-center`, container: `${prefix}-earth`, margin: 70,
      }),
    )
  }
  const [cx, cy] = screen(P)
  objects.push({
    id: `${prefix}-small-orbit`, type: 'svgPath',
    d: `M ${cx + smallRadius} ${cy} A ${smallRadius} ${smallRadius} 0 1 1 ${cx - smallRadius} ${cy} A ${smallRadius} ${smallRadius} 0 1 1 ${cx + smallRadius} ${cy}`,
    stroke: 'ink', strokeWidth: 2, dash: 2, fillOpacity: 0,
  }, {
    id: `${prefix}-rmoon`, type: 'arrow',
    from: `${prefix}-P`, to: `${prefix}-rmoon-end`,
    stroke: 'ink', strokeWidth: 1.8,
  }, {
    id: `${prefix}-rearth`, type: 'segment',
    from: `${prefix}-P`, to: `${prefix}-earth-center`,
    stroke: 'ink', strokeWidth: 1.8,
  })
  assertions.push(
    relationship(`${prefix}-rmoon-connects`, 'connects', {
      arrow: `${prefix}-rmoon`, from: `${prefix}-P`, to: `${prefix}-rmoon-end`,
    }),
    relationship(`${prefix}-rearth-connects`, 'connects', {
      arrow: `${prefix}-rearth`, from: `${prefix}-P`, to: `${prefix}-earth-center`,
    }),
  )
  for (const [name, radius] of [['large', 271], ['small', 83]]) {
    const start = polar(radius, 35)
    const end = polar(radius, 68)
    const radians = 68 * Math.PI / 180
    const tangent = [-Math.sin(radians), Math.cos(radians)]
    const arrowFrom = [end[0] - tangent[0] * 13, end[1] - tangent[1] * 13]
    const id = `${prefix}-motion-${name}`
    objects.push(
      {
        id, type: 'svgPath',
        d: `M ${numbers(screen(start))} A ${radius} ${radius} 0 0 0 ${numbers(screen(end))}`,
        stroke: 'ink', strokeWidth: 2, fillOpacity: 0,
      },
      point(`${id}-tail`, arrowFrom, 1),
      point(`${id}-tip`, end, 1),
      {
        id: `${id}-head`, type: 'arrow', from: `${id}-tail`, to: `${id}-tip`,
        stroke: 'ink', strokeWidth: 2,
      },
    )
    assertions.push(
      relationship(`${id}-connects`, 'connects', {
        arrow: `${id}-head`, from: `${id}-tail`, to: `${id}-tip`,
      }),
      relationship(`${id}-tip-on-arc`, 'pointOnCircle', {
        point: `${id}-tip`, center: `${prefix}-P`, radius,
      }),
      relationship(`${id}-ccw-tangent`, 'perpendicular', {
        a: `${prefix}-P`, b: `${id}-tip`, c: `${id}-tail`, d: `${id}-tip`,
      }),
    )
  }
  objects.push(
    text(`${prefix}-P-text`, [313, 332], 'P', 'NE'),
    text(`${prefix}-earth-text`, sphere ? [270, 165] : [243, 243],
      sphere ? 'Earth' : 'Earth center', sphere ? 'CENTER' : 'W'),
    text(`${prefix}-moon-text`, [467, 448], sphere ? 'Moon' : 'Moon center', 'W'),
    text(`${prefix}-rearth-text`, [266, 312], 'r_E', 'CENTER', 20),
    text(`${prefix}-rmoon-text`, [440, 332], 'r_M', 'CENTER', 20),
    text(`${prefix}-scale-notice`, [310, 29], 'Not to scale', 'CENTER', 18),
  )
  const physicalIds = [
    `${prefix}-P`, `${prefix}-earth-center`, `${prefix}-moon-center`,
    `${prefix}-rmoon-end`, `${prefix}-large-orbit`, `${prefix}-small-orbit`,
    `${prefix}-rmoon`, `${prefix}-rearth`,
  ]
  const inventory = [{
    id: `${prefix}-source-point-relations`,
    description: 'Original pixels: P is the common orbit center; Earth center is below-left on the small dashed path; Moon center is upper-right on the large solid path. The Moon-radius marker goes horizontally right to the circumference, not to the displayed Moon. Modern regularization: exact circles and opposite rays, schematic coordinate lengths, and Earth-radius segment joined to Earth center.',
    requires: ['pointRelationships'],
    objects: physicalIds,
    assertions: assertions.filter(item => !item.id.includes('inside')).map(item => item.id),
  }, {
    id: `${prefix}-source-orbit-structure`,
    description: 'One solid large unfilled orbit and one small dashed unfilled orbit around P; no panels, axes, grid, ticks or numerical distance labels. Styles and common centers are additionally checked in source-assertions.json; objectCount is only supplementary.',
    objects: [`${prefix}-large-orbit`, `${prefix}-small-orbit`],
    assertions: [`${prefix}-moon-on-large`, `${prefix}-earth-on-small`],
  }, {
    id: `${prefix}-source-radii-labels`,
    description: 'Source Moon-radius line points right to large circumference, Earth-radius marking points toward Earth-center region. English/math label mapping P, r_E and r_M; exact tiny Earth arrowhead unresolved, so a plain segment is used without claiming that detail.',
    objects: [
      `${prefix}-rmoon`, `${prefix}-rearth`, `${prefix}-P-text`,
      `${prefix}-rearth-text`, `${prefix}-rmoon-text`,
    ],
    assertions: [`${prefix}-rmoon-connects`, `${prefix}-rearth-connects`, `${prefix}-rmoon-horizontal`],
  }]
  for (const name of ['large', 'small']) {
    const id = `${prefix}-motion-${name}`
    inventory.push({
      id: `${id}-source-group`,
      description: 'Source upper-right curved counterclockwise motion arrow. Modern exact arc with short terminal tangent arrow, no force or acceleration vector. Executable supplemental check proves positive angular direction, correct arc flags, visible arrowhead and tangent orientation.',
      requires: ['pointRelationships'],
      objects: [`${prefix}-P`, id, `${id}-tail`, `${id}-tip`, `${id}-head`],
      assertions: [`${id}-connects`, `${id}-tip-on-arc`, `${id}-ccw-tangent`],
    })
  }
  // Keep source point groups wholly closed under their mapped assertions.
  inventory[0].assertions = assertions
    .filter(item => !item.id.includes('inside') && !item.id.includes('motion'))
    .map(item => item.id)
  if (sphere) {
    inventory.push({
      id: `${prefix}-source-bodies`,
      description: 'Source shaded Earth sphere contains both its own center and the offset P; smaller dark Moon disk is on the large path. Modern smooth geometric shading replaces scan hatching/geography, without artwork or calibrated body-size claims.',
      objects: [
        `${prefix}-P`, `${prefix}-earth-center`, `${prefix}-moon-center`,
        `${prefix}-earth`, `${prefix}-earth-shade`, `${prefix}-earth-latitude`,
        `${prefix}-earth-meridian`, `${prefix}-moon`,
        `${prefix}-earth-text`, `${prefix}-moon-text`,
      ],
      assertions: [`${prefix}-P-inside-earth`, `${prefix}-center-inside-earth`, `${prefix}-moon-on-large`],
    })
  } else {
    inventory.push({
      id: `${prefix}-source-centers-no-globe`,
      description: 'Source schematic has three marked points only, labelled P, Earth center and Moon center. No sphere or body disk. The no-globe and exact English label constraints are executable in source-assertions.json.',
      objects: [
        `${prefix}-P`, `${prefix}-earth-center`, `${prefix}-moon-center`,
        `${prefix}-earth-text`, `${prefix}-moon-text`,
      ],
      assertions: [`${prefix}-earth-on-small`, `${prefix}-moon-on-large`],
    })
  }
  inventory.push({
    id: `${prefix}-modern-scale-disclaimer`,
    description: 'Modern addition, not printed source: Not to scale. Coordinate radii and sphere sizes are schematic, never kilometer measurements. External figure caption remains parent-owned.',
    objects: [`${prefix}-scale-notice`],
    assertions: [],
  })
  const spec = {
    schema: 'ld-s10y-image/figure-spec@2',
    id: `fig-${number}`,
    mode: 'deterministic',
    description: sphere
      ? 'Earth and Moon revolve about P inside Earth; distinguish body centers from their common orbital center, preserving solid/dashed paths and counterclockwise motion. Schematic, not to scale.'
      : 'Compare Earth-center and Moon-center paths about common P without a globe, using two radii and counterclockwise motion. Schematic, not to scale.',
    source: {
      image: { path: sourcePath, sha256: context.sourceImage.sha256 },
      authoritativeText: fullText,
      inventory,
    },
    canvas: { width: 640, height: 640, boundingBox: [0, 640, 640, 0], background: 'paper' },
    display: { layout: 'scroll', purpose: 'instructional', minTextPx: 16, widths: [640] },
    objects,
    assertions,
    layout: { minLabelGap: 3 },
  }
  out(directory, 'spec.json', spec)
  out(directory, 'full-text-context.json', {
    role: 'AUTHOR ONLY',
    lesson: lesson.id,
    lessonPath,
    lessonSha256: hash(lessonBytes),
    title: lesson.title,
    completeEditionText: fullText,
    routing: 'deterministic; instructional; fixed 640px horizontal scroll media',
    sourceImage: spec.source.image,
    pixelInventory: `${directory}/original-pixel-inventory.md`,
    pixelInventorySha256: hash(fs.readFileSync(`${directory}/original-pixel-inventory.md`)),
    unresolvedSourceFacts: ['Exact tiny Earth-radius terminal arrowhead is not resolved; no terminal head is claimed.'],
    modernAdditions: [
      'Exact circular regularization and opposite rays; all coordinate lengths are schematic.',
      'English/math labels at readable fixed media width.',
      'Earth radius terminates at Earth center for instructional clarity.',
      'Not to scale notice.',
      ...(sphere ? ['Smooth geometric sphere shading, no geography or semantic artwork.'] : []),
    ],
    independentApproval: false,
  })
  console.log(`authored ${directory}/spec.json`)
}
