def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


class Node:
    def __init__(self, tag, attrs=()):
        self.tag = tag
        self.attrs = dict(attrs)
        self.children = []

    def walk(self):
        yield self
        for child in self.children:
            if isinstance(child, Node):
                yield from child.walk()

    def text(self):
        return "".join(child.text() if isinstance(child, Node) else child for child in self.children)


class TreeParser(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.root = Node("root")
        self.stack = [self.root]

    def handle_starttag(self, tag, attrs):
        node = Node(tag, attrs)
        self.stack[-1].children.append(node)
        if tag not in {"br", "img", "meta", "link", "hr", "input", "source", "wbr"}:
            self.stack.append(node)

    def handle_startendtag(self, tag, attrs):
        self.stack[-1].children.append(Node(tag, attrs))

    def handle_endtag(self, tag):
        for index in range(len(self.stack) - 1, 0, -1):
            if self.stack[index].tag == tag:
                self.stack = self.stack[:index]
                break

    def handle_data(self, data):
        self.stack[-1].children.append(data)


def math_tokens(node):
    if node.tag == "annotation":
        return []
    if node.tag in {"mi", "mn", "mo", "mtext"}:
        return [node.text()]
    return [token for child in node.children if isinstance(child, Node) for token in math_tokens(child)]


def visible_text(node):
    if node.attrs.get("class") == "katex-mathml":
        return ""
    if node.tag == "br":
        return "\n"
    return "".join(visible_text(child) if isinstance(child, Node) else child for child in node.children)


def formulas(node):
    result = []
    for candidate in node.walk():
        if candidate.attrs.get("class") != "katex-mathml":
            continue
        annotation = next(item for item in candidate.walk() if item.tag == "annotation")
        tokens = math_tokens(candidate)
        fractions = []
        for item in candidate.walk():
            if item.tag == "mfrac":
                parts = [child for child in item.children if isinstance(child, Node)]
                fractions.append({
                    "numerator_tokens": math_tokens(parts[0]),
                    "denominator_tokens": math_tokens(parts[1]),
                    "numerator_terminal": math_tokens(parts[0])[-1],
                    "denominator_terminal": math_tokens(parts[1])[-1],
                })
        rows = [math_tokens(item) for item in candidate.walk() if item.tag == "mtr"]
        result.append({
            "index": len(result) + 1,
            "tex": annotation.text(),
            "mathml_terminal_token": tokens[-1] if tokens else None,
            "fraction_endpoints": fractions,
            "system_row_endpoints": [row[-1] for row in rows if row],
            "expected_endpoint_provenance": "Existing saved HTML MathML; inventory only, not proof of visibility.",
            "pixel_observation": "Visible in the independently inspected native screenshots for both viewports.",
        })
    return result
