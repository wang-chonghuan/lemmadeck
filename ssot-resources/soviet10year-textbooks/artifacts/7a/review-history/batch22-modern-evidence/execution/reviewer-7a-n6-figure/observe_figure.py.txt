import hashlib
import json

import xml.etree.ElementTree as ET

from PIL import Image, ImageDraw


OWN = Path(__file__).resolve().parent
BASE = OWN.parent / "batch22-modern-7a"
FIGURES = BASE / "root/7a/editions/modern-us-neutral/figures"
SOURCE = BASE / "root/7a/figures/fig-02.png"
PREVIEW = BASE / "work/fig-02.preview.png"
SPEC = FIGURES / "fig-02.spec.json"
RENDER = FIGURES / "fig-02.render.json"
SVG = FIGURES / "fig-02.svg"
CONTEXT = BASE / "work/figure-02-context.json"
LESSON = BASE / "work/adapt/7a/modern-us-neutral/lessons/alg7-c1-s2-n6/lesson.template.json"
EXERCISES = LESSON.with_name("exercises.template.json")
MONTAGE = OWN / "declared-widths-existing-preview.png"


def bound(path):
    return {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}


def inspect():
    spec = json.loads(SPEC.read_text())
    render = json.loads(RENDER.read_text())
    context = json.loads(CONTEXT.read_text())
    lesson = json.loads(LESSON.read_text())
    exercises = json.loads(EXERCISES.read_text())
    svg = ET.parse(SVG).getroot()
    texts = [{
        "text": "".join(t.itertext()),
        "label_id": t.get("data-label-id"),
        "font_size_at_canvas_px": float(t.get("font-size")),
        "x": float(t.get("x")), "y": float(t.get("y")),
        "fill": t.get("fill"),
    } for t in svg.iter("{http://www.w3.org/2000/svg}text")]
    assert [t["text"] for t in texts] == ["A", "B", "A \u2282 B"]
    assert spec["source"]["image"]["sha256"] == bound(SOURCE)["sha256"]
    assert render["spec"]["sha256"] == bound(SPEC)["sha256"]
    assert render["output"]["svg"]["sha256"] == bound(SVG)["sha256"]
    assert spec["source"]["authoritativeText"] == context["authoritativeText"]
    assert [p["text"] for p in lesson["prose"] if p["text"]] == [
        item["text"] for item in context["authoritativeText"] if "exercise" not in item
    ]
    assert [e["text"] for e in exercises["exercises"]] == [
        item["text"] for item in context["authoritativeText"] if "exercise" in item
    ]
    objects = {o["id"]: o for o in spec["objects"]}
    boundary_assertions = [
        a for a in spec["assertions"] if a["id"].startswith("a-boundary-")
    ]
    assert len(boundary_assertions) == len(objects["a-set"]["points"]) == 64
    assert [a["point"] for a in boundary_assertions] == objects["a-set"]["points"]
    assert all(a["container"] == "b-set" and a["margin"] == 0.5 for a in boundary_assertions)
    assert len(objects["b-set"]["points"]) == 64
    with Image.open(SOURCE) as source_image, Image.open(PREVIEW) as preview:
        sizes = {"source": list(source_image.size), "preview": list(preview.size)}
        assert preview.size == (480, 480)
    checks = [{
        "declared_width_px": width,
        "svg_font_size_after_width_transform_px": [
            t["font_size_at_canvas_px"] * width / float(svg.get("width")) for t in texts
        ],
        "existing_render_report_check": next(c for c in render["displayChecks"] if c["width"] == width),
        "width_origin": "Declared authoring width, not measured headed-product media.",
    } for width in spec["display"]["widths"]]
    return {
        "bindings": [bound(p) for p in (SOURCE, PREVIEW, SPEC, RENDER, SVG, CONTEXT, LESSON, EXERCISES)],
        "image_dimensions_px": sizes,
        "svg_texts": texts,
        "declared_display": spec["display"],
        "declared_width_font_checks": checks,
        "full_boundary_coverage": {
            "a_vertices": 64, "b_vertices": 64,
            "a_vertex_inside_b_assertions": len(boundary_assertions),
            "minimum_clearance_lower_bound_user_units": 3.6 * math.cos(math.pi / 64) - 1.1 - 1.65,
            "proof": "The convex 64-vertex B regularization contains the disk centered (4.9,5.5) with radius 3.6*cos(pi/64). All of A lies within radius 1.1+1.65=2.75 of that center, strictly smaller than B's inradius. Hence all segments, not merely sampled anchors, lie strictly inside.",
        },
        "current_render_status": render["status"],
        "new_renderer_or_standalone_validator_invocations": 0,
    }


def main():

        assert not MONTAGE.exists()
        canvas = Image.new("RGB", (888, 360), "white")
        draw = ImageDraw.Draw(canvas)
        x = 12
        with Image.open(PREVIEW) as image:
            image = image.convert("RGB")
            for width in (240, 280, 320):
                draw.text((x, 8), f"Existing PNG scaled to {width}px; font {36*width/480:g}px", fill="black")
                tile = image.resize((width, width), Image.Resampling.LANCZOS)
                canvas.paste(tile, (x, 32))
                assert canvas.crop((x, 32, x + width, 32 + width)).tobytes() == tile.tobytes()
                x += width + 12
