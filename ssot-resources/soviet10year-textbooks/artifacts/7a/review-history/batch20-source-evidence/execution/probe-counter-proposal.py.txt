def count(mask):
    columns = np.flatnonzero(mask.any(axis=0))
    width = int(columns[-1] - columns[0] + 1) if columns.size else mask.shape[1]

def inspect_helper(function, mask, a, b, glyph_h):
    captured = {}

    def monitor(frame, event, arg):
        if frame.f_code is function.__code__ and event == "return":
            local = frame.f_locals
            remaining = local.get("remaining")
            captured.update({
                "return": bool(arg),
                "rules": local.get("rules", []),
                "remaining_y": (
                    [int(remaining[0]) + a, int(remaining[-1]) + a]
                    if remaining is not None and remaining.size else []
                ),
                "remaining_span": (
                    int(remaining[-1] - remaining[0] + 1)
                    if remaining is not None and remaining.size else 0
                ),
            })
        return monitor

    sys.settrace(monitor)
    try:
        function(mask, a, b, glyph_h)
    finally:
        sys.settrace(None)
    return captured
