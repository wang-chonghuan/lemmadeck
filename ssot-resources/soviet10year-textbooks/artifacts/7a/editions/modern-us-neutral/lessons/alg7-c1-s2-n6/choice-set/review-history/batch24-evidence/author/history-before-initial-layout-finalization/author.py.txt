import hashlib
import json
from pathlib import Path


OWN = Path(__file__).resolve().parent
seed = json.loads((OWN / "seed.json").read_bytes())
source_record = json.loads((OWN / "source-solutions.json").read_bytes())
sources = {row["number"]: row for row in source_record["sources"]}
image_observation = json.loads((OWN / "source-assets" / "actual-observations.json").read_bytes())
assert image_observation["actually_opened"]
questions = []


def option(text, truth, derivation, misconception=None):
    assert truth == (misconception is None)
    return {"text": text, "truth": truth, "derivation": derivation, "misconception": misconception}


def add(suffix, number, mode, stem, task, unit_reason, alternatives, figure=False, extra_losses=()):
    qid = f"n6-{suffix}"
    ordered = sorted(alternatives, key=lambda row: hashlib.sha256(
        f"initial-position-v1:{qid}:{row['text']}".encode()).hexdigest())
    options, rationales, correct = [], [], []
    for index, row in enumerate(ordered):
        oid = chr(65 + index)
        options.append({"id": oid, "text": row["text"], "figure_ids": []})
        rationales.append({
            "option_id": oid, "truth": "correct" if row["truth"] else "incorrect",
            "derivation": row["derivation"], "misconception": row["misconception"],
        })
        if row["truth"]:
            correct.append(oid)
    assert correct and (mode != "single" or len(correct) == 1)
    questions.append({
        "id": qid, "source_numbers": [number],
        "view": {"mode": mode, "stem": stem, "figure_ids": ["fig-02"] if figure else [], "options": options},
        "intent": {
            "task": task, "source_solution": sources[number]["solution"],
            "unit_reason": unit_reason, "concept_blocks": [1, 2, 6, 12, 14, 15],
            "limitations": [
                f"Choice recognition does not demonstrate this original production: {item}"
                for item in sources[number]["production_requirements"]
            ] + list(extra_losses),
            "figure_observations": [
                {"figure_id": "fig-02", "observation": image_observation["observation"]}
            ] if figure else [],
        },
        "key": {"correct": correct, "rationales": rationales},
    })

coverage = []
for number, source in sources.items():
    omitted = [f"Independent production not assessed: {item}" for item in source["production_requirements"]]
    if number == "83":
        omitted.append("83c original unconditional complete enumeration remains convention-blocked. The conditional full-set candidate retains both possibilities without selecting whether0 belongs to N.")
    if number == "85":
        omitted.append("Three original containment drawings are replaced by labeling recognition of the actually opened existing fig-02; original a,b,d preserved, absent c not invented.")
    coverage.append({
        "source_number": number,
        "question_ids": [q["id"] for q in questions if number in q["source_numbers"]],
        "omitted_tasks": omitted,
        "reason": "Complete mathematical truth sets and original-domain relationships are retained except the explicitly unresolved unconditional convention; proof, model and number-line production remain named losses. No new figure was created.",
    })
digest = hashlib.sha256(json.dumps(seed, sort_keys=True, ensure_ascii=True, separators=(",", ":")).encode()).hexdigest()
assert digest == source_record["seed_sha256"]
draft = {"schema": "s10y-choice-draft-v1", "seed_sha256": digest, "new_figures": [], "questions": questions, "coverage": coverage}
view = {"questions": [{"id": q["id"], **q["view"]} for q in questions]}
for filename, value in (("draft.json", draft), ("view.json", view)):
    with (OWN / filename).open("x", encoding="utf-8") as output:
        json.dump(value, output, ensure_ascii=True, indent=2)
        output.write("\n")
