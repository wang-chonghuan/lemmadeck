import hashlib
import json
from pathlib import Path

from PIL import Image, ImageDraw

scratch = Path(".intentfold/tickets/STEMROBIN-182/tmp")
for lesson in ("alg7-c1-s2-n5", "alg7-c1-s2-n6"):
    own = scratch / f"reviewer-{lesson}-actual-display"
    manifest = json.loads((scratch / f"batch24-render/7a-{lesson}-capture-manifest.json").read_text())
    supplement = json.loads((own / "supplement-inventory.json").read_text())
    files = [(Path(row["path"]), "primary") for row in manifest["screenshots"]]
    files += [(Path(row["path"]), "supplement") for row in supplement["screenshots"]]
    buckets = {"desktop": [], "mobile": [], "small": []}
    index = []
    for file, kind in files:
        image = Image.open(file).convert("RGB")
        width, height = image.size
        crop = [0, 0, width, height]
        if width == 1440:
            crop = [340, 0, 1100, height]
            image = image.crop(crop)
        bucket = "small" if height < 100 else "mobile" if width == 390 else "desktop"
        row = {
            "path": str(file.resolve()), "kind": kind,
            "sha256": hashlib.sha256(file.read_bytes()).hexdigest(),
            "original_dimensions": [width, height],
            "native_scale": 1, "preview_crop": crop,
            "preview_does_not_modify_original": True,
        }
        buckets[bucket].append((image, row))
    for bucket, entries in buckets.items():
        per_page = 12 if bucket == "small" else 6 if bucket == "mobile" else 4
        columns = 3 if bucket in ("small", "mobile") else 2
        for start in range(0, len(entries), per_page):
            group = entries[start:start + per_page]
            cell_width = max(image.width for image, _ in group) + 12
            row_heights = [
                max(image.height for image, _ in group[i:i + columns]) + 30
                for i in range(0, len(group), columns)
            ]
            canvas = Image.new("RGB", (cell_width * columns, sum(row_heights)), "#e5e7eb")
            draw = ImageDraw.Draw(canvas)
            montage = own / f"native-{bucket}-{start // per_page + 1:02}.png"
            assert not montage.exists()
            y = 0
            for offset, (image, row) in enumerate(group):
                column = offset % columns
                if offset and column == 0:
                    y += row_heights[offset // columns - 1]
                x = column * cell_width + 6
                draw.text((x, y + 3), f"{offset + 1}: {Path(row['path']).name}", fill="#111827")
                canvas.paste(image, (x, y + 24))
                row["opened_via_native_montage"] = str(montage.resolve())
                row["montage_slot"] = offset + 1
                index.append(row)
            canvas.save(montage)
    destination = own / "native-montage-index.json"
    assert not destination.exists()
    destination.write_text(json.dumps({
        "lesson": lesson, "status": "pending_actual_pixel_observation",
        "native_scale": 1, "files": index,
        "primary_count": len(manifest["screenshots"]),
        "supplement_count": len(supplement["screenshots"]),
    }, indent=2) + "\n")
    print(json.dumps({
        "lesson": lesson, "files": len(index),
        "montages": len(set(row["opened_via_native_montage"] for row in index)),
    }))
