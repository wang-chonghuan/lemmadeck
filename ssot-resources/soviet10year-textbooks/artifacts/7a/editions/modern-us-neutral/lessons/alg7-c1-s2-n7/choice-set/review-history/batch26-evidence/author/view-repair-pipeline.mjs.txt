function canonical(value) {
  if (typeof value === 'string') return ascii(value);
  if (Array.isArray(value)) return `[${value.map(canonical).join(',')}]`;
  if (value !== null && typeof value === 'object')
    return `{${Object.keys(value).sort().map(key => `${ascii(key)}:${canonical(value[key])}`).join(',')}}`;
  return JSON.stringify(value);
}

function retained(n) {

    assert.equal(record(item.preserved_path).sha256, item.sha256, 'Frozen history changed');
    if (!mutable.has(item.relative_path))
      assert.equal(record(item.original_path).sha256, item.sha256, `Unrelated prior file changed: ${item.relative_path}`);
  }

function differences(before, after, segments = []) {
  if (canonical(before) === canonical(after)) return [];
  if (before !== null && after !== null && typeof before === 'object' && typeof after === 'object' &&
      Array.isArray(before) === Array.isArray(after) &&
      canonical(Object.keys(before)) === canonical(Object.keys(after))) {
    return Object.keys(before).flatMap(key => differences(before[key], after[key], [...segments, key]));
  }
  return [{ op: 'replace', path: '/' + segments.map(x => x.replaceAll('~', '~0').replaceAll('/', '~1')).join('/'),
    segments, before, after }];
}
function reverse(value, delta) {
  const output = structuredClone(value);
  for (const change of [...delta].reverse()) {
    let target = output;
    for (const key of change.segments.slice(0, -1)) target = target[key];
    const key = change.segments.at(-1);
    assert.equal(canonical(target[key]), canonical(change.after));
    target[key] = change.before;
  }
  return output;
}

  }

}
function runOnce(n, name, argv, boundInputs, stdin = undefined) {
  const directory = commandDir(n);
  fs.mkdirSync(directory, { recursive: true });
  const startedPath = path.join(directory, `${name}.started.json`);
  const before = boundInputs.map(record);
  write(startedPath, { argv, cwd: repo, input_hashes: before,
    stdin_sha256: stdin === undefined ? null : sha(stdin), once_guard: true, started_at: new Date().toISOString() });
  const result = spawnSync(argv[0], argv.slice(1), { cwd: repo, input: stdin, encoding: 'utf8',
    env: { ...process.env, PYTHONDONTWRITEBYTECODE: '1', PWDEBUG: '', PLAYWRIGHT_HTML_OPEN: 'never' },
    maxBuffer: 8 * 1024 * 1024 });
  const stdout = path.join(directory, `${name}.stdout.txt`), stderr = path.join(directory, `${name}.stderr.txt`);
  fs.writeFileSync(stdout, result.stdout ?? '', { flag: 'wx' });
  fs.writeFileSync(stderr, result.stderr ?? '', { flag: 'wx' });
  const after = boundInputs.map(record);
  const immutable = canonical(before) === canonical(after);
  const receipt = { argv, cwd: repo, exit_code: result.status, signal: result.signal, error: result.error?.message ?? null,
    input_hashes: before, after_input_hashes: after, inputs_unchanged: immutable,
    output_hashes: [record(stdout), record(stderr)], pass: result.status === 0 && immutable,
    finished_at: new Date().toISOString(), no_review_self_approval: true };
  write(path.join(directory, `${name}.json`), receipt);
  assert(receipt.pass, `Stopped; preserve failed ${name} command evidence for diagnosis`);
  return receipt;
}
