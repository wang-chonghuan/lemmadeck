def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def digest(value):
    return sha(json.dumps(value, sort_keys=True, ensure_ascii=True,
                          separators=(",", ":")).encode())


def read(path):
    return json.loads(path.read_bytes())


def token_spans(text):
    spans = {}

    def skip(position):
        while position < len(text) and text[position].isspace():
            position += 1
        return position

    def walk(position, pointer):
        position = skip(position)
        start = position
        if text[position] == "{":
            position = skip(position + 1)
            if text[position] == "}":
                return position + 1
            while True:
                key, position = DECODER.raw_decode(text, position)
                assert isinstance(key, str)
                position = skip(position)
                assert text[position] == ":"
                escaped = key.replace("~", "~0").replace("/", "~1")
                position = skip(walk(position + 1, pointer + "/" + escaped))
                if text[position] == "}":
                    return position + 1
                assert text[position] == ","
                position = skip(position + 1)
        elif text[position] == "[":
            position = skip(position + 1)
            if text[position] == "]":
                return position + 1
            index = 0
            while True:
                position = skip(walk(position, pointer + "/" + str(index)))
                if text[position] == "]":
                    return position + 1
                assert text[position] == ","
                index += 1
                position = skip(position + 1)
        else:
            _, end = DECODER.raw_decode(text, position)
            spans[pointer] = (start, end)
            return end

    assert skip(walk(0, "")) == len(text)
    return spans


def differences(before, after, pointer=""):
    if isinstance(before, dict) and isinstance(after, dict):
        assert before.keys() == after.keys()
        result = []
        for key in before:
            escaped = key.replace("~", "~0").replace("/", "~1")
            result += differences(before[key], after[key], pointer + "/" + escaped)
        return result
    if isinstance(before, list) and isinstance(after, list):
        assert len(before) == len(after)
        return [leaf for i, pair in enumerate(zip(before, after))
                for leaf in differences(*pair, pointer + "/" + str(i))]
    return [] if before == after else [pointer]


def inverse_proof(before_path, after_path, expected):
    before_bytes, after_bytes = before_path.read_bytes(), after_path.read_bytes()
    before_text, after_text = before_bytes.decode(), after_bytes.decode()
    changed = differences(json.loads(before_text), json.loads(after_text))
    assert set(changed) == set(expected), (changed, expected)
    before_spans, after_spans = token_spans(before_text), token_spans(after_text)
    inverse = after_text
    evidence = []
    for pointer in sorted(changed, key=lambda item: after_spans[item][0], reverse=True):
        old_start, old_end = before_spans[pointer]
        new_start, new_end = after_spans[pointer]
        old_token = before_text[old_start:old_end]
        new_token = after_text[new_start:new_end]
        evidence.append({
            "pointer": pointer,
            "before_raw_token": old_token,
            "after_raw_token": new_token,
            "before_token_sha256": sha(old_token.encode()),
            "after_token_sha256": sha(new_token.encode()),
        })
        inverse = inverse[:new_start] + old_token + inverse[new_end:]
    assert inverse.encode() == before_bytes
    return {
        "before_path": str(before_path),
        "after_path": str(after_path),
        "before_byte_sha256": sha(before_bytes),
        "after_byte_sha256": sha(after_bytes),
        "before_canonical_sha256": digest(json.loads(before_text)),
        "after_canonical_sha256": digest(json.loads(after_text)),
        "changed_leaves": sorted(changed),
        "raw_tokens": evidence,
        "inverse_raw_bytes_identical": True,
        "inverse_byte_sha256": sha(inverse.encode()),
        "all_other_bytes_including_whitespace_unchanged": True,
    }


def save(path, value):
    with path.open("x", encoding="utf-8") as stream:
        json.dump(value, stream, ensure_ascii=True, indent=2)
        stream.write("\n")
