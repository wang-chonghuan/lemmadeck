def digest(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return digest(
        json.dumps(
            value, sort_keys=True, ensure_ascii=True, separators=(",", ":")
        ).encode("utf-8")
    )


def file_record(path):
    data = path.read_bytes()
    return {
        "path": str(path),
        "sha256": digest(data),
        "bytes": len(data),
    }


def save_new(path, value):
    with path.open("x", encoding="utf-8") as handle:
        json.dump(value, handle, indent=2, ensure_ascii=True)
        handle.write("\n")

graph = TMP / "7a-n7/graph95"
spec_path = graph / "spec.json"
svg_path = graph / "fig-choice-7a-n7-95.svg"
png_path = graph / "fig-choice-7a-n7-95.640.png"
spec = json.loads(spec_path.read_bytes())
root = ET.parse(svg_path).getroot()
paths = [
    element
    for element in root.iter()
    if element.tag.endswith("path") and element.get("stroke-width") == "3"
]
parents = {child: parent for parent in root.iter() for child in parent}
spec_paths = [obj for obj in spec["objects"] if obj["type"] == "svgPath"]
assert len(paths) == len(spec_paths) == 2

curve_evidence = []
for element, obj, coefficient in zip(paths, spec_paths, (3, 2)):
    assert element.get("d") == obj["d"]
    transform = parents[element].get("transform")
    assert transform == "scale(100 25)"
    tokens = re.findall(r"[ML]|[-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?", element.get("d"))
    assert len(tokens) % 3 == 0
    assert all(tokens[i] == ("M" if i == 0 else "L") for i in range(0, len(tokens), 3))
    points = [
        (float(tokens[i + 1]) - 4, 28 - float(tokens[i + 2]))
        for i in range(0, len(tokens), 3)
    ]
    assert all(points[i + 1][0] > points[i][0] for i in range(len(points) - 1))
    queried = []
    for query in (-2, -1, 0, 1, 2):
        for left, right in zip(points, points[1:]):
            if left[0] <= query <= right[0]:
                ratio = (query - left[0]) / (right[0] - left[0])
                interpolated = left[1] + ratio * (right[1] - left[1])
                queried.append(
                    {
                        "x": query,
                        "exact_source_y": coefficient * query * query,
                        "rendered_polyline_y": interpolated,
                        "position_at_640": [80 * (query + 4), 20 * (28 - interpolated)],
                    }
                )
                break
    curve_evidence.append(
        {
            "spec_object_id": obj["id"],
            "spec_and_svg_path_bytes_equal_as_strings": True,
            "svg_parent_transform": transform,
            "screen_to_math": "x=raw_x-4; y=28-raw_y, with native SVG scale(100 25)",
            "independently_tested_formula": f"y={coefficient}x^2",
            "sampled_vertices_checked": len(points),
            "max_abs_equation_residual": max(abs(y - coefficient * x * x) for x, y in points),
            "max_abs_symmetry_x_residual": max(abs(a[0] + b[0]) for a, b in zip(points, reversed(points))),
            "max_abs_symmetry_y_residual": max(abs(a[1] - b[1]) for a, b in zip(points, reversed(points))),
            "math_x_range": [points[0][0], points[-1][0]],
            "math_y_range": [min(y for _, y in points), max(y for _, y in points)],
            "visible_top_limit": 28,
            "expected_top_abs_x": math.sqrt(28 / coefficient),
            "max_chord_interpolation_error_bound_math_y": max(
                coefficient * (b[0] - a[0]) ** 2 / 4 for a, b in zip(points, points[1:])
            ),
            "queried_coordinates": queried,
        }
    )
