"""Execute only the five revised author procedures, not an independent review."""
from fractions import Fraction as F
import json
import sys


def graph95(swapped_coefficients):
    coefficients = {"f": F(2), "g": F(3)}
    if swapped_coefficients:
        coefficients["f"], coefficients["g"] = coefficients["g"], coefficients["f"]
    identities = {"I": "f", "II": "g"}
    values = {name: coefficient * F(-2) ** 2 for name, coefficient in coefficients.items()}
    above = values[identities["I"]] > values[identities["II"]]
    return {
        **identities,
        "I_relation": "higher" if above else "lower",
        "f_relation": "greater" if values["f"] > values["g"] else "less",
    }, {
        "input": F(-2), "coefficients_after_operation": coefficients,
        "assigned_identities": identities, "derived_values": values,
        "operation": "swap_coefficients_and_identity" if swapped_coefficients else "swap_identity_only",
    }


def classify98(mode):
    numerator = [
        F("3.6") / F("1.2") - 3,
        (F(11, 9) / F(22, 3) - F(1, 6)) * F(".23"),
        -8 / F(".4") + F(".02") * 1000,
        (F(1, 4) - F(1, 6)) * 8 - F(2, 3),
    ]
    denominator = [
        F("5.6") * F("2.1") - F("11.76"),
        F(17, 8) + F("1.2"),
        F("1.753") * F(".8") + F("1.2"),
        F("1.85") - F("1.62"),
    ]
    operations = []
    if mode == "ignore_denominator_misread_c_drop_d8":
        numerator[2] = -8 / F(".4") + F(".02") * 100
        numerator[3] = F(1, 4) - F(1, 6) - F(2, 3)
        operations = ["a: assign 0 to 0/0", "c: read multiplier 1000 as 100", "d: omit multiplier 8"]
    elif mode == "mixed_number_bug_round_d":
        # This b operation is unchanged from the frozen author's original program.
        numerator[1] = (F(1 + 2, 9) / F(7 + 1, 3) - F(1, 6)) * F(".23")
        denominator[3] = F(round(F("1.85"))) - F(round(F("1.62")))
        operations = ["b: mixed integers added to numerators before division",
                      "d: separately round 1.85 and 1.62 to integers before subtracting"]
    else:
        raise ValueError(mode)
    states = ["undefined" if d == 0 else "zero" if n == 0 else "nonzero"
              for n, d in zip(numerator, denominator)]
    values = ["undefined" if d == 0 else n / d for n, d in zip(numerator, denominator)]
    if mode == "ignore_denominator_misread_c_drop_d8":
        states[0] = "zero"
        values[0] = F(0)
        assert values[2] == F(-22500, 3253) and values[3] == F(-175, 69)
    else:
        assert values[1] == F(-23, 7980) and denominator[3] == 0
    return states, {"operations": operations, "numerators": numerator,
                    "denominators": denominator, "claimed_values": values}


def axis_product_abs_y():
    mistaken_constants = [F(-5), F(7)]
    roots = [-constant for constant in mistaken_constants]
    y_before_abs = mistaken_constants[0] * mistaken_constants[1] / F(10)
    y_after_abs = abs(y_before_abs)
    return {"x_axis": [[value, F(0)] for value in roots], "y_axis": [[F(0), y_after_abs]]}, {
        "operations": ["reverse both factor-root signs", "take absolute value of negative y-intercept"],
        "mistaken_factor_constants": mistaken_constants, "derived_roots": roots,
        "y_before_abs": y_before_abs, "y_after_abs": y_after_abs,
    }


def normalize(value):
    if isinstance(value, F):
        return str(value)
    if isinstance(value, list):
        return [normalize(item) for item in value]
    if isinstance(value, dict):
        return {key: normalize(item) for key, item in value.items()}
    return value


def visible(result, question):
    if question == "q95":
        height = "高于" if result["I_relation"] == "higher" else "低于"
        relation = ">" if result["f_relation"] == "greater" else "<"
        return (f'I 对应 ${result["I"]}$、II 对应 ${result["II"]}$；'
                f'在 $x=-2$ 处 I {height} II，$f(-2){relation}g(-2)$。')
    if question == "q98":
        words = {"zero": "零", "nonzero": "非零", "undefined": "没有意义"}
        return "；".join(words[state] for state in result) + "。"
    x_points = ",".join(f"({x},{y})" for x, y in result["x_axis"])
    y = F(result["y_axis"][0][1])
    assert y == F(7, 2)
    return f"$x$ 轴：${x_points}$；$y$ 轴：$(0,\\frac72)$。"


payload = json.load(sys.stdin)
programs = {
    "graph95_swapped_coefficients_and_identity(-2)": lambda: graph95(True),
    "graph95_swapped_identity_only(-2)": lambda: graph95(False),
    "classify98_repair('ignore_denominator_misread_c_drop_d8')":
        lambda: classify98("ignore_denominator_misread_c_drop_d8"),
    "classify98_repair('mixed_number_bug_round_d')": lambda: classify98("mixed_number_bug_round_d"),
    "axis_product_abs_y(-5,7,10)": axis_product_abs_y,
}
assert len(payload["programs"]) == 5
assert {entry["expr"] for entry in payload["programs"]} == set(programs)
results = []
for entry in payload["programs"]:
    result, trace = programs[entry["expr"]]()
    normalized = normalize(result)
    assert normalized == entry["expected"], (entry, normalized)
    generated_text = visible(normalized, entry["question_id"])
    assert generated_text == entry["displayed_text"], (entry, generated_text)
    results.append({**entry, "actual": normalized, "actual_displayed_text": generated_text,
                    "trace": normalize(trace), "matches_expected_and_display": True})
print(json.dumps({
    "scope": "Five changed authored false procedures only; no semantic self-approval",
    "executed": len(results), "results": results,
}, ensure_ascii=False, indent=2))
