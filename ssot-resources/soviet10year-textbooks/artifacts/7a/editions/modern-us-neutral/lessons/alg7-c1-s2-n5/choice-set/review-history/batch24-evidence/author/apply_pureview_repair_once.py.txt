def bind(path):
    return {"path": str(path.relative_to(REPO)), "sha256": sha256(path.read_bytes()).hexdigest()}


def encode(value):
    return (json.dumps(value, ensure_ascii=True, indent=2) + "\n").encode()


def canonical_digest(value):
    return sha256(
        json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(",", ":")).encode()
    ).hexdigest()


def save_new(path, value):
    with path.open("xb") as out:
        out.write(encode(value))


def differences(before, after, pointer=""):
    if type(before) is not type(after):
        return [{"path": pointer or "/", "before": before, "after": after}]
    if isinstance(before, dict):
        result = []
        for key in sorted(before.keys() | after.keys()):
            child = pointer + "/" + key.replace("~", "~0").replace("/", "~1")
            if key not in before or key not in after:
                result.append({
                    "path": child,
                    "before": before.get(key),
                    "after": after.get(key),
                    "before_present": key in before,
                    "after_present": key in after,
                })
            else:
                result.extend(differences(before[key], after[key], child))
        return result
    if isinstance(before, list):
        if len(before) != len(after):
            return [{"path": pointer or "/", "before": before, "after": after}]
        return [
            row for i, (left, right) in enumerate(zip(before, after))
            for row in differences(left, right, pointer + "/" + str(i))
        ]
    return [] if before == after else [{"path": pointer or "/", "before": before, "after": after}]

def decimal(value):
    value = F(value)
    return format(Decimal(value.numerator) / Decimal(value.denominator), "f").rstrip("0").rstrip(".") if value.denominator != 1 else str(value.numerator)


def integers(values):
    return r"\{" + ",".join(str(x) for x in sorted(values)) + r"\}"


def pairs(values):
    return r"\{" + ",".join(f"({x},{y})" for x, y in values) + r"\}"


def quadratic_integer_roots(b, c):
    discriminant = b * b - 4 * c
    root = isqrt(discriminant)
    assert root * root == discriminant
    answers = {F(-b - root, 2), F(-b + root, 2)}
    assert all(x.denominator == 1 for x in answers)
    return {int(x) for x in answers}

def sum_pairs(total, zero_allowed, ordered_filter=False):
    low = 0 if zero_allowed else 1
    high = total if zero_allowed else total - 1
    return [(x, total - x) for x in range(low, high + 1) if not ordered_filter or x <= total - x]
