import hashlib
import json
from pathlib import Path


OWN = Path(__file__).resolve().parent
seed = json.loads((OWN / "seed.json").read_bytes())
source_record = json.loads((OWN / "source-solutions.json").read_bytes())
sources = {row["number"]: row for row in source_record["sources"]}
questions = []


def option(text, truth, derivation, misconception=None):
    assert truth == (misconception is None)
    return {"text": text, "truth": truth, "derivation": derivation, "misconception": misconception}


def add(suffix, number, mode, stem, task, unit_reason, alternatives, extra_losses=()):
    qid = f"n5-{suffix}"
    ordered = sorted(alternatives, key=lambda row: hashlib.sha256(
        f"initial-position-v1:{qid}:{row['text']}".encode()).hexdigest())
    options, rationales, correct = [], [], []
    for index, row in enumerate(ordered):
        oid = chr(65 + index)
        options.append({"id": oid, "text": row["text"], "figure_ids": []})
        rationales.append({
            "option_id": oid, "truth": "correct" if row["truth"] else "incorrect",
            "derivation": row["derivation"], "misconception": row["misconception"],
        })
        if row["truth"]:
            correct.append(oid)
    assert correct and (mode != "single" or len(correct) == 1)
    losses = [
        f"Choice recognition does not demonstrate this original production: {item}"
        for item in sources[number]["production_requirements"]
    ] + list(extra_losses)
    questions.append({
        "id": qid, "source_numbers": [number],
        "view": {"mode": mode, "stem": stem, "figure_ids": [], "options": options},
        "intent": {
            "task": task, "source_solution": sources[number]["solution"],
            "unit_reason": unit_reason, "concept_blocks": [2, 9, 15],
            "limitations": losses, "figure_observations": [],
        },
        "key": {"correct": correct, "rationales": rationales},
    })
