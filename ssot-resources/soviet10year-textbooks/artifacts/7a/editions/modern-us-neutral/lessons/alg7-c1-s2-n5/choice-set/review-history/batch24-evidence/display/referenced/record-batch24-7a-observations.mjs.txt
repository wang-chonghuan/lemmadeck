for (const [lesson, evidence] of Object.entries(opened)) {
  const own = `${scratch}/reviewer-${lesson}-actual-display`
  const inventory = read(`${own}/supplement-inventory.json`)
  const index = read(`${own}/native-montage-index.json`)
  const manifest = read(`${scratch}/batch24-render/7a-${lesson}-capture-manifest.json`)
  const actualMontages = [...new Set(index.files.map(row => path.basename(row.opened_via_native_montage)))].sort()
  assert.deepEqual(actualMontages, Object.keys(evidence).sort())
  const observations = index.files.map(row => {
    assert.equal(hash(row.path), row.sha256)
    const small = row.original_dimensions[1] < 100
    return {
      ...row, actually_opened: true, display_window_status: 'pass',
      evidence: evidence[path.basename(row.opened_via_native_montage)],
      standalone_full_object_claim: false,
      fragment_only_not_full_formula: small && lesson.endsWith('n5'),
    }
  })
  const coverage = inventory.rows.map(row => {
    const ranges = row.primary_ranges.map(r => ({
      path: manifest.screenshots.find(s => s.path.endsWith(r.filename)).path,
      start: r.y, end: Math.min(row.document_height, r.y + r.height),
    })).concat(inventory.screenshots.filter(s => s.surface === row.surface &&
      s.viewport.width === row.viewport.width && !s.horizontal_offset).map(s => ({
        path: s.path, start: s.document_rect.y, end: s.document_rect.y + s.document_rect.height,
      }))).sort((a, b) => a.start - b.start)
    let end = 0
    for (const range of ranges) {
      assert(range.start <= end + 1, 'Unobserved vertical gap')
      end = Math.max(end, range.end)
    }
    assert(end >= row.document_height - 1)
    const filesFor = r => ranges.filter(s => s.start < r.y + r.height && s.end > r.y).map(s => s.path)
    return {
      surface: row.surface, viewport: row.viewport,
      rendered_path: row.input.path, rendered_sha256: row.input.sha256,
      vertical_document_coverage_complete: true,
      units: row.units.map(unit => ({
        index: unit.index, label: unit.label, status: 'pass-observed-across-listed-windows',
        windows: filesFor(unit.rect),
        options: unit.options.map(option => ({ index: option.index, windows: filesFor(option.rect),
          status: 'pass-observed-across-listed-windows' })),
      })),
      horizontal_regions: row.overflows.map(region => {
        const ends = inventory.screenshots.filter(s => s.surface === row.surface &&
          s.viewport.width === row.viewport.width && s.path.endsWith(`overflow-${region.index + 1}-end.png`))
        assert.equal(ends.length, 1)
        assert(region.scroll_width <= 2 * region.client_width)
        return { index: region.index, initial_left_covered_by_vertical_windows: true,
          right_end: ends[0].path, client_width: region.client_width,
          scroll_width: region.scroll_width, continuous_union_no_middle_gap: true }
      }),
    }
  })
  const destination = `${own}/actual-observations.json`
  assert(!fs.existsSync(destination))
  fs.writeFileSync(destination, JSON.stringify({
    lesson, status: 'pass', observer: { kind: 'self_review', identity: 'parent actual pixel observer' },
    scope: 'Actual offline display pixels only. Does not repeat mathematical, source, pureview, canonical, figure-source or app acceptance review.',
    observations, coverage, opened_native_montages: Object.entries(evidence).map(([name, text]) => ({
      path: `${own}/${name}`, sha256: hash(`${own}/${name}`), actually_opened: true, evidence: text,
    })),
    counts: {
      primary_PNGs: index.primary_count, supplement_PNGs: index.supplement_count,
      actual_PNG_files_observed: index.files.length, native_montages: actualMontages.length,
      unique_surface_viewport_combinations: 6, additional_combinations: 0,
      missing_requested_objects: 0, incomplete_horizontal_regions: 0,
    },
    limitations: [
      'Screenshots at adjacent window boundaries are intentionally partial; full coverage is their union, not each screenshot alone.',
      'Native desktop montage previews crop only outer whitespace, preserving the full sheet without image scaling; original screenshots remain unchanged.',
      'Small KaTeX fragments do not prove complete mathematical figures or formulas.',
      'This is offline rendered content, not app navigation, saving, grading, or production verification.',
    ],
    evidence_helpers: [
      `${scratch}/capture-batch24-7a-gaps.mjs`, `${scratch}/montage-batch24-7a.py`,
      `${scratch}/record-batch24-7a-observations.mjs`,
    ],
    browser_closed_in_finally: true, headless_isolated: true, new_semantic_or_renderer_runs: 0,
  }, null, 2) + '\n', { flag: 'wx' })
  console.log(JSON.stringify({ lesson, status: 'pass', primary: index.primary_count,
    supplement: index.supplement_count, montages: actualMontages.length }))
}
