import hashlib
import json
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


ROOT = Path(__file__).resolve().parents[5]
FONT = ImageFont.load_default(size=12)
LABEL_H = 40
GAP = 12
MAX_W = 1440
MAX_H = 1980


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main(number):
    output = ROOT / (
        ".intentfold/tickets/STEMROBIN-182/tmp/"
        f"reviewer-6a-n{number}-actual-display"
    )
    log_path = output / "supplemental-capture-log.json"
    log = json.loads(log_path.read_text())
    if log["status"] != "capture-complete-pixel-review-pending":
        raise ValueError("Supplement capture failed; preserve and diagnose first")
    groups = {}
    for capture in log["captures"]:
        source = Path(capture["path"])
        if sha(source) != capture["sha256"]:
            raise ValueError(f"Source hash changed: {source}")
        groups.setdefault(capture["group"], []).append(capture)
    montages = []
    for group, captures in groups.items():
        pages = []
        placements = []
        x = y = row_height = 0
        for capture in captures:
            width = capture["width"]
            height = capture["height"] + LABEL_H
            if width > MAX_W or height > MAX_H:
                raise ValueError("Native image exceeds montage bounds")
            if x and x + width > MAX_W:
                y += row_height + GAP
                x = row_height = 0
            if y + height > MAX_H:
                pages.append(placements)
                placements = []
                x = y = row_height = 0
            placements.append((capture, x, y))
            x += width + GAP
            row_height = max(row_height, height)
        if placements:
            pages.append(placements)
        for index, placements in enumerate(pages, 1):
            width = max(x + item["width"] for item, x, y in placements)
            height = max(y + LABEL_H + item["height"]
                         for item, x, y in placements)
            target = output / f"supplement-{group}-{index:02d}.png"
            if target.exists():
                raise FileExistsError(target)
            canvas = Image.new("RGB", (width, height), "#d8d8d8")
            draw = ImageDraw.Draw(canvas)
            regions = []
            for capture, x, y in placements:
                label = Path(capture["path"]).name
                chunks = [label[i:i + 44] for i in range(0, len(label), 44)]
                draw.multiline_text(
                    (x + 2, y + 2), "\n".join(chunks[:3]),
                    font=FONT, fill="black", spacing=1,
                )
                with Image.open(capture["path"]) as opened:
                    source = opened.convert("RGB")
                    canvas.paste(source, (x, y + LABEL_H))
                    pasted = canvas.crop((
                        x, y + LABEL_H, x + source.width,
                        y + LABEL_H + source.height,
                    ))
                    if pasted.tobytes() != source.tobytes():
                        raise ValueError("Native source pixels changed")
                regions.append({
                    "source_path": capture["path"],
                    "source_sha256": capture["sha256"],
                    "x": x,
                    "y": y + LABEL_H,
                    "width": capture["width"],
                    "height": capture["height"],
                    "source_RGB_pixels_identical": True,
                })
            canvas.save(target)
            montages.append({
                "path": str(target),
                "sha256": sha(target),
                "width": width,
                "height": height,
                "regions": regions,
            })
    inventory = {
        "capture_log_path": str(log_path),
        "capture_log_sha256": sha(log_path),
        "method": "Native RGB paste; no resize, source clipping or browser capture.",
        "supplement_file_count": len(log["captures"]),
        "supplement_files": log["captures"],
        "montage_count": len(montages),
        "montages": montages,
    }
    target = output / "supplement-montage-inventory.json"
    if target.exists():
        raise FileExistsError(target)
    target.write_text(json.dumps(inventory, ensure_ascii=True, indent=2) + "\n")
    print(json.dumps({
        "number": number,
        "supplement_count": len(log["captures"]),
        "montage_count": len(montages),
        "montages": [item["path"] for item in montages],
    }, indent=2))


if __name__ == "__main__":
    main(int(sys.argv[1]))
