import hashlib
import json
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


ROOT = Path(__file__).resolve().parents[5]
LIMIT_W = 1440
LIMIT_H = 1980
LABEL_H = 24
GAP = 12
FONT = ImageFont.load_default(size=14)


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main(number):
    ticket = ROOT / ".intentfold/tickets/STEMROBIN-182/tmp"
    manifest_path = ticket / "batch20-render" / (
        f"6a-alg6-c4-s2-n{number}-capture-manifest.json"
    )
    output = ticket / f"reviewer-6a-n{number}-actual-display"
    output.mkdir(exist_ok=True)
    manifest = json.loads(manifest_path.read_text())
    entries = []
    for entry in manifest["screenshots"]:
        path = ROOT / entry["path"]
        actual_sha = digest(path)
        if actual_sha != entry["sha256"]:
            raise ValueError(f"Capture hash changed: {path}")
        with Image.open(path) as image:
            width, height = image.size
        label = path.name.replace(f"alg6-c4-s2-n{number}-", "")
        surface, viewport = label.split("-")[:2]
        entries.append({
            "path": str(path),
            "sha256": actual_sha,
            "width": width,
            "height": height,
            "label": label,
            "group": f"{surface}-{viewport}-"
            + ("viewport" if width >= 1000 else "native-crops"),
        })
    groups = {}
    for entry in entries:
        groups.setdefault(entry["group"], []).append(entry)
    montages = []
    for group, members in groups.items():
        pages = []
        placements = []
        x = y = row_height = 0
        for entry in members:
            width = entry["width"]
            height = entry["height"] + LABEL_H
            if width > LIMIT_W or height > LIMIT_H:
                raise ValueError("Native capture exceeds montage bounds")
            if x and x + width > LIMIT_W:
                y += row_height + GAP
                x = row_height = 0
            if y + height > LIMIT_H:
                pages.append(placements)
                placements = []
                x = y = row_height = 0
            placements.append((entry, x, y))
            x += width + GAP
            row_height = max(row_height, height)
        if placements:
            pages.append(placements)
        for index, placements in enumerate(pages, 1):
            width = max(x + entry["width"] for entry, x, y in placements)
            height = max(y + LABEL_H + entry["height"]
                         for entry, x, y in placements)
            target = output / f"primary-{group}-{index:02d}.png"
            if target.exists():
                raise FileExistsError(target)
            canvas = Image.new("RGB", (width, height), "#d8d8d8")
            draw = ImageDraw.Draw(canvas)
            regions = []
            for entry, x, y in placements:
                draw.text((x + 2, y + 3), entry["label"], font=FONT, fill="black")
                with Image.open(entry["path"]) as opened:
                    original = opened.convert("RGB")
                    canvas.paste(original, (x, y + LABEL_H))
                    pasted = canvas.crop((
                        x, y + LABEL_H,
                        x + original.width, y + LABEL_H + original.height,
                    ))
                    if pasted.tobytes() != original.tobytes():
                        raise ValueError("Native pixels changed")
                regions.append({
                    "source_path": entry["path"],
                    "source_sha256": entry["sha256"],
                    "x": x,
                    "y": y + LABEL_H,
                    "width": entry["width"],
                    "height": entry["height"],
                    "source_RGB_pixels_identical": True,
                })
            canvas.save(target)
            montages.append({
                "path": str(target),
                "sha256": digest(target),
                "width": width,
                "height": height,
                "regions": regions,
            })
    inventory = {
        "manifest_path": str(manifest_path),
        "manifest_sha256": digest(manifest_path),
        "method": "Native RGB paste only; no resizing, clipping or new browser capture.",
        "primary_file_count": len(entries),
        "primary_unique_sha256_count": len({entry["sha256"] for entry in entries}),
        "primary_files": entries,
        "montage_count": len(montages),
        "montages": montages,
    }
    target = output / "primary-montage-inventory.json"
    if target.exists():
        raise FileExistsError(target)
    target.write_text(json.dumps(inventory, ensure_ascii=True, indent=2) + "\n")
    print(json.dumps({
        "number": number,
        "primary_file_count": len(entries),
        "montage_count": len(montages),
        "montages": [item["path"] for item in montages],
    }, indent=2))


if __name__ == "__main__":
    main(int(sys.argv[1]))
