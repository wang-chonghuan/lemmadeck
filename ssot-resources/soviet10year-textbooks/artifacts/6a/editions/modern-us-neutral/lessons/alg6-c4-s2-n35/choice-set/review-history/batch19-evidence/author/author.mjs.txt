const questions = []
function add(id, number, stem, candidates, task, reason, loss, figures = []) {
  const ids = ['A', 'B', 'C', 'D']
  questions.push({
    id, source_numbers: [number],
    view: { mode: 'single', stem, figure_ids: figures,
      options: candidates.map((row, i) => ({ id: ids[i], text: row.text, figure_ids: [] })) },
    intent: {
      task, source_solution: source.sources.find(row => row.number === number).solution,
      unit_reason: reason, concept_blocks: [], limitations: [loss],
      figure_observations: figures.map(id => ({
        figure_id: id, observation: source.figures.find(row => row.id === id).observation,
      })),
    },
    key: { correct: candidates.flatMap((row, i) => row.truth === 'correct' ? [ids[i]] : []),
      rationales: candidates.map((row, i) => ({
        option_id: ids[i], truth: row.truth, derivation: row.derivation,
        misconception: row.misconception,
      })) },
  })
}
