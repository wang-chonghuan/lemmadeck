"""Bind independently observed display notes to exact screenshot and native-copy evidence."""

import hashlib
import json
import re
from pathlib import Path

from PIL import Image

from prepare_primary import OUT, ROOT, sha


def read(name):
    return json.loads((OUT / name).read_text())


def primary_evidence(entry):
    surface, width, kind = entry["surface"], entry["viewport_width"], entry["kind"]
    if surface == "text":
        return (
            "Title, prose and example fractions/superscripts are painted with no observed overlap. "
            + ("Complete short lesson and final exercise link fit the desktop screen; first/last are identical pixels."
               if width == 1440 else
               "Mobile first/last collectively cover the examples and final exercise link; formula lines wrap within the column.")
        )
    if kind in ("first", "last"):
        if surface == "choices":
            return "Final coefficient options A-D and page bottom are painted at this width; labels, formula powers and text do not overlap."
        if kind == "first":
            return "Desktop title and 610/611 blocks paint all twelve reductions and eight nested-power formulas, with separate labels and complete fraction bars."
        return "Graph and final 614 stem/formulas and return link paint through the page bottom; axes/curve and paragraph boundaries are visible."
    if kind.startswith("table-"):
        return "Intentional mobile right-end window shows final numeric columns, value row and closing bottom/right table borders; complete table windows are also independently observed in the supplement."
    if "figure-1" in kind:
        return (
            "Source blank table display is painted: x and f(x) labels, header grid and intentionally empty value row. "
            + ("Right-end window paints headers through 3 and outer/bottom borders."
               if kind.endswith("-end") else
               "Initial window paints the left labels/headers and bottom border; any overflow is covered by recorded native end/middle windows.")
        )
    return (
        "Source graph display paints grid, curve, x/y axes and tick labels. "
        + ("Right-end window paints positive-x branch, endpoint, arrow and bottom grid extent."
           if kind.endswith("-end") else
           "Initial window paints the left curve/hump and axis ticks. Whole extent is covered collectively by observed native windows; mobile exercises additionally have an observed central-axis/valley window.")
    )


def main():
    primary = read("primary-native-manifest.json")
    supplement = read("supplement-native-manifest.json")
    notes = read("actual-pixel-notes.json")
    rejected = read("primary-observations-before-supplement.json")
    entries = primary["screenshots"] + supplement["screenshots"]
    montages = primary["montages"] + supplement["montages"]
    assert len(entries) == 109 and len(montages) == 11
    assert {Path(item["path"]).name for item in montages} == set(notes["actually_opened_montages"])
    assert len({item["path"] for item in entries}) == len(entries)
    membership = {}
    for montage in montages:
        montage_path = ROOT / montage["path"]
        assert sha(montage_path) == montage["sha256"]
        with Image.open(montage_path) as image:
            canvas = image.convert("RGB")
        for member in montage["members"]:
            assert member["path"] not in membership
            source_path = ROOT / member["path"]
            assert sha(source_path) == member["sha256"]
            with Image.open(source_path) as source:
                original = source.convert("RGB")
            x, y, width, height = member["member_rect"]
            cropped = canvas.crop((x, y, x + width, y + height))
            assert original.size == (width, height)
            assert original.tobytes() == cropped.tobytes()
            pixel_hash = hashlib.sha256(original.tobytes()).hexdigest()
            membership[member["path"]] = {
                "path": str(montage_path), "sha256": montage["sha256"],
                "member_rect": member["member_rect"], "resized": False,
                "montage_actually_opened": True,
                "source_rgb_pixel_sha256": pixel_hash,
                "montage_member_rgb_pixel_sha256": hashlib.sha256(cropped.tobytes()).hexdigest(),
                "all_native_member_pixels_match_after_reopening_saved_png": True,
            }
    assert set(membership) == {item["path"] for item in entries}
    rejected_by_name = {item["filename"]: item for item in rejected["rejected_originals"]}
    observations = []
    groups = {}
    for entry in entries:
        assert sha(ROOT / entry["path"]) == entry["sha256"]
        original_rejection = rejected_by_name.get(entry["filename"])
        evidence = original_rejection["evidence"] if original_rejection else None
        if not evidence and entry["origin"] == "primary":
            evidence = primary_evidence(entry)
        if not evidence and entry.get("candidate_table_number"):
            number = entry["candidate_table_number"]
            evidence = notes["candidate_table_distinguishing_pixels"][str(number)] + " " + notes["candidate_table_common"]
            if entry["window"] != "full":
                evidence += f" This member is the intentional native {entry['window']} window; cropped opposite edge is covered by its overlapping counterpart, not claimed visible in this member."
            group = (entry["viewport_width"], number)
            groups.setdefault(group, []).append(entry)
        elif entry.get("candidate_table_number"):
            groups.setdefault((entry["viewport_width"], entry["candidate_table_number"]), []).append(entry)
        if not evidence:
            kind = entry["kind"]
            if kind == "fresh-first":
                key = f'{entry["surface"]}-fresh-first'
            else:
                key = kind
            evidence = notes["supplement_region_evidence"][key]
        observation = {
            "filename": entry["filename"], "path": str(ROOT / entry["path"]),
            "sha256": entry["sha256"], "actually_opened": True,
            "observer": "independent_agent", "status": "rejected" if original_rejection else "pass",
            "scope": "specified native display window, not semantic approval",
            "surface": entry["surface"], "viewport_width": entry["viewport_width"],
            "origin": entry["origin"], "size": entry["size"], "evidence": evidence,
            "observed_via_native_pixel_montage": membership[entry["path"]],
        }
        if original_rejection:
            observation["failure_kind"] = original_rejection["failure_kind"]
        for key in ("candidate_table_number", "window", "scroll_left", "scroll_width",
                    "client_width", "covered_interval", "source_figure_number"):
            if key in entry:
                observation[key] = entry[key]
        observations.append(observation)
    table_coverage = []
    assert len(groups) == 48
    for (width, number), members in sorted(groups.items()):
        windows = [member for member in members if "covered_interval" in member]
        if windows:
            intervals = sorted(member["covered_interval"] for member in windows)
            end = 0
            for left, right in intervals:
                assert left <= end
                end = max(end, right)
            assert end == windows[0]["scroll_width"]
        table_coverage.append({
            "candidate_table_number": number, "viewport_width": width,
            "status": "pass", "full_height_and_all_horizontal_columns_actually_observed": True,
            "members": [str(ROOT / member["path"]) for member in members],
            "unique_display_combination_increment": 0,
        })
    html_checks = []
    for item in primary["rendered_html"]:
        current_hash = sha(ROOT / item["path"])
        assert current_hash == item["sha256"]
        html_checks.append({**item, "current_sha256": current_hash, "unchanged": True})
    for item in primary["source_evidence_preserved"]:
        assert sha(ROOT / item["original"]) == item["sha256"]
        assert sha(ROOT / item["preserved"]) == item["sha256"]
    for item in supplement["capture_reports"]:
        assert sha(ROOT / item["path"]) == item["sha256"]
    combinations = sorted({(entry["surface"], entry["viewport_width"]) for entry in entries})
    assert len(combinations) == 6
    resolutions = []
    for original in rejected["rejected_originals"]:
        surface = "exercises" if "exercises" in original["filename"] else "choices"
        width = 1440 if "-1440-" in original["filename"] else 390
        fresh = next(item for item in observations if item["origin"] != "primary"
                     and item["surface"] == surface and item["viewport_width"] == width
                     and "fresh-first" in item["filename"])
        resolutions.append({
            **original, "original_status_remains": "rejected",
            "accepted_replacement_path": fresh["path"], "accepted_replacement_sha256": fresh["sha256"],
            "replacement_actually_opened": True,
            "diagnosis": "original capture partial-paint artifact; same-HTML fresh capture and full regions paint"
                         if original["failure_kind"] == "partial_paint"
                         else "original endpoint mispositioning; same-HTML fresh endpoint shows document top",
        })
    payload = {
        "lesson": "alg6-c4-s2-n35", "status": "pass",
        "observer": "independent_agent", "approval_scope": "actual display only",
        "semantic_approval": False, "source_or_new_figure_approval": False, "new_figure_reviews": [],
        "html_unchanged": True, "rendered_html_checks": html_checks,
        "unique_surface_viewport_combinations": 6,
        "surface_viewport_combinations": [{"surface": surface, "width": width,
                                         "height": 960 if width == 1440 else 844}
                                        for surface, width in combinations],
        "additional_unique_surface_viewport_combinations": 0,
        "actual_png_count": len(entries), "actually_opened_png_count": len(entries),
        "primary_png_count": 29, "supplementary_png_count": 80,
        "native_montage_png_count": 11, "actually_opened_montage_count": 11,
        "counting_note": "109 distinct screenshot files, not 109 unique displays; montage containers are excluded from screenshot counts. Two identical desktop text endpoint files remain two opened files, not extra combinations.",
        "original_endpoint_png_count": 12, "fresh_endpoint_png_count": 3,
        "accepted_endpoint_positions_after_replacement": 12,
        "rejected_original_png_count": 3, "accepted_observation_png_count": 106,
        "rejected_history_preserved": True,
        "before_supplement_record": {"path": str(OUT / "primary-observations-before-supplement.json"),
                                    "sha256": sha(OUT / "primary-observations-before-supplement.json")},
        "rejected_originals_and_resolutions": resolutions,
        "candidate_tables": {
            "unique_candidate_table_count": 24, "viewport_coverage_group_count": 48,
            "supplementary_table_window_png_count": 51,
            "primary_table_end_png_count": 3, "coverage": table_coverage,
        },
        "formula_observations": {
            "non_table_horizontal_overflow_regions_in_same_html_scan": 0,
            "numeric_option_png_count": 6, "choice_stem_png_count": 10,
            "full_exercise_region_png_count": 4,
            "evidence": "Actual numeric tuples, intermediate choice stems, and all 610/611/612 exercise formulas are painted. Existing text endpoint formulas fit/wrap. No unnecessary formula-end captures added for nonoverflowing rows.",
        },
        "source_figure_display_observations": {
            "primary_png_count": 14, "supplementary_native_window_png_count": 6,
            "distinct_actual_figures": 2,
            "evidence": "Source blank table and graph display whole extents through observed overlapping windows at desktop/mobile in exercises/choices. Mobile exercise table 0/1/2 center cells and central graph y-axis/valley are explicitly observed.",
            "source_semantics_reviewed": False,
        },
        "capture_reports": supplement["capture_reports"],
        "opening_history": notes["opening_history"],
        "observations": observations,
        "handoff": "Display-only pass after same-HTML replacement captures. Keep all three original captures rejected and unchanged. No content, render, semantic, canonical, promotion, app, DB or dependency work performed.",
    }
    with (OUT / "observations.json").open("x") as handle:
        json.dump(payload, handle, indent=2)
        handle.write("\n")
    print(json.dumps({
        "saved": str(OUT / "observations.json"), "status": payload["status"],
        "actually_opened_png_count": len(entries), "native_montage_count": len(montages),
        "unique_combinations": len(combinations), "unique_candidate_tables": 24,
        "rejected_originals_retained": 3,
        "all_109_source_and_montage_member_pixels_reopened_verified": True,
    }))


if __name__ == "__main__":
    main()
