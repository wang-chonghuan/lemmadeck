async function settle(page) {
  await page.evaluate(() => document.fonts.ready)
  await page.evaluate(() => new Promise(resolve =>
    requestAnimationFrame(() => requestAnimationFrame(resolve))))
  await page.waitForTimeout(180)
}
async function bounds(locator) {
  return locator.evaluate(node => {
    const r = node.getBoundingClientRect()
    return { x: r.x + scrollX, y: r.y + scrollY, width: r.width, height: r.height }
  })
}
async function region(page, surface, viewport, label, rectangle, reason) {
  const tileHeight = viewport.height - 32
  const count = Math.ceil(rectangle.height / tileHeight)
  for (let i = 0; i < count; i++) {
    const offset = i * tileHeight
    const height = Math.min(tileHeight, rectangle.height - offset)
    await page.evaluate(top => scrollTo({ top, left: 0, behavior: 'instant' }),
      Math.max(0, rectangle.y + offset - 16))
    await settle(page)
    const scroll = await page.evaluate(() => ({ x: scrollX, y: scrollY }))
    const left = Math.max(0, Math.floor(rectangle.x - scroll.x))
    const top = Math.max(0, Math.floor(rectangle.y + offset - scroll.y))
    const width = Math.min(Math.ceil(rectangle.width), viewport.width - left)
    const cropHeight = Math.min(Math.ceil(height), viewport.height - top)
    if (width < 1 || cropHeight < height - 2) throw new Error(`Invisible target ${label}`)
    const pixels = await sharp(await page.screenshot({ animations: 'disabled' }))
      .extract({ left, top, width, height: cropHeight }).png().toBuffer()
    const stats = await sharp(pixels).stats()
    if (stats.channels.slice(0, 3).every(c => c.min > 245)) {
      throw new Error(`Blank target ${label}`)
    }
    const file = path.join(output, 'supplements',
      `${surface}-${viewport.width}-${label}${count > 1 ? `-tile-${i + 1}` : ''}.png`)
    fs.writeFileSync(file, pixels, { flag: 'wx' })
    log.captures.push({
      path: file, sha256: sha(pixels), width, height: cropHeight,
      group: `${surface}-${viewport.width}`, surface, viewport,
      target: label, missing_coverage_reason: reason,
      document_bounds: rectangle, scroll,
      native_viewport_crop: { left, top, width, height: cropHeight },
      tile_index: i + 1, tile_count: count, resized: false
    })
  }
}
async function capture(page, surface, viewport, label, locator, reason) {
  await locator.scrollIntoViewIfNeeded()
  await settle(page)
  await region(page, surface, viewport, label, await bounds(locator), reason)
}
async function range(page, surface, viewport, label, first, last, reason) {
  await first.scrollIntoViewIfNeeded()
  await settle(page)
  const a = await bounds(first)
  const b = await bounds(last)
  await region(page, surface, viewport, label,
    { x: a.x, y: a.y, width: Math.max(a.width, b.width), height: b.y + b.height - a.y }, reason)
}
async function horizontalEnds(page, surface, viewport, label, locator) {
  const handles = await locator.evaluateHandle(node => {
    const elements = [node, ...node.querySelectorAll('*')]
    return elements.filter(el => el.scrollWidth > el.clientWidth + 2
      && ['auto', 'scroll'].includes(getComputedStyle(el).overflowX))
      .filter(el => !elements.some(parent => parent !== el
        && parent.contains(el) && parent.scrollWidth > parent.clientWidth + 2
        && ['auto', 'scroll'].includes(getComputedStyle(parent).overflowX)))
  })
  const properties = await handles.getProperties()
  let index = 0
  for (const handle of properties.values()) {
    const element = handle.asElement()
    if (!element) { await handle.dispose(); continue }
    index++
    const metrics = await element.evaluate(el => ({
      clientWidth: el.clientWidth, scrollWidth: el.scrollWidth, tag: el.tagName, class: el.className
    }))
    log.measurements.push({ label, viewport, ...metrics })
    const maximum = metrics.scrollWidth - metrics.clientWidth
    const positions = []
    for (let p = metrics.clientWidth; p < maximum; p += metrics.clientWidth) positions.push(p)
    positions.push(maximum)
    for (let i = 0; i < positions.length; i++) {
      await element.evaluate((el, p) => { el.scrollLeft = p }, positions[i])
      await settle(page)
      await capture(page, surface, viewport, `${label}-horizontal-${index}-${i + 1}`,
        element, 'Unseen horizontal end of newly supplemented missing text/options, native scroll reachability.')
    }
    await element.evaluate(el => { el.scrollLeft = 0 })
    await handle.dispose()
  }
  await handles.dispose()
}
