const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const { createRequire } = require('node:module');
const { pathToFileURL } = require('node:url');

delete process.env.PWDEBUG;
process.env.PLAYWRIGHT_HTML_OPEN = 'never';
process.env.PLAYWRIGHT_HTML_REPORT_OPEN = 'never';
const repo = path.resolve(__dirname, '../../../../..');
const { chromium } = createRequire(path.join(repo, 'app/package.json'))('@playwright/test');
const sha256 = bytes => crypto.createHash('sha256').update(bytes).digest('hex');

async function main() {
  const browser = await chromium.launch({ headless: true });
  try {
    for (const number of [39, 40]) {
      const scratch = path.join(repo, `.intentfold/tickets/STEMROBIN-182/tmp/6a-n${number}`);
      const seed = JSON.parse(fs.readFileSync(path.join(scratch, 'seed.json'), 'utf8'));
      const context = await browser.newContext({
        viewport: { width: 1400, height: 1400 },
        deviceScaleFactor: 1,
      });
      const captures = [];
      try {
        await context.route(/https?:\/\//, route => route.abort());
        const page = await context.newPage();
        for (const figure of seed.figures) {
          for (const asset of figure.assets) {
            const sourcePath = path.join(repo, asset.path);
            const sourceHash = sha256(fs.readFileSync(sourcePath));
            if (sourceHash !== asset.sha256) throw new Error(`Changed source asset: ${asset.path}`);
            await page.goto(pathToFileURL(sourcePath).href);
            await page.evaluate(() => document.fonts.ready);
            const svg = page.locator('svg').first();
            const box = await svg.boundingBox();
            await page.setViewportSize({
              width: Math.max(400, Math.ceil(box.x + box.width + 32)),
              height: Math.max(400, Math.ceil(box.y + box.height + 32)),
            });
            const output = path.join(scratch, `${figure.id}-actual-full.png`);
            await svg.screenshot({ path: output, scale: 'css' });
            const bytes = fs.readFileSync(output);
            captures.push({
              figure_id: figure.id,
              actual_source_path: asset.path,
              actual_source_sha256: sourceHash,
              complete_render_path: path.relative(repo, output),
              complete_render_sha256: sha256(bytes),
              width: bytes.readUInt32BE(16),
              height: bytes.readUInt32BE(20),
              source_bytes_unchanged: sha256(fs.readFileSync(sourcePath)) === sourceHash,
            });
          }
        }
      } finally {
        await context.close();
      }
      fs.writeFileSync(path.join(scratch, 'actual-figure-captures.json'),
        JSON.stringify({
          scope: 'Full actual seed SVG rendering only; no new diagrams or mathematical source reconstruction.',
          browser: { headless: true, isolated_context: true, user_profile: false, custom_launcher: false },
          PWDEBUG_present: Boolean(process.env.PWDEBUG),
          PLAYWRIGHT_HTML_OPEN: process.env.PLAYWRIGHT_HTML_OPEN,
          PLAYWRIGHT_HTML_REPORT_OPEN: process.env.PLAYWRIGHT_HTML_REPORT_OPEN,
          captures,
          model_usage: null,
        }, null, 2) + '\n');
    }
  } finally {
    await browser.close();
  }
}

main().catch(error => {
  console.error(error);
  process.exitCode = 1;
});
