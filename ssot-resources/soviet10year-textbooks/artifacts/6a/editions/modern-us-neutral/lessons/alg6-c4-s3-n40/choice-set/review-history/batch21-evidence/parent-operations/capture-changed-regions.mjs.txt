function png(file, details) {
  const bytes = fs.readFileSync(file);
  return { path: file, sha256: sha(file), width: bytes.readUInt32BE(16), height: bytes.readUInt32BE(20),
    native_scale: 1, ...details };
}
async function geometry(page, surface) {
  return page.evaluate((surface) => {
    const selectors = { text: 'p.para', exercises: 'li.ex', choices: 'section.choice' };
    const box = (node) => {
      const rect = node.getBoundingClientRect();
      return { x: rect.x, y: rect.y + scrollY, width: rect.width, height: rect.height,
        client_width: node.clientWidth, scroll_width: node.scrollWidth };
    };
    return { headers: { h1: box(document.querySelector('h1')), crumb: box(document.querySelector('.crumb')) },
      document: { viewport_width: innerWidth, client_width: document.documentElement.clientWidth,
        scroll_width: document.documentElement.scrollWidth },
      objects: [...document.querySelectorAll(selectors[surface])].map((node, index) => ({
        index: index + 1, id: node.id || `${surface}-${index + 1}`, ...box(node),
        body: node.querySelector('.body') ? box(node.querySelector('.body')) : null,
        formulae: [...node.querySelectorAll('.katex-html')].map((formula) => box(formula)),
        figures: [...node.querySelectorAll('.fig')].map((figure) => ({ id: figure.id, ...box(figure),
          media: box(figure.querySelector('.figure-media')) })),
      })) };
  }, surface);
}
