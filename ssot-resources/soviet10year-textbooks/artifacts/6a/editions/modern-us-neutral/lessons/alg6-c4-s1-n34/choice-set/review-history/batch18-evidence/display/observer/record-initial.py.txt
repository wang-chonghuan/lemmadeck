def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    manifest = json.loads((OUT / "endpoint-manifest.json").read_text())
    capture = json.loads((OUT.parent / "batch18-render/capture-manifest.json").read_text())
    observations = []
    for montage in manifest["montages"]:
        if sha(ROOT / montage["path"]) != montage["sha256"]:
            raise RuntimeError("Opened montage changed")
        for entry in montage["members"]:
            if sha(ROOT / entry["path"]) != entry["sha256"]:
                raise RuntimeError("Actual screenshot changed")
            observations.append({
                "filename": entry["filename"],
                "sha256": entry["sha256"],
                "actually_opened": True,
                "evidence": {
                    "method": "All four native endpoint members actually opened in labeled surface montage; no resizing.",
                    "display_observation": NOTES[(entry["surface"], entry["viewport_width"], entry["endpoint"])],
                    "surface": entry["surface"],
                    "viewport_width": entry["viewport_width"],
                    "endpoint": entry["endpoint"],
                    "montage": {
                        "path": montage["path"], "sha256": montage["sha256"],
                        "member_rect": entry["member_rect"], "resized": False,
                    },
                    "report": capture["report"],
                    "scope": "Actual display only; no semantic/core, source-figure or publication approval.",
                },
                "observer": "Codex independent actual-display-only observer, current chat",
            })
    payload = {
        "lesson": manifest["lesson"],
        "status": "blocked",
        "observations": observations,
        "unique_surface_viewport_combinations": 6,
        "blocked_findings": [{
            "filename": "alg6-c4-s1-n34-choices-390-first.png",
            "finding": "Option 1B final exponent/punctuation is off-window at starting scroll.",
            "disposition": "Retain initial incomplete evidence; require actually opened right-end pixels.",
        }],
        "partial_paint_failures": [],
        "inputs": capture["inputs"],
        "rendered": capture["rendered"],
        "scope": "Actual supplied endpoint pixels only; not semantic approval or unseen intermediate-row inspection.",
    }
    with (OUT / "observations-before-supplement.json").open("x") as handle:
        json.dump(payload, handle, indent=2)
        handle.write("\n")
    print("Retained initial blocked endpoint observations: 12 opened PNGs, six combinations.")


if __name__ == "__main__":
    main()
