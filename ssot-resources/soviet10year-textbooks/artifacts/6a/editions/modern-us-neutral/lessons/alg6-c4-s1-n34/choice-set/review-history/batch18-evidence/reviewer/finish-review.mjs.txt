import fs from 'node:fs'

const own = path.dirname(fileURLToPath(import.meta.url))
const author = path.join(own, '../6a-n34')
const read = file => JSON.parse(fs.readFileSync(file, 'utf8'))

const view = read(path.join(own, 'view-checks.json'))
const draft = read(path.join(author, 'draft.json'))
const seed = read(path.join(author, 'seed.json'))
const digest = file => {
  const result = spawnSync('python3', ['.agents/skills/ld-s10y-choice/scripts/prepare.py', 'digest', file], { encoding: 'utf8' })
  assert.equal(result.status, 0, result.stderr)
  return result.stdout.trim()
}
assert.equal(view.counts.option_truths, 31)
assert.equal(draft.questions.length, 9)
const comparisons = draft.questions.map(question => {
  const independent = view.questions.find(row => row.id === question.id)
  assert(independent)
  assert.deepEqual(independent.correct, question.key.correct)
  for (const check of independent.options) {
    const rationale = question.key.rationales.find(row => row.option_id === check.id)
    assert.equal(rationale.truth, check.truth)
    assert.equal(rationale.misconception === null, check.truth === 'correct')
  }
  assert(question.intent.limitations.length > 0)
  return { question_id: question.id, true_sets_match: true,
    all_internal_derivations_and_misprocedures_read: true,
    option_count: independent.options.length,
    local_limitations_match_source_to_choice_production_loss: true }
})
