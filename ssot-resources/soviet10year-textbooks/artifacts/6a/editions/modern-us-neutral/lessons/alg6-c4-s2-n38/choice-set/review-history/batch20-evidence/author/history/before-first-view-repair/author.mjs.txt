const table = (letters, values) => '| 小问 | 结果 |\n| --- | --- |\n' +
  values.map((value, i) => `| ${letters[i]}) | ${value} |`).join('\n')
const mathTable = (letters, values) => table(letters, values.map(v => `$${v}$`))
const tuple = values => '$(' + values.join(',\\ ') + ')$'
const questions = []
function add(id, number, stem, candidates, task, reason, losses, figures = []) {
  const ids = ['A', 'B', 'C', 'D']
  questions.push({
    id, source_numbers: [number],
    view: { mode: 'single', stem, figure_ids: figures,
      options: candidates.map((row, i) => ({ id: ids[i], text: row.text, figure_ids: [] })) },
    intent: { task, source_solution: source.sources.find(row => row.number === number).solution,
      unit_reason: reason, concept_blocks: ['624', '625'].includes(number) ? [1] : [],
      limitations: losses, figure_observations: figures.map(id => ({
        figure_id: id, observation: source.figures.find(row => row.id === id).observation,
      })) },
    key: { correct: candidates.flatMap((row, i) => row.truth === 'correct' ? [ids[i]] : []),
      rationales: candidates.map((row, i) => ({
        option_id: ids[i], truth: row.truth, derivation: row.derivation,
        misconception: row.misconception,
      })) },
  })
}
