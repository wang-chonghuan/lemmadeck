async function settle(page) {
  await page.evaluate(() => document.fonts.ready)
  await page.evaluate(() => new Promise((resolve) => {
    requestAnimationFrame(() => requestAnimationFrame(resolve))
  }))
  await page.waitForTimeout(180)
}

async function box(locator) {
  return locator.evaluate((node) => {
    const bounds = node.getBoundingClientRect()
    return {
      x: bounds.x + scrollX,
      y: bounds.y + scrollY,
      width: bounds.width,
      height: bounds.height,
    }
  })
}

async function captureBox(page, number, surface, viewport, label, bounds, reason) {
  const available = viewport.height - 24
  const tiles = Math.ceil(bounds.height / available)
  for (let index = 0; index < tiles; index++) {
    const offset = index * available
    const height = Math.min(available, bounds.height - offset)
    await page.evaluate((top) => scrollTo({ top, left: 0, behavior: 'instant' }),
      Math.max(0, bounds.y + offset - 12))
    await settle(page)
    const origin = await page.evaluate(() => ({ x: scrollX, y: scrollY }))
    const x = Math.max(0, Math.floor(bounds.x - origin.x))
    const y = Math.max(0, Math.floor(bounds.y + offset - origin.y))
    const width = Math.min(Math.ceil(bounds.width), viewport.width - x)
    const clipHeight = Math.min(Math.ceil(height), viewport.height - y)
    if (width < 1 || clipHeight < height - 2) {
      throw new Error(`Requested native region not visible: ${label}`)
    }
    const filename = `${surface}-${viewport.width}-${label}`
      + (tiles > 1 ? `-tile-${index + 1}` : '') + '.png'
    const output = path.join(ticket,
      `reviewer-6a-n${number}-actual-display/supplements`, filename)
    if (fs.existsSync(output)) throw new Error(`No capture overwrite: ${output}`)
    const viewportPixels = await page.screenshot({ animations: 'disabled' })
    const pixels = await sharp(viewportPixels)
      .extract({ left: x, top: y, width, height: clipHeight }).png().toBuffer()
    const stats = await sharp(pixels).stats()
    if (stats.channels.slice(0, 3).every((channel) => channel.min > 245)) {
      throw new Error(`Blank pixel capture rejected: ${label}`)
    }
    fs.writeFileSync(output, pixels, { flag: 'wx' })
    logs.get(number).captures.push({
      path: output,
      sha256: sha(pixels),
      width,
      height: clipHeight,
      label: `${surface}-${viewport.width}-${label}`,
      group: `${surface}-${viewport.width}`,
      viewport,
      surface,
      target: label,
      missing_coverage_reason: reason,
      tile_index: index + 1,
      tile_count: tiles,
      document_bounds: bounds,
      scroll: origin,
      native_viewport_crop: { x, y, width, height: clipHeight },
      resized: false,
    })
  }
}

async function capture(page, number, surface, viewport, label, locator, reason) {
  await locator.scrollIntoViewIfNeeded()
  await settle(page)
  await captureBox(page, number, surface, viewport, label, await box(locator), reason)
}

async function range(page, number, surface, viewport, label, first, last, reason) {
  await first.scrollIntoViewIfNeeded()
  await settle(page)
  const a = await box(first)
  const b = await box(last)
  await captureBox(page, number, surface, viewport, label, {
    x: a.x,
    y: a.y,
    width: Math.max(a.width, b.width),
    height: b.y + b.height - a.y,
  }, reason)
}

async function horizontalRegion(locator) {
  return locator.evaluateHandle((node) => {
    for (let parent = node; parent; parent = parent.parentElement) {
      if (parent.scrollWidth > parent.clientWidth + 1
        && ['auto', 'scroll'].includes(getComputedStyle(parent).overflowX)) {
        return parent
      }
    }
    return null
  })
}

async function extraHorizontal(page, number, surface, viewport, label, locator,
  skipKnownRight, reason) {
  const handle = await horizontalRegion(locator)
  const element = handle.asElement()
  if (!element) {
    await handle.dispose()
    return
  }
  const metrics = await element.evaluate((node) => ({
    clientWidth: node.clientWidth,
    scrollWidth: node.scrollWidth,
  }))
  logs.get(number).measurements.push({ label, ...metrics })
  const max = metrics.scrollWidth - metrics.clientWidth
  const positions = max > metrics.clientWidth ? [max / 2, max] : [max]
  for (const position of positions) {
    if (skipKnownRight && position === max) continue
    await element.evaluate((node, value) => {
      node.scrollLeft = value
    }, position)
    await settle(page)
    await capture(page, number, surface, viewport,
      `${label}-${position === max ? 'right' : 'middle'}`,
      element, reason)
  }
  await element.evaluate((node) => { node.scrollLeft = 0 })
  await handle.dispose()
}
