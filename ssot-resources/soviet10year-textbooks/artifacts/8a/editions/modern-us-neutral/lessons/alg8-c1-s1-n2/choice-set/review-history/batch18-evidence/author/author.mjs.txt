const correct = (text, derivation) => ({ text, truth: 'correct', derivation, misconception: null })
const wrong = (text, derivation, misconception) => ({ text, truth: 'incorrect', derivation, misconception })
function question(id, number, mode, stem, options, task, unitReason, limitation, conceptBlocks = [3]) {
  const ids = ['A', 'B', 'C', 'D']
  questions.push({
    id, source_numbers: [number],
    view: { mode, stem, figure_ids: [], options: options.map((option, index) => ({
      id: ids[index], text: option.text, figure_ids: [],
    })) },
    intent: { task, source_solution: solutions.get(number), unit_reason: unitReason,
      concept_blocks: conceptBlocks, limitations: [limitation], figure_observations: [] },
    key: { correct: options.flatMap((option, index) => option.truth === 'correct' ? [ids[index]] : []),
      rationales: options.map((option, index) => ({
        option_id: ids[index], truth: option.truth, derivation: option.derivation,
        misconception: option.misconception,
      })) },
  })
}
