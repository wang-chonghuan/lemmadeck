const sha = b => crypto.createHash("sha256").update(b).digest("hex");
const objectSha = o => sha(Buffer.from(JSON.stringify(o)));
function bind(p, expected) {
  const b = fs.readFileSync(p);
  return {
    path: p,
    sha256: sha(b),
    bytes: b.length,
    ...(expected ? { receipt_sha256: expected, matches_receipt: sha(b) === expected } : {})
  };
}
function pngSize(p) {
  const b = fs.readFileSync(p);
  return { width: b.readUInt32BE(16), height: b.readUInt32BE(20) };
}

function sourceBlock(ref) {
  const match = /^p(\d+)#(\d+)$/.exec(ref);
  if (!match) return { ref, binding_error: "unrecognized source ref" };
  const page = Number(match[1]);
  const block = pageJson.get(page)?.blocks[Number(match[2]) - 1];
  return block ? { ref, pdf_page: page, kind: block.kind, block_json_sha256: objectSha(block), continuation: block.cont, open: block.open } :
    { ref, binding_error: "missing source block" };
}

function collect(value) {
  if (!value || typeof value !== "object") return;
  if (typeof value.path === "string" && typeof value.sha256 === "string" && typeof value.bytes === "number") allBound.push(value);
  Object.values(value).forEach(collect);
}
