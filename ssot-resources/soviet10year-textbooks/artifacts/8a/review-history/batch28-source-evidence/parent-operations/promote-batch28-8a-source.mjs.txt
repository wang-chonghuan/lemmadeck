const pending = []
const retained = {}
const files = new Map()
function retain(file, relative) {
  if (files.has(path.resolve(file))) return files.get(path.resolve(file))
  const target = `${evidence}/${relative}`
  assert(!fs.existsSync(target), target)
  const bytes = fs.readFileSync(file)
  const digest = hash(file)
  files.set(path.resolve(file), target)
  pending.push({ target, bytes })
  retained[relative] = { original_bytes_sha256: digest, retained_bytes_sha256: digest,
    inactive_history: target.endsWith('.txt') }
  return target
}
