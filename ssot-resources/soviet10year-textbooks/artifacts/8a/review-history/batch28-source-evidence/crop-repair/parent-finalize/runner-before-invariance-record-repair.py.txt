def read(file):
    return json.loads(file.read_text())


def sha(file):
    return hashlib.sha256(file.read_bytes()).hexdigest()


def save(file, value):
    with file.open("x") as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


review_file = scratch / "reviewer-8a-fig08-crop-repair/observations.json"
review = read(review_file)
assert review["status"] == "pass" and review["verdict"]["status"] == "pass"
receipt = read(repair / "repair-receipt.json")
old_box = receipt["geometry"]["old_logical_box_xywh"]
new_box = receipt["geometry"]["proposed_logical_box_xywh"]
assert receipt["geometry"]["logical_box_delta_xywh"] == [0, 0, 0, -9]
page = candidate / "root/8a/pages/0029"
work = candidate / "work/8a/pages/0029"
original = {}
for name in ("page.md", "page.json", "audit.json"):
    original[name] = (page / name).read_bytes()
    shutil.copyfile(page / name, evidence / ("before-" + name))
for name in ("fig-08.png", "fig-08.svg"):
    shutil.copyfile(work / "figures" / name, evidence / ("before-" + name))

meta, blocks = B.parse(original["page.md"].decode())
before_meta, before_blocks = copy.deepcopy(meta), copy.deepcopy(blocks)
figures = [block for block in blocks if block["kind"] == "fig"]
assert len(figures) == 1 and figures[0]["box"] == old_box
figures[0]["box"] = new_box
assert len(meta["figures"]) == 1 and meta["figures"][0]["box"] == old_box
meta["figures"][0]["box"] = new_box

# The reviewed machine crop is an exact geometry cache, not a new page-gate result.
cache = read(page / "audit.json")
assert cache["figures"][0]["box"] == old_box
cache["figures"][0]["box"] = new_box
cache["geometry_binding"] = {
    "purpose": "Reuse independently reviewed machine crop; not a new finalize result.",
    "review": str(review_file.relative_to(repo)),
    "review_sha256": sha(review_file),
    "generator_receipt_sha256": sha(repair / "repair-receipt.json"),
}
(page / "page.md").write_text(B.dump(meta, blocks, 29))
(page / "audit.json").write_text(json.dumps(cache, ensure_ascii=False, indent=2) + "\n")
save(evidence / "geometry-cache-input.json", cache)
python = repo / ".claude/skills/ld-s10y-lesson/.venv/bin/python"


def run(name, args):
    command = [str(python), str(tools / "p2c.py"), *args,
               "--book", "8a", "--root", str(candidate / "root"),
               "--work", str(candidate / "work")]
    started = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(command, cwd=repo, text=True, capture_output=True)
    save(evidence / (name + "-command.json"), {
        "command": command, "cwd": str(repo), "started": started,
        "finished": datetime.now(timezone.utc).isoformat(),
        "exit_code": result.returncode, "stdout": result.stdout,
        "stderr": result.stderr, "execution_count": 1,
    })
    assert result.returncode == 0, result.stdout + result.stderr
