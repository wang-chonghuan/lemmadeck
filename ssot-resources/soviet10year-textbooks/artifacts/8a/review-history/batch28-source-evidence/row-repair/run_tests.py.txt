"""Run only selected counting tests and retain exclusive command receipts."""
import argparse
import contextlib
import io
import json
import sys
import unittest
from pathlib import Path

REPO = Path("/Users/yong/.codex/worktrees/d9b1/lemmadeck")
OWN = Path(__file__).resolve().parent
TESTS = REPO / ".claude/skills/ld-s10y-lesson/tests"
FOCUSED = "test_four_row_pixel_support.py"
AFFECTED = [
    FOCUSED,
    "test_layout.py",
    "test_fraction_line_spacing.py",
    "test_fraction_component_support.py",
    "test_isolated_scan_streak.py",
    "test_nested_fraction_scan_noise.py",
    "test_compound_fraction_rows.py",
    "test_short_numeric_rows.py",
    "test_fraction_dominated_short_digits.py",
    "test_inline_fraction_row.py",
    "test_standalone_quotient.py",
    "test_adjacent_exercise_subpart.py",
    "test_formula_columns.py",
    "test_6a_printed_rows.py",
]

parser = argparse.ArgumentParser()
parser.add_argument("name")
parser.add_argument("--affected", action="store_true")
args = parser.parse_args()
assert Path.cwd() == REPO
patterns = AFFECTED if args.affected else [FOCUSED]
suite = unittest.TestSuite()
loader = unittest.TestLoader()
for pattern in patterns:
    assert (TESTS / pattern).is_file(), pattern
    suite.addTests(loader.discover(str(TESTS), pattern=pattern))
log = io.StringIO()
with contextlib.redirect_stdout(log), contextlib.redirect_stderr(log):
    result = unittest.TextTestRunner(stream=log, verbosity=2).run(suite)
with (OWN / f"{args.name}.txt").open("x") as stream:
    stream.write(log.getvalue())
receipt = {
    "workdir": str(REPO), "python": sys.executable,
    "patterns": patterns, "tests_run": result.testsRun,
    "failures": len(result.failures), "errors": len(result.errors),
    "skipped": len(result.skipped), "success": result.wasSuccessful(),
    "output": str(OWN / f"{args.name}.txt"),
}
with (OWN / f"{args.name}.json").open("x") as stream:
    json.dump(receipt, stream, indent=2)
    stream.write("\n")
print(json.dumps(receipt))
if not result.wasSuccessful():
    print(log.getvalue())
sys.exit(not result.wasSuccessful())
