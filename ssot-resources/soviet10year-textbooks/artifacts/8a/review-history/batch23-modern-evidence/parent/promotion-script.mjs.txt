const pending = new Map()

function remember(file, target) {
  file = relative(file)
  if (mappings.has(file)) return
  assert(!fs.existsSync(target))
  const data = bytes(file)
  pending.set(target, data)
  mappings.set(file, target)
  retained[path.relative(evidence, target)] = {
    original_bytes_sha256: hash(data), retained_bytes_sha256: hash(data),
    inactive_history: target.endsWith('.txt'),
  }
}

function rebind(value) {
  if (Array.isArray(value)) return value.map(rebind)
  if (value && typeof value === 'object') return Object.fromEntries(Object.entries(value)
    .map(([key, child]) => [key, rebind(child)]))
  if (typeof value !== 'string') return value
  const normalized = relative(value)
  if (mappings.has(normalized)) return mappings.get(normalized)
  if (normalized.startsWith(`${scratch}/`) || normalized.startsWith('.tmp/')) {
    return `Historical execution location, not retained dependency: ${normalized.slice(normalized.indexOf('/') + 1)}`
  }
  return value
}
function rebindNotes(text) {
  for (const [from, to] of [...mappings].sort((a, b) => b[0].length - a[0].length)) {
    text = text.split(path.resolve(from)).join(to).split(from).join(to)
  }
  assert.doesNotMatch(text, /\.intentfold\/|\.tmp\/|\/Users\//)
  return text
}
