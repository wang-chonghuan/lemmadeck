import fs from 'node:fs'
import path from 'node:path'
import crypto from 'node:crypto'
import assert from 'node:assert/strict'
import { createRequire } from 'node:module'
import { spawnSync } from 'node:child_process'
import { fileURLToPath } from 'node:url'

const work = path.dirname(fileURLToPath(import.meta.url))
const repo = '/Users/yong/.codex/worktrees/d9b1/lemmadeck'
const tool = path.join(repo, '.claude/skills/ld-s10y-lesson/tools/prose_flow.mjs')
const require = createRequire(import.meta.url)
const { proseOutline, sectionBreakIndices } = require(path.join(repo, '.claude/skills/ld-s10y-lesson/tools/htmlfrag.js'))
const sha = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')
for (const [id, starts] of [
  ['alg8-c1-s1-n3', [1, 4, 8, 10, 12, 14]],
  ['alg8-c1-s2-n4', [10, 22, 23, 25]]
]) {
  const file = path.join(work, 'adapt/8a/modern-us-neutral/lessons', id, 'lesson.template.json')
  const output = path.join(work, `prose-flow-${id}.json`)
  assert(!fs.existsSync(output))
  const before = sha(file)
  const result = spawnSync('node', [tool, '--json', file], { encoding: 'utf8' })
  assert.equal(result.status, 0, result.stderr)
  const outline = JSON.parse(result.stdout)
  const boundaries = starts.map(blockIndex => {
    const index = outline.paragraphs.findIndex(p => p.blockIndex === blockIndex)
    assert(index > 0, `No adjacent transition for ${id} block${blockIndex}`)
    return { before: outline.paragraphs[index - 1].text, after: outline.paragraphs[index].text }
  })
  const lesson = JSON.parse(fs.readFileSync(file, 'utf8'))
  lesson.section_breaks = boundaries
  const resolved = sectionBreakIndices(lesson.prose, lesson.section_breaks)
  assert.equal(resolved.length, boundaries.length)
  assert.deepEqual(proseOutline(lesson.prose), outline.paragraphs)
  fs.writeFileSync(file, JSON.stringify(lesson, null, 2) + '\n')
  fs.writeFileSync(output, JSON.stringify({
    argv: ['node', tool, '--json', file], exit_code: result.status,
    stdout: result.stdout, stderr: result.stderr, template_sha256_before: before,
    actual_prose_flow: outline, selected_adjacent_boundaries: boundaries,
    resolved_boundaries_after: resolved, template_sha256_after: sha(file)
  }, null, 2) + '\n', { flag: 'wx' })
  console.log(JSON.stringify({ id, source_blocks: lesson.prose.length,
    final_paragraphs: outline.paragraphs.length, boundaries: boundaries.length, output }))
}
