import assert from 'node:assert/strict'
import crypto from 'node:crypto'
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const work = path.dirname(fileURLToPath(import.meta.url))
const history = path.join(work, 'history/before-three-text-review-repairs')
const records = []
const allowed = {
  'alg8-c1-s1-n3/lesson.template.json': ['/prose/7/text'],
  'alg8-c1-s1-n3/exercises.template.json': ['/exercises/8/text', '/exercises/8/lines/0', '/exercises/8/changes'],
  'alg8-c1-s2-n4/lesson.template.json': ['/prose/20/text'],
}
function diff(a, b, pointer = '') {
  if (JSON.stringify(a) === JSON.stringify(b)) return []
  if (a == null || b == null || typeof a !== 'object' || typeof b !== 'object'
      || Array.isArray(a) !== Array.isArray(b)
      || Array.isArray(a) && a.length !== b.length) return [{ pointer, before: a, after: b }]
  return [...new Set([...Object.keys(a), ...Object.keys(b)])]
    .flatMap(key => diff(a[key], b[key], `${pointer}/${key}`))
}
for (const id of ['alg8-c1-s1-n3', 'alg8-c1-s2-n4']) {
  for (const name of ['lesson.template.json', 'exercises.template.json', 'figures.template.json']) {
    const relative = `adapt/8a/modern-us-neutral/lessons/${id}/${name}`
    const original = path.join(history, relative)
    const current = path.join(work, relative)
    const differences = diff(JSON.parse(fs.readFileSync(original)), JSON.parse(fs.readFileSync(current)))
    assert.deepEqual(differences.map(row => row.pointer).sort(), (allowed[`${id}/${name}`] ?? []).sort())
    records.push({ path: current, original,
      sha256: crypto.createHash('sha256').update(fs.readFileSync(current)).digest('hex'),
      original_sha256: crypto.createHash('sha256').update(fs.readFileSync(original)).digest('hex'),
      differences })
  }
}
fs.writeFileSync(path.join(work, 'three-text-repair-ready.json'), JSON.stringify({
  status: 'ready for independent changed-text-only review',
  actual_changed_semantic_objects: 3, transport_or_changes_metadata_leaves: 2,
  RAW_modified: false, source_metadata_modified: false, figure_templates_modified: false,
  already_passing_text_not_re_reviewed: true, records,
}, null, 2) + '\n', { flag: 'wx' })
console.log(JSON.stringify({ ready: path.join(work, 'three-text-repair-ready.json'),
  changed_objects: 3 }))
