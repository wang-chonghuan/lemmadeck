import assert from 'node:assert/strict'

const sha = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')
const read = file => JSON.parse(fs.readFileSync(file, 'utf8'))
const record = file => ({ path: file, sha256: sha(file), bytes: fs.statSync(file).size })
const save = (file, data) => fs.writeFileSync(file, JSON.stringify(data, null, 2) + '\n', { flag: 'wx' })
const frozenFile = path.join(work, 'inputs-before.json')

function check() {
  const frozen = read(frozenFile)
  for (const item of frozen.inputs) assert.equal(sha(item.path), item.sha256, `Input changed: ${item.path}`)
  for (const item of frozen.copies) assert.equal(sha(item.destination), item.sha256, `Immutable copy changed: ${item.destination}`)
  return frozen
}
function once(name, executable, args, outputs = []) {
  check()
  const start = path.join(work, `${name}.started.json`), output = path.join(work, `${name}.json`)
  assert(!fs.existsSync(start) && !fs.existsSync(output), `Already attempted: ${name}`)
  save(start, { at: new Date().toISOString(), argv: [executable, ...args] })
  const temporary = path.join(work, 'runtime-temp')
  fs.mkdirSync(temporary, { recursive: true })
  const result = spawnSync(executable, args, {
    cwd: repo, encoding: 'utf8', maxBuffer: 16 * 1024 * 1024,
    env: { ...process.env, PYTHONDONTWRITEBYTECODE: '1', PYTHON: python, TMPDIR: temporary }
  })
  let failure = result.error?.message ?? (result.status === 0 ? null : `Exit ${result.status}`)
  try { check() } catch (error) { failure = error.message }
  save(output, {
    at: new Date().toISOString(), argv: [executable, ...args], status: result.status,
    stdout: result.stdout, stderr: result.stderr, pass: !failure, failure,
    outputs: outputs.filter(fs.existsSync).map(record)
  })
  process.stdout.write(result.stdout ?? '')
  process.stderr.write(result.stderr ?? '')
  if (failure) process.exitCode = result.status || 1
}
