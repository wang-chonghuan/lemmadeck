const point = (id, at, label, placement = 'SE') => ({
  id, type: 'point', at, size: 2, stroke: 'ink', fill: 'ink', strokeWidth: 1,

function sourceText(lessonId) {
  const lesson = read(path.join(templates(lessonId), 'lesson.template.json'))
  const exercises = read(path.join(templates(lessonId), 'exercises.template.json'))
  return [
    ...lesson.prose.filter(b => b.text).map(b => ({ lesson: lessonId, text: b.text })),
    ...exercises.exercises.map(e => ({ lesson: lessonId, exercise: e.number, text: e.text }))
  ]
}

function gridFrame(spec, bounds, step, columns, rows, xTicks, yTicks) {
  const [xmin, ymax, xmax, ymin] = bounds
  spec.objects.push(
    { id: 'source-grid', type: 'grid', bounds, xStep: step, yStep: step,
      xOffset: 0, yOffset: 0, stroke: 'grid', strokeWidth: 1.2 },
    { id: 'source-frame', type: 'polygon',
      points: [[xmin, ymin], [xmax, ymin], [xmax, ymax], [xmin, ymax]],
      stroke: 'ink', strokeWidth: 2, fillOpacity: 0 },
    { id: 'x-axis', type: 'axis', from: [xmin, 0], to: [xmax, 0], stroke: 'ink', strokeWidth: 2.5,
      ticks: xTicks.map(([value, label]) => ({ at: [value, 0], label, size: .09,
        fontSize: 28, labelPlacement: { position: 'S', offset: [0, 12] } })) },
    { id: 'y-axis', type: 'axis', from: [0, ymin], to: [0, ymax], stroke: 'ink', strokeWidth: 2.5,
      ticks: yTicks.map(([value, label]) => ({ at: [0, value], label, size: .09,
        fontSize: 28, labelPlacement: { position: 'W', offset: [-12, 0] } })) },
    arrow('x-positive-arrow', [xmax - .6, 0], [xmax, 0]),
    arrow('y-positive-arrow', [0, ymax - .6], [0, ymax]),
    text('x-label', [xmax + .48, -.25], 'x'),
    text('y-label', [-.25, ymax + .48], 'y')
  )
  spec.assertions.push(
    { id: 'source-grid-dimensions', type: 'gridDimensions', grid: 'source-grid', columns, rows },
    { id: 'axes-perpendicular', type: 'perpendicular',
      a: [xmin, 0], b: [xmax, 0], c: [0, ymin], d: [0, ymax] },
    connects('x-positive-arrow', [xmax - .6, 0], [xmax, 0]),
    connects('y-positive-arrow', [0, ymax - .6], [0, ymax])
  )
  spec.source.inventory.push(group('source-grid-and-axes',
    `Source has one shared${columns}-column by${rows}-row grid;step${step};bounds${JSON.stringify(bounds)}. Preserve its finite border,axes,positive arrows,and the source tick labels. No new numbered ticks.`,
    ['source-grid', 'source-frame', 'x-axis', 'y-axis', 'x-positive-arrow', 'y-positive-arrow', 'x-label', 'y-label'],
    ['source-grid-dimensions', 'axes-perpendicular', 'x-positive-arrow-connects', 'y-positive-arrow-connects'],
    ['gridDimensions']))
}
function parabola(spec, id, a, m, n, bounds) {
  const [xmin, ymax, xmax, ymin] = bounds
  const radius = Math.sqrt(((a > 0 ? ymax : ymin) - n) / a)
  const lo = Math.max(xmin, m - radius), hi = Math.min(xmax, m + radius)
  const evaluate = x => a * (x - m) ** 2 + n
  const control = [(lo + hi) / 2, evaluate(lo) + a * (lo - m) * (hi - lo)]
  const [bxmin, bymax, bxmax, bymin] = spec.canvas.boundingBox
  assert.equal(round((bxmax - bxmin) / (bymax - bymin)),
    round(spec.canvas.width / spec.canvas.height), 'Raw path and JSXGraph must share the exact screen transform')
  const screen = ([x, y]) => [
    round((x - bxmin) / (bxmax - bxmin) * spec.canvas.width),
    round((bymax - y) / (bymax - bymin) * spec.canvas.height)
  ]
  const p0 = [lo, evaluate(lo)], p2 = [hi, evaluate(hi)]
  const xy = p => screen(p).join(' ')
  spec.objects.push({ id, type: 'svgPath',
    d: `M ${xy(p0)} Q ${xy(control)} ${xy(p2)}`, stroke: 'ink', strokeWidth: 3.5 })
  const checks = [0, .25, .5, .75, 1].map(t => {
    const x = (1 - t) ** 2 * p0[0] + 2 * (1 - t) * t * control[0] + t ** 2 * p2[0]
    const y = (1 - t) ** 2 * p0[1] + 2 * (1 - t) * t * control[1] + t ** 2 * p2[1]
    assert(Math.abs(y - evaluate(x)) < 1e-8)
    return { t, x: round(x), y: round(y), equation_residual: round(y - evaluate(x)) }
  })
  return { id, coefficients: { a, m, n }, shown_window: bounds,
    source_space_quadratic_bezier: [p0, control, p2].map(p => p.map(round)), exact_curve_checks: checks }
}
const outputRecords = []
function retain(spec, lessonId, curves, additions) {
  const file = path.join(destination, `${spec.id}.spec.json`)
  save(file, spec)
  const observation = interpretation.actual_figure_observations.find(o => o.id === spec.id)
  save(path.join(work, `${spec.id}-author-context.json`), {
    source: record(spec.source.image.path), source_only_interpretation_sha256: frozen.interpretation_sha256,
    original_png_opened_individually: true, actual_source_observation: observation,
    templates: ['lesson.template.json', 'exercises.template.json'].map(name => record(path.join(templates(lessonId), name))),
    authoritativeText: spec.source.authoritativeText, source_inventory: spec.source.inventory,
    curve_geometry: curves, modern_additions_or_regularization: additions,
    curve_assertion_limit: 'Existing FigureSpec uses svgPath for exact quadratic Bezier curves. Independent review must compare the actual curve with source readings;sample residuals and renderer determinism are not source-fidelity approval.',
    independent_figure_review: 'pending parent dispatch;no review pass created', spec: record(file)
  })
  const manifestFile = path.join(templates(lessonId), 'figures.template.json')
  const manifest = read(manifestFile), index = manifest.figures.findIndex(f => f.id === spec.id)
  assert(index >= 0)
  const entry = manifest.figures[index]
  for (const field of ['png', 'artwork', 'generation']) delete entry[field]
  Object.assign(entry, {
    spec: `figures/${spec.id}.spec.json`, svg: `figures/${spec.id}.svg`,
    render: `figures/${spec.id}.render.json`, review: `figures/${spec.id}.review.json`
  })
  fs.writeFileSync(manifestFile, JSON.stringify(manifest, null, 2) + '\n')
  outputRecords.push({ id: spec.id, lesson: lessonId, spec: record(file),
    independent_review: 'pending', actual_source_png: record(spec.source.image.path) })
}
