def latin_parts(text):
    labels = dict(zip(edition.CYRILLIC_PARTS, edition.LATIN_PARTS))
    masked = edition.masked_math(text)
    depths = edition.nesting_depths(masked)
    markers = [
        m for m in edition.PART_MARKER.finditer(masked)
        if m.group("label") in labels and depths[m.start()] == 0
    ]
    for marker in reversed(markers):
        a, b = marker.span("label")
        text = text[:a] + labels[marker.group("label")] + text[b:]
    return text


results = []
for lesson_id in TARGETS:
    lesson_path = BASE / lesson_id / "lesson.template.json"
    exercise_path = BASE / lesson_id / "exercises.template.json"
    lesson = json.loads(lesson_path.read_text())
    exercises = json.loads(exercise_path.read_text())
    for index, block in enumerate(lesson["prose"]):
        if block["kind"] != "p":
            continue
        text = block["source_text"]
        for old, new in replacements[lesson_id].get(index, []):
            assert old in text, (lesson_id, index, old)
            text = text.replace(old, new)
        if (lesson_id, index) in appendices:
            text += "\n" + appendices[lesson_id, index]
        text = edition.normalize_prose_layout(latin_parts(text))
        errors = edition.validate_text(
            block["source_text"], text, ["layout", "wording"], [], f"{lesson_id}.prose[{index}]", []
        )
        assert not errors, errors
        block["text"] = text
        block["changes"] = ["layout", "wording"] if text != block["source_text"] else []
    for index, exercise in enumerate(exercises["exercises"]):
        text = latin_parts(exercise["source_text"])
        if exercise["number"] == "15":
            text = text.replace("约分：", "约分，并保留原分母不为零的限制：")
        elif exercise["number"] == "16":
            text = text.replace("作出函数的图象：", "作出函数的图象，保留原分式的定义域：")
        elif exercise["number"] == "28":
            text = text.replace("利用这个模型板作出", "括号中的逗号是小数分隔符。利用这个模型板作出")
        text = edition.normalize_numbered_subparts(text)
        changes = ["layout", "wording"] if text != exercise["source_text"] else []
        errors = edition.validate_text(
            exercise["source_text"], text, changes, [], f"{lesson_id}.exercise[{index}]", [],
            source_lines=exercises["source"]["data"]["exercises"][index]["lines"]
        )
        assert not errors, errors
        exercise["text"], exercise["lines"], exercise["changes"] = text, text.splitlines(), changes
    assert len(lesson["prose"]) == (16 if lesson_id.endswith("n3") else 26)
    assert len(exercises["exercises"]) == 11
    for path, value in ((lesson_path, lesson), (exercise_path, exercises)):
        path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")
        results.append({"path": str(path), "sha256": sha(path)})
