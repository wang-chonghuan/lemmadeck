const figureDir = path.join(base, 'root/8a/editions/modern-us-neutral/figures')

const read = file => JSON.parse(fs.readFileSync(file, 'utf8'))
const sha = file => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')
const record = file => ({ path: file, sha256: sha(file), bytes: fs.statSync(file).size })
const save = (file, value) => fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n', { flag: 'wx' })
const receiptFile = path.join(history, 'receipt.json')
const candidateFile = path.join(work, 'candidate-ready.json')

const outputFiles = id => [
  ...['spec.json', 'svg', 'render.json'].map(suffix => path.join(figureDir, `${id}.${suffix}`)),
  path.join(work, `${id}.preview.png`),
]
const allowed = new Set(ids.flatMap(outputFiles))

function check() {
  const frozen = read(receiptFile)
  for (const item of frozen.snapshots) {
    assert.equal(sha(item.preserved), item.sha256, `History changed: ${item.preserved}`)
    if (!allowed.has(item.path)) assert.equal(sha(item.path), item.sha256, `Unchanged file changed: ${item.path}`)
  }
  const inputs = read(path.join(work, 'inputs-before.json'))
  for (const item of inputs.inputs) assert.equal(sha(item.path), item.sha256)
  for (const item of inputs.copies) assert.equal(sha(item.destination), item.sha256)
  return frozen
}

function leafDiff(before, after, pointer = '') {
  if (JSON.stringify(before) === JSON.stringify(after)) return []
  if (before && after && typeof before === 'object' && typeof after === 'object'
    && Array.isArray(before) === Array.isArray(after)) {
    return [...new Set([...Object.keys(before), ...Object.keys(after)])].flatMap(key =>
      leafDiff(before[key], after[key], `${pointer}/${key.replaceAll('~', '~0').replaceAll('/', '~1')}`))
  }
  return [{ path: pointer, ...(before === undefined ? {} : { before }),
    ...(after === undefined ? {} : { after }),
    op: before === undefined ? 'add' : after === undefined ? 'remove' : 'replace' }]
}

    fs.mkdirSync(path.dirname(preserved), { recursive: true })
    fs.copyFileSync(file, preserved, fs.constants.COPYFILE_EXCL)

  })

} else if (action === 'diff') {
  const frozen = check()
  const diffs = ids.map(id => {
    const specFile = path.join(figureDir, `${id}.spec.json`)
    const old = frozen.snapshots.find(item => item.path === specFile)
    const before = read(old.preserved), after = read(specFile)
    assert.deepEqual(before.source.image, after.source.image)
    assert.deepEqual(before.source.authoritativeText, after.source.authoritativeText)
    assert.deepEqual(before.canvas, after.canvas)
    assert.deepEqual(before.display, after.display)
    if (id === 'fig-01') {
      const copy = structuredClone(after)
      const oldAxis = before.objects.find(o => o.id === 'x-axis')
      const newAxis = copy.objects.find(o => o.id === 'x-axis')
      for (const tick of newAxis.ticks) if (['-1', '1'].includes(tick.label)) {
        tick.labelPlacement = oldAxis.ticks.find(t => t.label === tick.label).labelPlacement
      }
      assert.deepEqual(copy, before, 'fig01 must differ only at two labelPlacement objects')
    } else if (id === 'fig-02') {
      assert.deepEqual(after.objects.filter(o => o.id !== 'source-horizontal-projection'), before.objects)
      assert.deepEqual(after.assertions.filter(a => !a.id.startsWith('source-horizontal-projection')), before.assertions)
      assert.deepEqual(after.source.inventory.filter(i => i.id !== 'source-horizontal-projection'), before.source.inventory)
    } else {
      assert.deepEqual(after.objects.filter(o => o.type === 'text'), before.objects.filter(o => o.type === 'text'))
      const segments = after.objects.filter(o => o.type === 'segment')
      assert.equal(segments.length, 10)
      assert.deepEqual(segments.filter(o => o.from[1] === o.to[1]).map(o => [o.from, o.to]),
        [0, 1, 2].map(y => [[0, y], [8, y]]))
      assert.deepEqual(segments.filter(o => o.from[0] === o.to[0]).map(o => [o.from, o.to]),
        [1, 2, 3, 4, 5, 6, 7].map(x => [[x, 0], [x, 2]]))
      assert.deepEqual(after.objects.filter(o => o.type !== 'text' && o.type !== 'segment'), [])
    }
    const file = path.join(work, `${id}-parent-authorized-region-leaf-diff.json`)
    const changes = leafDiff(before, after)
    save(file, { id, before: old, after: record(specFile), changes })
    return { id, before_sha256: old.sha256, after_sha256: sha(specFile), diff: record(file), changed_leaves: changes.length }
  })
  save(path.join(work, 'three-region-exact-diff-receipt.json'), { diffs, other_frozen_files_byte_identical: true })
  console.log(JSON.stringify(diffs))

  fs.mkdirSync(temp, { recursive: true })
  const result = spawnSync(executable, args, { cwd: repo, encoding: 'utf8', maxBuffer: 16 * 1024 * 1024,

  let failure = result.error?.message ?? (result.status === 0 ? null : `Exit ${result.status}`)
  try { check() } catch (error) { failure = error.message }
  save(resultFile, { at: new Date().toISOString(), argv: [executable, ...args],
    status: result.status, pass: !failure, failure, stdout: result.stdout, stderr: result.stderr,
    outputs: outputFiles(id).filter(fs.existsSync).map(record) })
  process.stdout.write(result.stdout ?? '')
  process.stderr.write(result.stderr ?? '')
  if (failure) process.exitCode = result.status || 1
