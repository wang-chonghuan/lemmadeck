const snapshotFile = async (absolute, destination, expected = null) => {
  const bytes = await fs.readFile(absolute);
  const sha256 = hash(bytes);
  if (expected && sha256 !== expected) throw new Error('Reviewed source bytes changed: ' + absolute);
  await fs.mkdir(path.dirname(destination), { recursive: true });
  await fs.copyFile(absolute, destination);
  if (hash(await fs.readFile(destination)) !== sha256) throw new Error('Snapshot copy changed bytes');
  return { path: absolute, sha256, snapshot_path: destination, copy_hash_verified: true };
};
