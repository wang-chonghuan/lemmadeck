const copyExact = (sourcePath, targetPath, expectedBytes) => {
  fs.mkdirSync(path.dirname(targetPath), { recursive: true });
  if (!fs.existsSync(targetPath)) {
    fs.copyFileSync(sourcePath, targetPath, fs.constants.COPYFILE_EXCL);
  }

};
const leafDifferences = (before, after, pointer = "") => {
  if (JSON.stringify(before) === JSON.stringify(after)) return [];
  if (Array.isArray(before) && Array.isArray(after)) {
    if (before.length !== after.length) return [{ pointer, before, after }];
    return before.flatMap((value, index) =>
      leafDifferences(value, after[index], `${pointer}/${index}`),
    );
  }
  if (before !== null && after !== null && typeof before === "object" && typeof after === "object") {
    const keys = new Set([...Object.keys(before), ...Object.keys(after)]);
    return [...keys].flatMap((key) =>
      leafDifferences(before[key], after[key], `${pointer}/${key}`),
    );
  }
  return [{ pointer, before, after }];
};
