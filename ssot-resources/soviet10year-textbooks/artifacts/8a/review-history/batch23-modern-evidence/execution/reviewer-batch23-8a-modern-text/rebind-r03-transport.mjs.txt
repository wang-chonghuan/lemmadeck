assert.equal(after.section_breaks[2].after, after.prose[8].text);
assert.deepEqual(after.prose, before.prose);

// Exact literal substitution proves that every other byte, not merely every JSON value, is retained.
const oldLiteral = `"before": ${JSON.stringify(before.section_breaks[2].before)}`;
const newLiteral = `"before": ${JSON.stringify(after.section_breaks[2].before)}`;
const oldString = oldCandidateBytes.toString("utf8");
assert.ok(oldString.includes(oldLiteral));
assert.equal(oldString.indexOf(oldLiteral), oldString.lastIndexOf(oldLiteral));
assert.ok(Buffer.from(oldString.replace(oldLiteral, newLiteral)).equals(currentBytes));

const preserveCopy = (sourcePath, targetPath, bytes) => {
  fs.mkdirSync(path.dirname(targetPath), { recursive: true });
  assert.ok(!fs.existsSync(targetPath), "Do not overwrite transport history.");
  fs.copyFileSync(sourcePath, targetPath, fs.constants.COPYFILE_EXCL);
  assert.ok(read(targetPath).equals(bytes));
};
