import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import { spawnSync } from "node:child_process";
import { fileURLToPath } from "node:url";

const own = path.dirname(fileURLToPath(import.meta.url));
const repo = "/Users/yong/.codex/worktrees/d9b1/lemmadeck";
const sha = bytes => crypto.createHash("sha256").update(bytes).digest("hex");
const record = file => ({ path: file, sha256: sha(fs.readFileSync(file)), bytes: fs.statSync(file).size });
const file = path.join(own, "observe-changed-regions.mjs");
const argv = ["node", file, "resume-after-anonymous-key-diagnosis"];
const prefix = path.join(own, "changed-region-resume");
const write = (suffix, bytes) => fs.writeFileSync(`${prefix}.${suffix}`, bytes, { flag: "wx" });
if (fs.existsSync(`${prefix}.started.json`)) throw Error("Resume wrapper already attempted.");
write("started.json", `${JSON.stringify({
  argv, cwd: repo, script: record(file),
  preserved_failed_command: record(path.join(own, "changed-region-observe.command.json")),
  preserved_failed_script: record(path.join(own, "observe-changed-regions.rejected-anonymous-svg-key.mjs")),
  diagnosis: "Global visible-node index was an unstable anonymous-path key after a new named line. Use per-tag anonymous ordinals; no SVG/candidate changes.",
  do_not_repeat_fig01_capture_or_view: true,
}, null, 2)}\n`);
const result = spawnSync(argv[0], argv.slice(1), { cwd: repo, maxBuffer: 16 * 1024 * 1024 });
write("stdout.txt", result.stdout ?? Buffer.alloc(0));
write("stderr.txt", result.stderr ?? Buffer.alloc(0));
write("command.json", `${JSON.stringify({
  argv, cwd: repo, script: record(file), exit_code: result.status, error: result.error?.message ?? null,
  stdout: record(`${prefix}.stdout.txt`), stderr: record(`${prefix}.stderr.txt`),
  output: fs.existsSync(path.join(own, "changed-region-display-evidence.json")) ? record(path.join(own, "changed-region-display-evidence.json")) : null,
}, null, 2)}\n`);
process.stdout.write(result.stdout ?? "");
process.stderr.write(result.stderr ?? "");
if (result.status !== 0) process.exitCode = result.status ?? 1;
