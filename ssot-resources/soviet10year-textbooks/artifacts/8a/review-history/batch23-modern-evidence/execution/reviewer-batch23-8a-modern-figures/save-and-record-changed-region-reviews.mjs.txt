    assert(record(item.path).sha256 === item.sha256, `Source/template/unchanged candidate drift: ${item.path}`);
    if (item.frozen_path) assert(record(item.frozen_path).sha256 === item.sha256, "Frozen copy drift.");
  }
  for (const candidate of inputs.candidates) {
    for (const item of [candidate.spec, candidate.svg, candidate.render]) {
      assert(record(item.path).sha256 === item.sha256 && record(item.frozen_path).sha256 === item.sha256, "Changed candidate drift.");
    }
  }
  for (const capture of displays.captures) for (const region of capture.regions) {
    assert(record(region.screenshot.path).sha256 === region.screenshot.sha256, "Actual region screenshot drift.");
  }
}
