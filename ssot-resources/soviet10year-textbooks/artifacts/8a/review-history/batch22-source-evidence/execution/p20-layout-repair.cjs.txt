const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const assert = require('node:assert/strict');
const { spawnSync } = require('node:child_process');
const base = '.intentfold/tickets/STEMROBIN-182/tmp';
const root = `${base}/batch22-source-8a`;
const work = `${base}/batch22-work-8a`;
const page = `${root}/8a/pages/0020`;
const history = `${work}/history/p0020-before-caption-row-repair`;
const sha = (file) => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
const read = (file) => JSON.parse(fs.readFileSync(file, 'utf8'));
const write = (file, value) => fs.writeFileSync(file, `${JSON.stringify(value, null, 2)}\n`, { flag: 'wx' });
const files = (dir) => fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))
  .flatMap((item) => item.isDirectory() ? files(`${dir}/${item.name}`) : [`${dir}/${item.name}`]);
const diffs = (a, b, pointer = '') => {
  if (JSON.stringify(a) === JSON.stringify(b)) return [];
  if (a && b && typeof a === 'object' && typeof b === 'object' && Array.isArray(a) === Array.isArray(b)) {
    return [...new Set([...Object.keys(a), ...Object.keys(b)])].flatMap((key) =>
      diffs(a[key], b[key], `${pointer}/${key.replace(/~/g, '~0').replace(/\//g, '~1')}`));
  }
  return [{ path: pointer, before: a === undefined ? null : a, after: b === undefined ? null : b }];
};

if (process.argv[2] === 'archive') {
  assert(!fs.existsSync(history), 'Repair history already exists; do not overwrite');
  fs.mkdirSync(history, { recursive: true });
  const originals = [
    ...['page.md', 'page.json', 'audit.json'].map((name) => `${page}/${name}`),
    `${work}/evidence/finalize-p0020.json`,
    `${work}/evidence/prepare-p0020.json`,
    `${work}/8a/pages/0020/layout.json`,
    `${work}/8a/pages/0020/page.template.md`,
    `${work}/source-observations.json`,
    `${work}/raw-progress.json`,
  ];
  const archived = originals.map((file) => {
    const destination = `${history}/${path.relative(base, file)}`;
    fs.mkdirSync(path.dirname(destination), { recursive: true });
    fs.copyFileSync(file, destination, fs.constants.COPYFILE_EXCL);
    assert.equal(sha(file), sha(destination));
    return { path: file, snapshot: destination, sha256: sha(file) };
  });
  const immutable = [
    ...files(`${root}/8a/pages`).filter((file) => !file.startsWith(`${page}/`)),
    ...files(`${work}/8a/pages/0020`).filter((file) => file.endsWith('.png') || file.endsWith('layout.json')),
  ].map((file) => ({ path: file, sha256: sha(file) }));
  write(`${history}/manifest.json`, { archived, immutable });
  console.log(JSON.stringify({ archived_files: archived.length, history }, null, 2));
} else if (process.argv[2] === 'finalize') {
  const manifest = read(`${history}/manifest.json`);
  const snapshot = (file) => manifest.archived.find((item) => item.path === file).snapshot;
  const oldMd = fs.readFileSync(snapshot(`${page}/page.md`), 'utf8');
  const expected = oldMd.replace('<!-- cap 图 2 -->\n图2\n\n', '')
    .replace('\n<!-- foot -->', '\n<!-- cap 图 2 samerow -->\n图2\n\n<!-- foot -->');
  assert.equal(fs.readFileSync(`${page}/page.md`, 'utf8'), expected, 'Only the authorized caption move is allowed');
  const record = `${work}/evidence/finalize-p0020-caption-row-repair.json`;
  assert(!fs.existsSync(record), 'Repaired finalize already attempted');
  const argv = ['.claude/skills/ld-s10y-lesson/.venv/bin/python',
    '.claude/skills/ld-s10y-lesson/tools/p2c.py', 'finalize', '--book', '8a', '--page', '20',
    '--root', root, '--work', work];
  const started = new Date().toISOString();
  const result = spawnSync(argv[0], argv.slice(1), { encoding: 'utf8' });
  write(record, { argv, cwd: process.cwd(), started_at: started, completed_at: new Date().toISOString(),
    exit_code: result.status, signal: result.signal, execution_error: result.error ? String(result.error) : null,
    stdout: result.stdout || '', stderr: result.stderr || '', execution_count: 1,
    supersedes_failed_command: `${history}/batch22-work-8a/evidence/finalize-p0020.json` });
  const before = read(snapshot(`${page}/page.json`));
  const after = read(`${page}/page.json`);
  const expectedBlocks = structuredClone(before.blocks);
  const caption = expectedBlocks.splice(8, 1)[0];
  assert.equal(caption.kind, 'cap');
  caption.samerow = true;
  expectedBlocks.splice(9, 0, caption);
  assert.deepEqual(after.blocks, expectedBlocks, 'Unexpected textual, mathematical or box change');
  const expectedMeta = { ...before.meta, printed_lines: 19 };
  assert.deepEqual(after.meta, expectedMeta);
  const immutable = manifest.immutable.map((item) => ({ ...item, current_sha256: sha(item.path) }));
  assert(immutable.every((item) => item.sha256 === item.current_sha256), 'Another page or original image changed');
  const markdownDiff = spawnSync('diff', ['-u', snapshot(`${page}/page.md`), `${page}/page.md`], { encoding: 'utf8' });
  fs.writeFileSync(`${work}/p0020-caption-row-repair.markdown.diff`, markdownDiff.stdout, { flag: 'wx' });
  write(`${work}/p0020-caption-row-repair-delta.json`, {
    schema: 'raw-page-layout-only-repair-v1',
    history, command_record: record,
    changed_json_files: ['page.json', 'audit.json'].map((name) => ({
      path: `${page}/${name}`, before_sha256: sha(snapshot(`${page}/${name}`)), after_sha256: sha(`${page}/${name}`),
      full_field_delta: diffs(read(snapshot(`${page}/${name}`)), read(`${page}/${name}`)),
    })),
    markdown: { before_sha256: sha(snapshot(`${page}/page.md`)), after_sha256: sha(`${page}/page.md`),
      complete_diff: `${work}/p0020-caption-row-repair.markdown.diff` },
    immutable_files: immutable,
    text_formula_boxes_and_image_bytes_unchanged: true,
    finalization_exit_code: result.status,
  });
  console.log(result.stdout || '');
  if (result.stderr) console.error(result.stderr);
  if (result.error || result.status !== 0) process.exit(1);
} else {
  throw new Error('Use archive or finalize only');
}
