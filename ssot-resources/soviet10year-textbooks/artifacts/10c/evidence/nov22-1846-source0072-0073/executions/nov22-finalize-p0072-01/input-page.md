<!-- ex 217 -->

<!-- ex 218 samerow -->

<!-- ex 219 -->

<!-- ex 224 -->

<!-- ex 225 -->

<!-- ex 226 -->

<!-- ex 227 -->

<!-- ex 229 -->

230. $\operatorname{arctg}0+\operatorname{arcctg}0$.

$\operatorname{arcctg}'x=-\frac{1}{1+x^2}$.
